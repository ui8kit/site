    Product Grid

    Products as cards in a grid, a list or a rail that scrolls sideways.

    Install

    npx ui8kit add product-grid

    Pulls in: box, grid, product-card, section-frame, utils

    Code

    Svelte

    Point the CLI at this runtime once, then add the item:

    npx ui8kit init --yes --framework svelte --dir src --registry-url https://registry.buildy.tw/svelte --strict-cdn
    npx ui8kit add product-grid

/**
 * A section with a heading and the products as cards in a grid, as a list of horizontal cards, or as a rail that
 * scrolls sideways. Axes: layout (grid, list, rail), columns, ratio, rating, action, tone, density.
 */
import { type ReactNode } from "react";
import { Box } from "../primitives/box/box";
import { Grid } from "../primitives/grid/grid";
import { ProductCard } from "../blocks/product-card";
import { SectionFrame } from "../components/section-frame";
import type { Product, ProductLabels } from "./product-card.types";

export type ProductGridProps = {
  id: string;
  title: string;
  products: Product[];
  labels: ProductLabels;
  badges: Record<string, string>;
  layout?: "grid" | "list" | "rail";
  cols?: "1-2" | "1-3" | "1-4";
  ratio?: "4/3" | "1/1" | "3/4";
  rating?: "show" | "hide";
  action?: "button" | "none";
  tone?: "default" | "muted";
  density?: "roomy" | "default" | "compact";
  onAdd?: (id: string) => void;
};

export function ProductGrid({
  id,
  title,
  products,
  labels,
  badges,
  layout = "grid",
  cols = "1-3",
  ratio = "4/3",
  rating = "show",
  action = "button",
  tone = "default",
  density = "default",
  onAdd,
}: ProductGridProps) {
  return (
    <SectionFrame id={id} title={title} tone={tone} density={density}>
      {((layout as string) === "rail") ? (
        <Box className="flex snap-x gap-4 overflow-x-auto pb-4" data-testid="items">
          {(products ?? []).map((p) => (
            <Box key={p.id} className="w-64 shrink-0 snap-start">
              <ProductCard
                product={p}
                labels={labels}
                badges={badges}
                layout={(((layout as string) === "list") ? "horizontal" : "vertical")}
                ratio={ratio}
                rating={rating}
                action={action}
                density={(((density as string) === "compact") ? "compact" : "default")}
                onAdd={() => onAdd?.(p.id)}
              />
            </Box>
          ))}
        </Box>
      ) : (
        <Grid cols={(((layout as string) === "list") ? "1" : cols)} className="gap-4" data-testid="items">
          {(products ?? []).map((p) => (
            <ProductCard key={p.id}
              product={p}
              labels={labels}
              badges={badges}
              layout={(((layout as string) === "list") ? "horizontal" : "vertical")}
              ratio={ratio}
              rating={rating}
              action={action}
              density={(((density as string) === "compact") ? "compact" : "default")}
              onAdd={() => onAdd?.(p.id)}
            />
          ))}
        </Grid>
      )}
    </SectionFrame>
  );
}
