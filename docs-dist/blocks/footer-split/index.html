    Footer Split

    The brand with figures and a sitemap link over link-group cards; or a split beside a picture with points, link columns, a newsletter card, the contacts or the community channels.

                    {(((layout as string) === "newsletter") ? labels.loopTitle : (((layout as string) === "channels") ? labels.joinTitle : labels.title))}
                  </Text>
                  <Text size="sm" class="text-muted-foreground">
                    {(((layout as string) === "newsletter") ? labels.loopText : (((layout as string) === "channels") ? labels.joinText : labels.text))}
                  </Text>
                </Stack>
              )}
              {((layout as string) === "points") ? (
                <FooterPoints extras={extras} title={labels.points} />
              ) : null}
              {((layout as string) === "newsletter") ? (
                <FooterSignupCard id={id} labels={labels} />
              ) : null}
              {(((layout as string) === "links") || ((layout as string) === "newsletter")) ? (

                </Text>
