    Delivery Info

    How delivery works: the facts, then the zones.

    Install

    npx ui8kit add delivery-info

    Pulls in: badge, box, feature-grid, grid, group, lucide-icon, section-frame, stack, text, title, utils, zone-table

import { Text } from "../primitives/text/text";
import { Title } from "../primitives/title/title";
import { SectionFrame } from "../components/section-frame";

import type { DeliveryLabels } from "./delivery-info.types";
import type { Feature } from "./feature-grid.types";
import type { Zone } from "./zone-table.types";

export type DeliveryInfoProps = {
  id: string;
  title: string;
  lead?: string;
  facts: Feature[];
  zones: Zone[];
  labels: DeliveryLabels;
  layout?: "full" | "facts" | "split";
  tone?: "default" | "muted";
  density?: "default" | "compact";
};

export function DeliveryInfo({
  id,
  title,
  lead = "",
  facts,
  zones,
  labels,
  layout = "full",
  tone = "default",
  density = "default",
}: DeliveryInfoProps) {
  return (
    <SectionFrame id={id} title={title} lead={lead} tone={tone} density={density}>
      <Box

        )}
      >
        <Grid cols={(((layout as string) === "split") ? "1-2" : "1-4")} className="gap-4" data-testid="facts">
          {(facts ?? []).map((f) => (
            <Stack key={f.id} gap="4" className="rounded bg-card inset-ring inset-ring-border p-6" data-testid="fact">
              <Box className="flex size-10 shrink-0 items-center justify-center rounded bg-muted">
                <LucideIcon name={f.icon} size={20} />
              </Box>
              <Stack gap="2">
                <Text size="sm" weight="semibold">

              className={
                (layout as string) === "split" ? "flex flex-col gap-4"
                : "grid grid-cols-1 gap-4 md:grid-cols-3"
              }
            >
              {(zones ?? []).map((z) => (
                <Stack key={z.id}
                  gap="2"
                  className={
                    (layout as string) === "split" ? "border-b border-border pb-4"
                    : "rounded bg-card inset-ring inset-ring-border p-6"
                  }
                  data-testid="zone"
                >
                  <Group gap="4" align="baseline" justify="between">
                    <Text size="sm" weight="semibold">
                      {z.name}
                    </Text>
                    <Badge variant="secondary" size="default">
                      {z.time}
