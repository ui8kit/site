    Page Intro

    The head of an inner page: the heading, one sentence and an optional action.

    Install

    npx ui8kit add page-intro

    Pulls in: action-group, section-frame, utils

 * The head of an inner page: the heading (h1 on a page of its own), one sentence and, optionally, an action.
 * Axes: align (start, center), tone, density.
 */
import { type ReactNode } from "react";
import { ActionGroup } from "../components/action-group";
import { SectionFrame } from "../components/section-frame";

export type PageIntroProps = {
  id: string;
  title: string;
  text: string;
  cta?: string;
  href?: string;
  level?: 1 | 2;
  align?: "start" | "center";
  tone?: "default" | "muted";
  density?: "default" | "compact";
};

export function PageIntro({
  id,
  title,
  text,
  cta,
  href,
  level = 1,
  align = "start",
  tone = "default",
  density = "default",
}: PageIntroProps) {
  return (
    <SectionFrame id={id} title={title} lead={text} heading={{ level }} align={align} tone={tone} density={density}>
      {cta ? (
        <ActionGroup primary={cta} primaryHref={href} align={align} />
      ) : null}
    </SectionFrame>
  );
}
