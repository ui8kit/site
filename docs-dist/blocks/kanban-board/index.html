    Kanban Board

    Work items in lanes with a count; cards with a tag and a person. Moving cards is a port contract (SortableList).

    Install

    npx ui8kit add kanban-board

    Pulls in: grid, group, list, pill, section-frame, stack, text, title, utils

import { type ReactNode } from "react";
import { Grid } from "../primitives/grid/grid";
import { Group } from "../primitives/group/group";
import { List } from "../primitives/list/list";

import type { Lane, BoardCard } from "./kanban-board.types";

export type KanbanBoardProps = {
  id: string;
  title: string;
  lanes: Lane[];
  cards: BoardCard[];
  countLabel: string;
  card?: "simple" | "detail";
  cols?: "2" | "3";
  tone?: "default" | "muted";
  density?: "default" | "compact";
};

export function KanbanBoard({
  id,
  title,
  lanes,
  cards,
  countLabel,
  card = "detail",
  cols = "3",
  tone = "default",
  density = "default",
}: KanbanBoardProps) {
  return (
    <SectionFrame id={id} title={title} tone={tone} density={density}>
      <Grid cols={(((cols as string) === "2") ? "1-2" : "1-3")} className="items-start gap-4" data-testid="lanes">
        {(lanes ?? []).map((lane) => (
          <Stack key={lane.id} gap="4" align="stretch" className="rounded bg-muted p-4" data-testid="lane">
            <Group gap="2" align="center" justify="between">
              <Title as={3} size="base" weight="bold">
                {lane.title}
              </Title>
              <Text size="xs" className="text-muted-foreground">
                {tpl(countLabel, (cards.filter((i) => !lane.key || i["lane"] === lane.key)?.length ?? 0))}
              </Text>
            </Group>
            <List className="m-0 flex list-none flex-col gap-2 p-0">
              {(cards.filter((i) => !lane.key || i["lane"] === lane.key) ?? []).map((c) => (
                <ListItem key={c.id}
                  className={cn(
                    "flex flex-col rounded bg-card inset-ring inset-ring-border",
                    (((density as string) === "compact") ? "gap-2 p-2" : "gap-2 p-4")
                  )}
                  data-testid="item"
                >
                  <Text size="sm" weight="semibold">
                    {c.title}
                  </Text>
                  {((card as string) === "detail") ? (
                    <Group gap="2" align="center" justify="between" className="flex-wrap">
                      <Pill tone="blue" icon="tag">
                        {c.tag}
                      </Pill>
                      <Text size="xs" className="text-muted-foreground">
                        {c.who}
                      </Text>
                    </Group>
                  ) : null}
                </ListItem>
              ))}
            </List>
          </Stack>
        ))}
      </Grid>
    </SectionFrame>
  );
}
