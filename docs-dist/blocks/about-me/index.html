    About Me

    A personal about section beside a portrait: key facts or skills.

    Install

    npx ui8kit add about-me

    Pulls in: action-group, error-page, form, group, list, media-frame, product-info, section-frame, stack, text, utils

    Code

    Svelte

    Point the CLI at this runtime once, then add the item:

    npx ui8kit init --yes --framework svelte --dir src --registry-url https://registry.buildy.tw/svelte --strict-cdn
    npx ui8kit add about-me

/**
 * A personal about section beside a portrait: a badge, the heading, a lead, then key facts and a list of

 * mediaSide, tone.

import { type ReactNode } from "react";
import { Progress } from "../primitives/form/progress";
import { Group } from "../primitives/group/group";
import { List } from "../primitives/list/list";
import { ListItem } from "../primitives/list/list-item";
import { Stack } from "../primitives/stack/stack";
import { Text } from "../primitives/text/text";
import { ActionGroup } from "../components/action-group";
import { MediaFrame } from "../components/media-frame";
import { SectionFrame } from "../components/section-frame";
import { tpl } from "../utils";
import type { Skill, AboutMeLabels } from "./about-me.types";
import type { Spec } from "./product-info.types";
import type { LinkActions } from "./error-page.types";

export type AboutMeProps = {
  id: string;
  title: string;
  lead?: string;
  badge?: string;
  level?: 1 | 2;
  facts: Spec[];
  skills: Skill[];
  labels: AboutMeLabels;
  actions: LinkActions;
  layout?: "facts" | "skills";
  mediaSide?: "start" | "end";
  tone?: "default" | "muted";
};

export function AboutMe({
  id,
  title,
  lead = "",
  badge = "",
  level = 1,
  facts,
  skills,
  labels,
  actions,
  layout = "facts",
  mediaSide = "start",
  tone = "default",
}: AboutMeProps) {
  return (
    <SectionFrame
      id={id}
      title={title}
      lead={lead}
      heading={{ badge, level }}
      layout="split"
      media={{ show: true, side: mediaSide }}
      tone={tone}
      density="default"
      aside={(
        <MediaFrame alt={title} ratio="3/4" />
      )}
    >
      {((layout as string) === "facts") ? (
        <List tag="ul" className="grid grid-cols-1 gap-4 sm:grid-cols-2" data-testid="facts">
          {(facts ?? []).map((f) => (
            <ListItem key={f.id} className="rounded bg-card inset-ring inset-ring-border flex flex-col gap-0 p-4" data-testid="fact">
              <Text size="xs" className="text-muted-foreground">
                {f.label}
              </Text>
              <Text size="sm" weight="semibold">
                {f.value}
              </Text>
            </ListItem>
          ))}
        </List>
      ) : (
        <Stack gap="4" align="stretch" data-testid="skills">
          {(skills ?? []).map((s) => (
            <Stack key={s.id} gap="2" align="stretch" data-testid="skill">
              <Group gap="4" justify="between">
                <Text size="sm" weight="semibold">
                  {s.label}
                </Text>
                <Text size="xs" className="text-muted-foreground">
                  {tpl(labels.level, s.level)}
                </Text>
              </Group>
              <Progress value={(s.level == null ? undefined : String(s.level))} max="100" aria-label={s.label} />
            </Stack>
          ))}
        </Stack>
      )}
      <ActionGroup
        primary={actions.cta}
        primaryHref={actions.href}
        secondary={((actions.mode === "two") ? actions.secondary : undefined)}
        secondaryHref={actions.secondaryHref}
        align="start"
      />
    </SectionFrame>
  );
}
