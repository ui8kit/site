    Cta Proof

    A call to action with logos, features, figures, icon rows or delivery zones; centered or beside a picture.

    Install

    npx ui8kit add cta-proof

    Pulls in: action-group, cta-proof-items, error-page, feature-grid, logo-cloud, media-frame, section-frame, stats-band, text, utils, zone-table

    Code

    Svelte

    Point the CLI at this runtime once, then add the item:

    npx ui8kit init --yes --framework svelte --dir src --registry-url https://registry.buildy.tw/svelte --strict-cdn
    npx ui8kit add cta-proof

import { type ReactNode } from "react";
import { Text } from "../primitives/text/text";

export type CtaProofProps = {
  id: string;

export function CtaProof({
  id,
  title,
  lead = "",
  badge = "",
  actions,
  content,
  note = "",
  layout = "center",
  proof = "stats",
  mediaSide = "end",
  tone = "default",
}: CtaProofProps) {
  return (
    <SectionFrame
      id={id}
      title={title}
      lead={lead}
      heading={{ badge }}
      tone={tone}
      density="default"
      align={(((layout as string) === "center") ? "center" : "start")}
      layout={(((layout as string) === "center") ? "stack" : "split")}
      media={{ show: ((layout as string) === "split"), side: mediaSide }}
      aside={(
        <MediaFrame alt={title} ratio="4/3" />
      )}
    >
      <CtaProofItems content={content} proof={proof} center={((layout as string) === "center")} />
      <ActionGroup
        primary={actions.cta}
        primaryHref={actions.href}
        secondary={((actions.mode === "two") ? actions.secondary : undefined)}
        secondaryHref={actions.secondaryHref}
        align={(((layout as string) === "center") ? "center" : "start")}
      />
      {note ? (
        <Text size="xs" className="text-muted-foreground" data-testid="cta-note">
          {note}
        </Text>
      ) : null}
    </SectionFrame>
  );
}
