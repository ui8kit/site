    Week Menu

    Meal delivery as a week: one column per day with its meal.

    Install

    npx ui8kit add week-menu

    Pulls in: box, compose, dish-showcase, empty-state, group, inline, kit-box, kit-text, media-frame, restaurant-menu, section-frame, stack, text, utils

/**
 * Meal delivery as a week: one column per day (its name and delivery note) with the meal of that day as a small
 * card (picture, name, weight, price). Axes: media (image, none), tone.
 */
import { Fragment, type ReactNode } from "react";
import { Box, Group, Inline, Stack, Text } from "../compose";
import { EmptyState } from "../components/empty-state";
import { MediaFrame } from "../components/media-frame";
import { SectionFrame } from "../components/section-frame";
import { fmtMoney } from "../utils";
import type { Dish, FoodLabels } from "./dish-showcase.types";
import type { DishSection } from "./restaurant-menu.types";

export type WeekMenuProps = {
  id: string;
  title: string;
  lead?: string;
  dishes: Dish[];
  sections: DishSection[];
  empty?: string;
  labels: FoodLabels;
  media?: "image" | "none";
  tone?: "default" | "muted";
  landmarkLabel?: string;
};

export function WeekMenu({
  id,
  title,
  lead = "",
  dishes,
  sections,
  empty = "",
  labels,
  media = "image",
  tone = "default",
  landmarkLabel,
}: WeekMenuProps) {
  return (
    <SectionFrame id={id} title={title} lead={lead} tone={tone} landmarkLabel={landmarkLabel}>
      {((sections?.length ?? 0) === 0) ? (
        <EmptyState icon="calendar" title={empty} />
      ) : (
        <Box className="grid grid-cols-1 gap-4 sm:grid-cols-2 lg:grid-cols-5" data-testid="items">
          {(sections ?? []).map((s) => (
            <Stack key={s.id} gap="2" className="border-t-2 border-primary pt-4" data-testid="day">
              <Text size="sm" weight="semibold">
                {s.title}
              </Text>
              <Text size="xs" className="text-muted-foreground">
                {s.note}
              </Text>
              {(dishes ?? []).map((d) => (
                <Fragment key={d.id}>
                  {(d.tag === s.key) ? (
                    <Stack gap="2" className="rounded bg-card inset-ring inset-ring-border p-4" data-testid="item">
                      {((media as string) === "image") ? (
                        <MediaFrame alt={d.name} ratio="4/3" />
                      ) : null}
                      <Text size="sm" weight="semibold">
                        {d.name}
                      </Text>
                      <Group gap="2" align="baseline" justify="between">
                        <Text size="xs" className="text-muted-foreground">
                          {d.weight}
                        </Text>
                        <Inline className="text-sm font-bold">
                          {fmtMoney(d.price, d.currency)}
                        </Inline>
                      </Group>
                    </Stack>
                  ) : null}
                </Fragment>
              ))}
            </Stack>
          ))}
        </Box>
      )}
    </SectionFrame>
  );
}
