    Feature Showcase

    Cards with a picture on top and a read link, or with a muted glyph panel on top like an index of categories, in three or four columns.

    Install

    npx ui8kit add feature-showcase

    Pulls in: box, card, feature-grid, link, lucide-icon, media-frame, section-frame, text, title, utils

/**
 * Features as cards under a centered heading. Axes: item (picture: a picture on top, then the title, the text

import { type ReactNode } from "react";
import { Box } from "../primitives/box/box";
import { Card } from "../primitives/card/card";
import { CardContent } from "../primitives/card/card-content";
import { Link } from "../primitives/link/link";
import { LucideIcon } from "../primitives/lucide-icon/lucide-icon";
import { Text } from "../primitives/text/text";
import { Title } from "../primitives/title/title";
import { MediaFrame } from "../components/media-frame";
import { SectionFrame } from "../components/section-frame";
import type { Feature } from "./feature-grid.types";

export type FeatureShowcaseProps = {
  id: string;
  title: string;
  lead?: string;
  features: Feature[];
  more: string;
  item?: "picture" | "panel";
  cols?: "3" | "4";
  tone?: "default" | "muted";
};

export function FeatureShowcase({
  id,
  title,
  lead = "",
  features,
  more,
  item = "picture",
  cols = "3",
  tone = "default",
}: FeatureShowcaseProps) {
  return (
    <SectionFrame id={id} title={title} lead={lead} tone={tone} align="center">
      <Box
        className={
          (cols as string) === "4" ? "grid grid-cols-1 gap-6 text-start sm:grid-cols-2 lg:grid-cols-4"
          : "grid grid-cols-1 gap-6 text-start sm:grid-cols-2 lg:grid-cols-3"
        }
      >
        {(features ?? []).map((f) => (
          <Card key={f.id} variant="default" className="h-full gap-4 overflow-hidden" data-testid="showcase-card">
            {((item as string) === "picture") ? (
              <MediaFrame alt={f.title} ratio="16/9" />
            ) : (
              <Box
                className="flex aspect-video items-center justify-center rounded bg-muted text-foreground"
                data-testid="showcase-panel"
              >
                <LucideIcon name={f.icon} size={32} />
              </Box>
            )}
            <CardContent className="flex flex-1 flex-col gap-2">
              <Title as={3} size="base" weight="semibold">
                {f.title}
              </Title>
              <Text size="sm" className="text-muted-foreground">
                {f.text}
              </Text>
              {((item as string) === "picture") ? (
                <Link href="#/" size="sm" className="mt-auto pt-2" data-testid="read">
                  {more}
                  <LucideIcon name="arrow-right" size={16} />
                </Link>
              ) : null}
            </CardContent>
          </Card>
        ))}
      </Box>
    </SectionFrame>
  );
}
