    Icon Wall

    Many small features as icons with a name: tiles or chips in three to six columns.

    Install

    npx ui8kit add icon-wall

    Pulls in: box, feature-grid, grid, lucide-icon, section-frame, text, utils

/**
 * A dense grid of small tiles: an icon and a short label. Axes: columns (3, 4, 6), item (tile: icon above the
 * label, chip: icon beside it), tone, density.
 */
import { type ReactNode } from "react";
import { Box } from "../primitives/box/box";
import { Grid } from "../primitives/grid/grid";
import { LucideIcon } from "../primitives/lucide-icon/lucide-icon";
import { Text } from "../primitives/text/text";
import { SectionFrame } from "../components/section-frame";
import { cn, fitCols } from "../utils";
import type { Feature } from "./feature-grid.types";

export type IconWallProps = {
  id: string;
  title: string;
  features: Feature[];
  cols?: "3" | "4" | "6";
  item?: "tile" | "chip";
  tone?: "default" | "muted";
  density?: "default" | "compact";
  landmarkLabel?: string;
};

export function IconWall({
  id,
  title,
  features,
  cols = "4",
  item = "tile",
  tone = "default",
  density = "default",
  landmarkLabel,
}: IconWallProps) {
  const askedCols =
    ((item as string) === "chip") && ((cols as string) === "6") ? "2-6"
    : (cols as string) === "2" ? "2"
    : (cols as string) === "3" ? "2-3"
    : (cols as string) === "4" ? "2-4"
    : "3-6";
  const wallCols = fitCols(askedCols, (features?.length ?? 0));
  return (
    <SectionFrame id={id} title={title} tone={tone} density={density} landmarkLabel={landmarkLabel}>
      <Grid cols={wallCols} className="gap-4" data-testid="items">
        {(features ?? []).map((f) => (
          <Box key={f.id}
            className={cn(
              "flex items-center gap-2",
              "rounded bg-card inset-ring inset-ring-border",
              (
                (item as string) === "tile" ? cn("flex-col justify-center text-center", (((density as string) === "compact") ? "p-4" : "p-6"))

              )

              {f.title}
            </Text>
          </Box>
        ))}
      </Grid>
    </SectionFrame>
  );
}
