    Icon Wall

    Many small features as icons with a name: tiles or chips in three to six columns.

    Install

    npx ui8kit add icon-wall

    Pulls in: box, feature-grid, grid, lucide-icon, section-frame, text, utils

  item?: "tile" | "chip";
  tone?: "default" | "muted";
  density?: "default" | "compact";

export function IconWall({
  id,
  title,
  features,
  cols = "4",
  item = "tile",
  tone = "default",
  density = "default",
}: IconWallProps) {
  const askedCols =
    ((item as string) === "chip") && ((cols as string) === "6") ? "2-6"
    : (cols as string) === "2" ? "2"
