    Feature Tabs

    Features as tabs; the chosen one fills the panel beside or below the tab list.

    Install

    npx ui8kit add feature-tabs

    Pulls in: feature-grid, grid, lucide-icon, media-frame, section-frame, stack, switcher, switcher-item, text, title, utils

    Preview

    One product, many jobs

    Clear structure

    One heading per page, landmarks and a skip link come built in.

    Code

    Point the CLI at this runtime once, then add the item:

    npx ui8kit init --yes --framework svelte --dir src --registry-url https://registry.buildy.tw/svelte --strict-cdn
    npx ui8kit add feature-tabs

/**
 * Features behind a tab list: the chosen tab shows its text and a picture. The page keeps the chosen tab
 * (`active`, empty for the first) and updates it on the `select` event. Axes: panel (split: text beside the
 * picture, stack: picture below), tone, density.
 */
import { Fragment, type ReactNode } from "react";
import { Grid } from "../primitives/grid/grid";
import { LucideIcon } from "../primitives/lucide-icon/lucide-icon";
import { Stack } from "../primitives/stack/stack";
import { Text } from "../primitives/text/text";
import { Title } from "../primitives/title/title";
import { MediaFrame } from "../components/media-frame";
import { SectionFrame } from "../components/section-frame";
import { Switcher } from "../components/switcher";
import { SwitcherItem } from "../components/switcher-item";
import type { Feature } from "./feature-grid.types";

export type FeatureTabsProps = {
  id: string;
  title: string;
  features: Feature[];
  tabsLabel: string;
  active?: string;
  panel?: "split" | "stack";
  tone?: "default" | "muted";
  density?: "default" | "compact";
  landmarkLabel?: string;
  onSelect?: (id: string) => void;
};

export function FeatureTabs({
  id,
  title,
  features,
  tabsLabel,
  active = "",
  panel = "split",
  tone = "default",
  density = "default",
  landmarkLabel,
  onSelect,
}: FeatureTabsProps) {
  return (
    <SectionFrame id={id} title={title} tone={tone} density={density} landmarkLabel={landmarkLabel}>
      <Stack gap="6" align="stretch">
        <Switcher label={tabsLabel} className="w-fit max-w-full flex-wrap">
          {(features ?? []).map((f, i) => (
            <SwitcherItem key={f.id}
              active={((f.id === active) || (((active as string) === "") && (i === 0)))}
              onClick={() => onSelect?.(f.id)}
            >
              {f.title}
            </SwitcherItem>
          ))}
        </Switcher>
        {(features ?? []).map((f, i) => (
          <Fragment key={f.id}>
            {((f.id === active) || (((active as string) === "") && (i === 0))) ? (
              <Grid
                cols={(((panel as string) === "split") ? "1-2" : "1")}
                className="items-center gap-8"
                data-testid="tab-panel"
                role="tabpanel"
              >
                <Stack gap="4">
                  <LucideIcon name={f.icon} size={24} />
                  <Title as={3} size="2xl" weight="bold">
                    {f.title}
                  </Title>
                  <Text size="base" className="text-muted-foreground">
                    {f.text}
                  </Text>
                </Stack>
                <MediaFrame alt={f.title} ratio="4/3" />
              </Grid>
            ) : null}
          </Fragment>
        ))}
      </Stack>
    </SectionFrame>
  );
}

templ FeatureTabs(p FeatureTabsProps) {
	{{ vId := p.Id }}
	{{ vTitle := p.Title }}
	{{ vFeatures := p.Features }}
	{{ vTabsLabel := p.TabsLabel }}
	{{ vActive := uiutils.Def(p.Active, "") }}
	{{ vPanel := uiutils.Def(p.Panel, "split") }}
	{{ vTone := uiutils.Def(p.Tone, "default") }}
	{{ vDensity := uiutils.Def(p.Density, "default") }}

				}
			}
