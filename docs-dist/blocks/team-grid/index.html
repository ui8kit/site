    Team Grid

    The people of a team as cards, a minimal grid, detailed cards, rows beside a picture or a directory.

    Install

    npx ui8kit add team-grid

    Pulls in: box, media-frame, member-row, section-frame, stack, text, utils

    Code

    Svelte

    Point the CLI at this runtime once, then add the item:

    npx ui8kit init --yes --framework svelte --dir src --registry-url https://registry.buildy.tw/svelte --strict-cdn
    npx ui8kit add team-grid

/**
 * The people of a team: portrait cards, a minimal grid of small portraits, detailed cards with skills, rows

 * cols, tone, density.

import { Fragment, type ReactNode } from "react";
import { Box } from "../primitives/box/box";
import { Stack } from "../primitives/stack/stack";
import { Text } from "../primitives/text/text";
import { MediaFrame } from "../components/media-frame";
import { MemberRow } from "../components/member-row";
import { SectionFrame } from "../components/section-frame";
import { cn } from "../utils";
import type { Member } from "./team-grid.types";

export type TeamGridProps = {
  id: string;
  title: string;
  lead?: string;
  members: Member[];
  depts: Record<string, string>;
  layout?: "cards" | "minimal" | "detailed" | "list" | "directory";
  cols?: "1-3" | "1-4";
  tone?: "default" | "muted";
  density?: "default" | "compact";
};

export function TeamGrid({
  id,
  title,
  lead = "",
  members,
  depts,
  layout = "cards",
  cols = "1-4",
  tone = "default",
  density = "default",
}: TeamGridProps) {
  return (
    <SectionFrame
      id={id}
      title={title}
      lead={lead}
      tone={tone}
      density={density}
      layout={(((layout as string) === "list") ? "split" : "stack")}
      media={{ show: ((layout as string) === "list"), side: "start", align: "start" }}
      aside={(
        <MediaFrame alt={title} ratio="3/4" />
      )}
    >
      <Box
        className={
          (layout as string) === "minimal" ? "grid grid-cols-2 gap-8 md:grid-cols-4"
          : (layout as string) === "directory" ? "grid grid-cols-1 gap-4 md:grid-cols-2"
          : (layout as string) === "list" ? "flex flex-col gap-4"
          : (cols as string) === "1-3" ? "grid grid-cols-1 gap-4 sm:grid-cols-2 lg:grid-cols-3"
          : "grid grid-cols-1 gap-4 sm:grid-cols-2 md:grid-cols-4"
        }
        data-testid="items"
      >
        {(members ?? []).map((m) => (
          <Fragment key={m.id}>
            {((layout as string) === "minimal") ? (
              <Stack gap="2" align="center" className="text-center" data-testid="item">
                <MediaFrame alt={m.name} ratio="1/1" className={cn("size-16", "shrink-0")} />
                <Text size="sm" weight="semibold">
                  {m.name}
                </Text>
                <Text size="xs" className="text-muted-foreground">
                  {m.role}
                </Text>
              </Stack>
            ) : ((layout as string) === "cards") ? (
              <Stack gap="4" className="rounded bg-card inset-ring inset-ring-border p-4" data-testid="item">
                <MediaFrame alt={m.name} ratio="1/1" />
                <Stack gap="2">
                  <Stack gap="0">
                    <Text size="base" weight="semibold">
                      {m.name}
                    </Text>
                    <Text size="sm" className="text-muted-foreground">
                      {m.role}
                    </Text>
                  </Stack>
                  <Text size="sm" className="text-muted-foreground">
                    {m.bio}
                  </Text>
                </Stack>
              </Stack>
            ) : (
              <MemberRow member={m} depts={depts} layout={layout} />
            )}
          </Fragment>
        ))}
      </Box>
    </SectionFrame>
  );
}
