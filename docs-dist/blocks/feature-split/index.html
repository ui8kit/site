    Feature Split

    Features beside a picture: plain in two columns, as cards with actions, two figures over checked rows, or four metric cards with actions.

    Install

    npx ui8kit add feature-split

    Pulls in: action-group, error-page, feature-cards, feature-grid, feature-metric-cards, feature-pairs, feature-stat-rows, media-frame, section-frame, stats-band, utils

import { ActionGroup } from "../components/action-group";
import { FeatureCards } from "../components/feature-cards";

import type { Feature } from "./feature-grid.types";

export type FeatureSplitProps = {
  id: string;
  title: string;
  lead?: string;
  badge?: string;
  level?: 1 | 2;
  features: Feature[];
  stats: Stat[];
  actions: LinkActions;

export function FeatureSplit({
  id,
  title,
  lead = "",
  badge = "",
  level = 2,
  features,
  stats,
  actions,
  layout = "grid",
  mediaSide = "end",
  tone = "default",
}: FeatureSplitProps) {
  return (
    <SectionFrame
      id={id}
      title={title}
      lead={lead}
      heading={{ badge, level }}
      tone={tone}
      layout="split"
      media={{ show: true, side: mediaSide }}
      aside={(
        <MediaFrame alt={title} ratio={(((layout as string) === "stats") ? "1/1" : "4/3")} />
      )}
    >
      {((layout as string) === "grid") ? (
        <FeaturePairs features={features} />
      ) : null}
      {((layout as string) === "cards") ? (
        <FeatureCards features={features} cols="1-2" item="tile" />
      ) : null}
      {((layout as string) === "stats") ? (
        <FeatureStatRows features={features} stats={stats} />
      ) : null}
      {((layout as string) === "metrics") ? (
        <FeatureMetricCards stats={stats} />
      ) : null}
      {(((layout as string) === "cards") || ((layout as string) === "metrics")) ? (
        <ActionGroup
          primary={actions.cta}
          primaryHref={actions.href}
          secondary={((actions.mode === "two") ? actions.secondary : undefined)}
          secondaryHref={actions.secondaryHref}
          align="start"
        />
      ) : null}
    </SectionFrame>
  );
}
