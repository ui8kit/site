    Contact Form

    A message form, alone or beside the ways to reach us.

    Install

    npx ui8kit add contact-form

    Pulls in: button, checkbox, feature-grid, field, form, grid, group, input, label, lucide-icon, section-frame, stack, text, textarea, utils

    Code

    Svelte

    Point the CLI at this runtime once, then add the item:

    npx ui8kit init --yes --framework svelte --dir src --registry-url https://registry.buildy.tw/svelte --strict-cdn
    npx ui8kit add contact-form

export function ContactForm({
  id,
  title,
  text,
  labels,

  level = 2,

  return (

              <Input type="text" name="name" autoComplete="name" id={tpl("{0}-name", id)} data-testid="field" />

          ) : null}

              <Input type="text" name="subject" id={tpl("{0}-subject", id)} data-testid="field" />
            </Field>
          ) : null}
          <Field label={labels.message} htmlFor={tpl("{0}-message", id)}>
