    Milestone Timeline

    Milestones on a vertical line beside a picture, in a narrow column, or in a row.

    Install

    npx ui8kit add milestone-timeline

    Pulls in: badge, group, inline, list, media-frame, section-frame, text, title, utils

    Code

    Svelte

    Point the CLI at this runtime once, then add the item:

    npx ui8kit init --yes --framework svelte --dir src --registry-url https://registry.buildy.tw/svelte --strict-cdn
    npx ui8kit add milestone-timeline

/**
 * Milestones on a line: the year, a tag, a title and a sentence each. Axes: layout (split: a vertical line

import { type ReactNode } from "react";
import { Badge } from "../primitives/badge/badge";
import { Group } from "../primitives/group/group";
import { Inline } from "../primitives/inline/inline";

import { MediaFrame } from "../components/media-frame";

import type { Milestone } from "./milestone-timeline.types";

export type MilestoneTimelineProps = {
  id: string;
  title: string;
  lead?: string;
  milestones: Milestone[];
  layout?: "split" | "stacked" | "row";
  tone?: "default" | "muted";
  density?: "default" | "compact";
};

export function MilestoneTimeline({
  id,
  title,
  lead = "",
  milestones,
  layout = "split",
  tone = "default",
  density = "default",
}: MilestoneTimelineProps) {
  return (
    <SectionFrame
      id={id}
      title={title}
      lead={lead}
      tone={tone}
      density={density}
      layout={(((layout as string) === "split") ? "split" : "stack")}
      media={{ show: ((layout as string) === "split"), side: "start", align: "start" }}
      measure={(((layout as string) === "stacked") ? "2xl" : "default")}
      aside={(
        <MediaFrame alt={title} ratio="3/4" />
      )}
    >
      <List
        tag="ol"
        className={
          (layout as string) === "row" ? "grid grid-cols-1 gap-8 border-t border-border sm:grid-cols-2 lg:grid-cols-4"
          : "flex flex-col gap-8 border-l border-border"
        }
        data-testid="items"
      >
        {(milestones ?? []).map((m) => (
          <ListItem key={m.id}
            className={
              (layout as string) === "row" ? "relative flex flex-col gap-2 pt-6"
              : "relative flex flex-col gap-2 pl-6"
            }
            data-testid="item"
          >
            <Inline
              className={
                (layout as string) === "row" ? "absolute top-0 left-0 size-2 -translate-y-1/2 rounded bg-primary"
                : "absolute top-2 left-0 size-2 -translate-x-1/2 rounded bg-primary"
              }
              aria-hidden="true"
            />
            <Group gap="2" align="center">
              <Text size="sm" weight="semibold">
                {m.date}
              </Text>
              <Badge variant="outline" size="sm">
                {m.tag}
              </Badge>
            </Group>
            <Title as={3} size="lg" weight="bold">
              {m.title}
            </Title>
            <Text size="sm" className="text-muted-foreground">
              {m.text}
            </Text>
          </ListItem>
        ))}
      </List>
    </SectionFrame>
  );
}
