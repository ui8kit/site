/**
 * A section with a heading and the posts as cards: a grid, a list of horizontal cards, or a magazine (the first
 * post wide, the rest in a grid). Axes: layout (grid, list, magazine), columns, media, meta (full, short, news),
 * filter mode (none, topics), tone, density.
 */
import { Fragment, type ReactNode } from "react";
import { Grid } from "../primitives/grid/grid";
import { Stack } from "../primitives/stack/stack";
import { PostCard } from "../blocks/post-card";
import { SectionFrame } from "../components/section-frame";
import { Switcher } from "../components/switcher";
import { SwitcherItem } from "../components/switcher-item";
import { fitCols } from "../utils";
import type { PostTopic, PostFilter } from "./post-grid.types";
import type { Post, PostLabels } from "./post-card.types";
import type { HeadLink } from "../components/section-frame.types";

export type PostGridProps = {
  id: string;
  title: string;
  posts: Post[];
  labels: PostLabels;
  action?: HeadLink;
  filter?: PostFilter;
  layout?: "grid" | "list" | "magazine";
  cols?: "1-2" | "1-3" | "1-4";
  media?: "image" | "none";
  meta?: "full" | "short" | "news";
  tone?: "default" | "muted";
  density?: "default" | "compact";
  onTopic?: (value: string) => void;
};

export function PostGrid({
  id,
  title,
  posts,
  labels,
  action,
  filter,
  layout = "grid",
  cols = "1-3",
  media = "image",
  meta = "full",
  tone = "default",
  density = "default",
  onTopic,
}: PostGridProps) {
  return (
    <SectionFrame
      id={id}
      title={title}
      action={action}
      tone={tone}
      density={(((density as string) === "compact") ? "compact" : "default")}
      measure={(((meta as string) === "news") ? "fill" : "default")}
      className={(((meta as string) === "news") ? "py-0" : "")}
    >
      {(filter?.mode === "topics") ? (
        <Switcher label={filter?.label}>
          {(filter?.topics ?? []).map((tab) => (
            <SwitcherItem key={tab.id} active={(tab.value === filter?.topic)} onClick={() => onTopic?.(tab.value)}>
              {tab.title}
            </SwitcherItem>
          ))}
        </Switcher>
      ) : null}
      {((layout as string) === "magazine") ? (
        <Stack gap="6" align="stretch" data-testid="items">
          {(posts.filter((i) => !filter?.topic || i["category"] === filter?.topic) ?? []).map((p, i) => (
            <Fragment key={p.id}>
              {(i === 0) ? (
                <PostCard post={p} labels={labels} layout="horizontal" media={media} meta={meta} density={density} />
              ) : null}
            </Fragment>
          ))}
          <Grid cols={cols} className="gap-6">
            {(posts.filter((i) => !filter?.topic || i["category"] === filter?.topic) ?? []).map((p, i) => (
              <Fragment key={p.id}>
                {(i > 0) ? (
                  <PostCard post={p} labels={labels} layout="vertical" media={media} meta={meta} density={density} />
                ) : null}
              </Fragment>
            ))}
          </Grid>
        </Stack>
      ) : (
        <Grid
          cols={
            (layout as string) === "list" ? "1"
            : fitCols(cols, (posts.filter((i) => !filter?.topic || i["category"] === filter?.topic)?.length ?? 0))
          }
          className="w-full gap-6"
          data-testid="items"
        >
          {(posts.filter((i) => !filter?.topic || i["category"] === filter?.topic) ?? []).map((p) => (
            <PostCard key={p.id}
              post={p}
              labels={labels}
              layout={(((layout as string) === "list") ? "horizontal" : "vertical")}
              media={media}
              meta={meta}
              density={density}
            />
          ))}
        </Grid>
      )}
    </SectionFrame>
  );
}
