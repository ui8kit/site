  id: string;
  title: string;
  channels: Feature[];

  layout?: "grid" | "list";
  cols?: "1-2" | "1-3";
  action?: "link" | "none";
  tone?: "default" | "muted";
};

export function SupportChannels({
  id,
  title,
  channels,
  open,
  layout = "grid",
  cols = "1-3",
  action = "link",
  tone = "default",

    <SectionFrame id={id} title={title} tone={tone}>
