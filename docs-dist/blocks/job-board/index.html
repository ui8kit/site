/**
 * Open jobs as cards in two columns, grouped by department, or as a table: the title, the department, the place,
 * the type, an optional salary and an apply link. Axes: layout (grid, grouped, table), salary (show, hide),
 * tone.
 */
import { type ReactNode } from "react";
import { Badge } from "../primitives/badge/badge";
import { Group } from "../primitives/group/group";
import { Link } from "../primitives/link/link";
import { List } from "../primitives/list/list";
import { ListItem } from "../primitives/list/list-item";
import { Stack } from "../primitives/stack/stack";
import { Text } from "../primitives/text/text";
import { Title } from "../primitives/title/title";
import { JobCards } from "../components/job-cards";
import { SectionFrame } from "../components/section-frame";
import { tpl } from "../utils";
import type { Job, Department, JobLabels } from "./job-board.types";

export type JobBoardProps = {
  id: string;
  title: string;
  lead?: string;
  jobs: Job[];
  departments: Department[];
  labels: JobLabels;
  layout?: "grid" | "grouped" | "table";
  salary?: "show" | "hide";
  tone?: "default" | "muted";
  landmarkLabel?: string;
};

export function JobBoard({
  id,
  title,
  lead = "",
  jobs,
  departments,
  labels,
  layout = "grid",
  salary = "hide",
  tone = "default",
  landmarkLabel,
}: JobBoardProps) {
  return (
    <SectionFrame
      id={id}
      title={title}
      lead={lead}
      tone={tone}
      measure={(((layout as string) === "grouped") ? "3xl" : "default")}
      landmarkLabel={landmarkLabel}
    >
      {((layout as string) === "grouped") ? (
        <Stack gap="8" align="stretch" data-testid="groups">
          {(departments ?? []).map((d) => (
            <Stack key={d.id} gap="2" align="stretch" data-testid="group">
              <Group gap="4" align="baseline" justify="between">
                <Title as={3} size="lg" weight="bold">
                  {d.name}
                </Title>
                <Text size="xs" className="text-muted-foreground">
                  {tpl(labels.count, (jobs.filter((i) => !d.id || i["dept"] === d.id)?.length ?? 0))}
                </Text>
              </Group>
              <List tag="ul" className="flex flex-col">
                {(jobs.filter((i) => !d.id || i["dept"] === d.id) ?? []).map((j) => (
                  <ListItem key={j.id}
                    className="flex flex-wrap items-center justify-between gap-4 border-b border-border py-4"
                    data-testid="item"
                  >
                    <Stack gap="0">
                      <Link href="#/careers" variant="default" className="font-medium">
                        {j.title}
                      </Link>
                      <Text size="xs" className="text-muted-foreground">
                        {tpl("{0}, {1}", j.location, j.type)}
                      </Text>
                    </Stack>
                    <Badge variant="outline" size="default">
                      {j.type}
                    </Badge>
                  </ListItem>
                ))}
              </List>
            </Stack>
          ))}
        </Stack>
      ) : (
        <JobCards jobs={jobs} departments={departments} labels={labels} layout={layout} salary={salary} />
      )}
    </SectionFrame>
  );
}
