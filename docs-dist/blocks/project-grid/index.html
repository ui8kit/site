    Project Grid

    Projects as cards, a compact grid, captions over pictures, a bento or a list beside a collage.

    Install

    npx ui8kit add project-grid

    Pulls in: box, group, media-frame, project-tiles, section-frame, stack, text, utils

    Code

    Svelte

    Point the CLI at this runtime once, then add the item:

    npx ui8kit init --yes --framework svelte --dir src --registry-url https://registry.buildy.tw/svelte --strict-cdn
    npx ui8kit add project-grid

/**
 * Projects of a studio or a person: cards with tags and links, a compact grid, tiles with the caption over the
 * picture, a bento with one big project, or a list beside a collage. Axes: layout (cards, compact, overlay,
 * bento, list), tone, density.
 */
import { type ReactNode } from "react";
import { Box } from "../primitives/box/box";
import { Group } from "../primitives/group/group";
import { Stack } from "../primitives/stack/stack";
import { Text } from "../primitives/text/text";
import { MediaFrame } from "../components/media-frame";
import { ProjectTiles } from "../components/project-tiles";
import { SectionFrame } from "../components/section-frame";
import type { Project, ProjectLabels } from "./project-grid.types";

export type ProjectGridProps = {
  id: string;
  title: string;
  lead?: string;
  projects: Project[];
  labels: ProjectLabels;
  layout?: "cards" | "compact" | "overlay" | "bento" | "list";
  tone?: "default" | "muted";
  density?: "default" | "compact";
};

export function ProjectGrid({
  id,
  title,
  lead = "",
  projects,
  labels,
  layout = "cards",
  tone = "default",
  density = "default",
}: ProjectGridProps) {
  return (
    <SectionFrame
      id={id}
      title={title}
      lead={lead}
      tone={tone}
      density={density}
      layout={(((layout as string) === "list") ? "split" : "stack")}
      media={{ show: ((layout as string) === "list"), align: "start" }}
      aside={(
        <Box className="grid grid-cols-2 gap-4" data-testid="collage">
          <MediaFrame alt={title} ratio="3/4" />
          <MediaFrame alt={title} ratio="3/4" className="mt-8" />
        </Box>
      )}
    >
      {((layout as string) === "list") ? (
        <Stack gap="4" align="stretch" data-testid="items">
          {(projects ?? []).map((p) => (
            <Group key={p.id} gap="4" align="start" className="border-b border-border pb-4" data-testid="item">
              <MediaFrame alt={p.title} ratio="1/1" className="size-16 shrink-0" />
              <Stack gap="2" align="stretch" className="flex-1">
                <Group gap="4" justify="between">
                  <Text size="sm" weight="semibold">
                    {p.title}
                  </Text>
                  <Text size="xs" className="text-muted-foreground">
                    {p.year}
                  </Text>
                </Group>
                <Text size="sm" className="text-muted-foreground">
                  {p.summary}
                </Text>
              </Stack>
            </Group>
          ))}
        </Stack>
      ) : (
        <ProjectTiles projects={projects} labels={labels} layout={layout} />
      )}
    </SectionFrame>
  );
}
