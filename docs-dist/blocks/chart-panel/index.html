    Chart Panel

    Orders per day as bars or meters.

    Install

    npx ui8kit add chart-panel

    Pulls in: bar-chart, container, meter-row, panel, stack, utils

    Code

    Svelte

    Point the CLI at this runtime once, then add the item:

    npx ui8kit init --yes --framework svelte --dir src --registry-url https://registry.buildy.tw/svelte --strict-cdn
    npx ui8kit add chart-panel

/**
 * A panel with a chart of one series: vertical bars with captions, or meter rows with a percent. Axes: chart
 * (bars, meters), size (default, compact), tone (default, muted).
 */
import { type ReactNode } from "react";
import { Container } from "../primitives/container/container";
import { Stack } from "../primitives/stack/stack";
import { BarChart } from "../components/bar-chart";
import { MeterRow } from "../components/meter-row";
import { Panel } from "../components/panel";
import { tpl } from "../utils";
import type { Point } from "./chart-panel.types";

export type ChartPanelProps = {
  id: string;
  title: string;
  text: string;
  series: Point[];
  summary: string;
  percent: string;
  chart?: "bars" | "meters";
  size?: "default" | "compact";
  tone?: "default" | "muted";
};

export function ChartPanel({
  id,
  title,
  text,
  series,
  summary,
  percent,
  chart = "bars",
  size = "default",
  tone = "default",
}: ChartPanelProps) {
  return (
    <Container className="mx-auto w-full max-w-6xl px-4 py-8" data-slot="container">
      <Panel title={title} description={text} id={id} className={(((tone as string) === "muted") ? "bg-muted" : "")}>
        {((chart as string) === "bars") ? (
          <BarChart bars={series} label={summary} height={(((size as string) === "compact") ? 96 : 176)} />
        ) : (
          <Stack gap={(((size as string) === "compact") ? "0" : "2")} align="stretch" data-testid="items">
            {(series ?? []).map((s) => (
              <MeterRow key={s.id} label={s.label} value={s.value} caption={tpl(percent, s.value)} />
            ))}
          </Stack>
        )}
      </Panel>
    </Container>
  );
}
