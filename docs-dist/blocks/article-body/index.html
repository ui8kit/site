    Article Body

    A long read in one reading column, or beside a table of contents that links to the sections.

    Install

    npx ui8kit add article-body

    Pulls in: block, grid, link, list, section-frame, stack, text, title, utils

    Code

    Svelte

    Point the CLI at this runtime once, then add the item:

    npx ui8kit init --yes --framework svelte --dir src --registry-url https://registry.buildy.tw/svelte --strict-cdn
    npx ui8kit add article-body

/**
 * A long read: sections with headings, text and an optional pull quote, in one reading column or beside a table
 * of contents that links to each section. Axes: layout (column, toc), quotes (show, hide), tone. Highlighting
 * the section being read is a port contract.
 */
import { type ReactNode } from "react";
import { Block } from "../primitives/block/block";
import { Grid } from "../primitives/grid/grid";
import { Link } from "../primitives/link/link";
import { List } from "../primitives/list/list";
import { ListItem } from "../primitives/list/list-item";
import { Stack } from "../primitives/stack/stack";
import { Text } from "../primitives/text/text";
import { Title } from "../primitives/title/title";
import { SectionFrame } from "../components/section-frame";
import { cn, tpl } from "../utils";
import type { Section } from "./article-body.types";

export type ArticleBodyProps = {
  id: string;
  title: string;
  sections: Section[];
  tocTitle: string;
  layout?: "column" | "toc";
  quotes?: "show" | "hide";
  tone?: "default" | "muted";
};

export function ArticleBody({
  id,
  title,
  sections,
  tocTitle,
  layout = "column",
  quotes = "show",
  tone = "default",
}: ArticleBodyProps) {
  return (
    <SectionFrame id={id} title={title} tone={tone} density="default">
      <Grid
        cols="1"
        className={cn(
          "items-start gap-8",
          (((layout as string) === "toc") ? "md:grid-cols-[14rem_minmax(0,1fr)]" : "")
        )}
      >
        {((layout as string) === "toc") ? (
          <Block tag="nav" className="md:sticky md:top-4" aria-label={tocTitle} data-testid="toc">
            <Stack gap="2">
              <Text size="sm" weight="semibold">
                {tocTitle}
              </Text>
              <List className="m-0 flex list-none flex-col gap-2 p-0">
                {(sections ?? []).map((s) => (
                  <ListItem key={s.id}>
                    <Link
                      href={tpl("#{0}-{1}", id, s.id)}
                      variant="default"
                      className="text-sm text-muted-foreground"
                      data-testid="toc-link"
                    >
                      {s.title}
                    </Link>
                  </ListItem>
                ))}
              </List>
            </Stack>
          </Block>
        ) : null}
        <Stack gap="8" align="stretch" className="mx-auto w-full max-w-2xl min-w-0" data-testid="article">
          {(sections ?? []).map((s) => (
            <Block key={s.id} tag="div" className="flex flex-col gap-4" id={tpl("{0}-{1}", id, s.id)} data-testid="item">
              <Title as={3} size="2xl" weight="bold">
                {s.title}
              </Title>
              <Text size="base">
                {s.text}
              </Text>
              {((quotes as string) === "show") ? (
                <>
                  {s.quote ? (
                    <Block tag="blockquote" className="m-0 border-l border-border pl-4 text-lg" data-testid="quote">
                      {s.quote}
                    </Block>
                  ) : null}
                </>
              ) : null}
            </Block>
          ))}
        </Stack>
      </Grid>
    </SectionFrame>
  );
}
