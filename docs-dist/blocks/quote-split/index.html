    Quote Split

    One quote beside a picture, a rating panel or over logo tiles; stacked quotes beside a picture; four compact quotes; figures over a featured quote and a stack.

    Install

    npx ui8kit add quote-split

    Pulls in: action-group, box, error-page, logo-cloud, media-frame, quote-figures, quote-logo-tiles, quote-rating-panel, quote-slot, section-frame, stack, stats-band, testimonial-grid, utils

    Code

    Svelte

    Point the CLI at this runtime once, then add the item:

    npx ui8kit init --yes --framework svelte --dir src --registry-url https://registry.buildy.tw/svelte --strict-cdn
    npx ui8kit add quote-split

export function QuoteSplit({
  id,
  title,
  lead = "",
  badge = "",
  quotes,
  labels,
  extras,
  actions,
  layout = "media",
  mediaSide = "end",
  tone = "default",
}: QuoteSplitProps) {
  return (
    <>
      {(((layout as string) === "compact") || ((layout as string) === "bento")) ? (
        <SectionFrame id={id} title={title} lead={lead} heading={{ badge }} tone={tone} density="default">
          {((layout as string) === "bento") ? (
            <QuoteFigures extras={extras} />
          ) : null}
          <Box
            className={
              (layout as string) === "bento" ? "grid grid-cols-1 gap-4 lg:grid-cols-2"
              : "grid grid-cols-1 gap-4 sm:grid-cols-2 lg:grid-cols-4"
            }
          >
            {(quotes ?? []).map((q, i) => (
              <QuoteSlot key={q.id}
                q={q}
                labels={labels}
                kind="card"
                layout={((((layout as string) === "bento") && (i === 0)) ? "single" : "grid")}
                className={((((layout as string) === "bento") && (i === 0)) ? "lg:row-span-3" : "")}
              />
            ))}
          </Box>
        </SectionFrame>
      ) : (
        <SectionFrame
          id={id}
          title={title}
          lead={lead}
          heading={{ badge }}
          tone={tone}
          density="default"
          layout="split"
          media={{ show: true, side: mediaSide }}
          aside={(
            <>
              {((layout as string) === "rating") ? (
                <QuoteRatingPanel extras={extras} labels={labels} />
              ) : (
                <MediaFrame alt={title} ratio="4/3" />
              )}
            </>
          )}
        >
          {(((layout as string) === "media") || ((layout as string) === "logos") || ((layout as string) === "rating")) ? (
            <>
              {(quotes ?? []).map((q, i) => (
                <Fragment key={q.id}>
                  {(i < 1) ? (
                    <QuoteSlot
                      q={q}
                      labels={labels}
                      kind={(((layout as string) === "media") ? "card" : "plain")}
                      layout="single"
                      className=""
                    />
                  ) : null}
                </Fragment>
              ))}
            </>
          ) : null}
          {((layout as string) === "logos") ? (
            <QuoteLogoTiles extras={extras} />
          ) : null}
          {((layout as string) === "stack") ? (
            <Stack gap="4" align="stretch">

                  ) : null}
                </Fragment>

              secondaryHref={actions.secondaryHref}
              align="start"
            />
          ) : null}
        </SectionFrame>
      )}
    </>
  );
}
