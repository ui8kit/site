    Error Page

    A page for something that went wrong, such as a missing page: a code, a heading, a sentence and the way back.

    Install

    npx ui8kit add error-page

    Pulls in: action-group, media-frame, section-frame, text, utils

/**
 * A page that tells the visitor something went wrong: an icon or a big code, a heading, a sentence and the way
 * back. Axes: layout (centered, split: the picture beside the text), code (show, hide), actions mode (one, two),
 * tone.
 */
import { type ReactNode } from "react";
import { Text } from "../primitives/text/text";
import { ActionGroup } from "../components/action-group";
import { MediaFrame } from "../components/media-frame";
import { SectionFrame } from "../components/section-frame";
import { cn } from "../utils";
import type { LinkActions } from "./error-page.types";

export type ErrorPageProps = {
  id: string;
  title: string;
  text: string;
  status: string;
  actions: LinkActions;
  level?: 1 | 2;
  layout?: "centered" | "split";
  code?: "show" | "hide";
  tone?: "default" | "muted";
  landmarkLabel?: string;
};

export function ErrorPage({
  id,
  title,
  text,
  status,
  actions,
  level = 1,
  layout = "centered",
  code = "show",
  tone = "default",
  landmarkLabel,
}: ErrorPageProps) {
  return (
    <SectionFrame
      id={id}
      title={title}
      lead={text}
      heading={{ level }}
      layout={(((layout as string) === "split") ? "split" : "stack")}
      media={{ show: ((layout as string) === "split") }}
      align={(((layout as string) === "split") ? "start" : "center")}
      tone={tone}
      density="roomy"
      landmarkLabel={landmarkLabel}
      aside={(
        <MediaFrame alt={title} ratio="4/3" />
      )}
    >
      {((code as string) === "show") ? (
        <Text
          size="4xl"
          weight="bold"
          className={cn(
            "text-8xl text-muted-foreground",
            (((layout as string) === "split") ? "" : "text-center")
          )}
          data-testid="error-code"
        >
          {status}
        </Text>
      ) : null}
      <ActionGroup
        primary={actions.cta}
        primaryHref={actions.href}
        secondary={((actions.mode === "two") ? actions.secondary : undefined)}
        secondaryHref={actions.secondaryHref}
        align={(((layout as string) === "split") ? "start" : "center")}
      />
    </SectionFrame>
  );
}
