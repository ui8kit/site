    Bottom Nav

    Pulls in: block, inline, link, list, lucide-icon, nav-bar, utils

  labels?: "show" | "hide";
  position?: "static" | "sticky";
  tone?: "default" | "muted";
};

export function BottomNav({
  label,
  tabs,
  current = "",
  labels = "show",
  position = "static",
  tone = "default",
}: BottomNavProps) {
  return (
    <Block
      tag="nav"
      className={cn(
