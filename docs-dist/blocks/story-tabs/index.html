    Story Tabs

    Year tabs over the panel of the chosen year, or the heading beside a picture with arrows and one dot per step. Choosing a step is an event.

    Install

    npx ui8kit add story-tabs

    Pulls in: box, button, group, lucide-icon, media-frame, milestone-timeline, section-frame, stack, story-dots, story-panel, text, utils

    Code

    Svelte

    Point the CLI at this runtime once, then add the item:

    npx ui8kit init --yes --framework svelte --dir src --registry-url https://registry.buildy.tw/svelte --strict-cdn
    npx ui8kit add story-tabs

/**
 * A story in steps (years or slides). Axes: layout (years: a row of year tabs over the panel of the active year
 * with its badge, step, title, text and tag; carousel: the heading beside a picture with previous and next
 * arrows and one dot per step), tone. Choosing a step emits `select`.

import { Fragment, type ReactNode } from "react";
import { Box } from "../primitives/box/box";
import { Button } from "../primitives/button/button";
import { Group } from "../primitives/group/group";
import { LucideIcon } from "../primitives/lucide-icon/lucide-icon";
import { Stack } from "../primitives/stack/stack";
import { Text } from "../primitives/text/text";
import { MediaFrame } from "../components/media-frame";
import { SectionFrame } from "../components/section-frame";
import { StoryDots } from "../components/story-dots";
import { StoryPanel } from "../components/story-panel";
import type { StoryLabels } from "./story-tabs.types";
import type { Milestone } from "./milestone-timeline.types";

export type StoryTabsProps = {
  id: string;
  title: string;
  lead?: string;
  badge?: string;
  steps: Milestone[];
  active: string;
  labels: StoryLabels;
  layout?: "years" | "carousel";
  tone?: "default" | "muted";
  onSelect?: (id: string) => void;
};

export function StoryTabs({
  id,
  title,
  lead = "",
  badge = "",
  steps,
  active,
  labels,
  layout = "years",
  tone = "default",
  onSelect,
}: StoryTabsProps) {
  return (
    <>
      {((layout as string) === "years") ? (
        <SectionFrame id={id} title={title} lead={lead} heading={{ badge }} tone={tone}>
          <Box className="grid grid-cols-2 gap-2 sm:grid-cols-4" data-testid="story-tabs">
            {(steps ?? []).map((s) => (
              <Button key={s.id}
                variant={((s.id === active) ? "default" : "outline")}
                size="lg"
                type="button"
                className="h-auto flex-col items-start gap-0 py-2 text-start"
                data-testid="story-tab"
                data-active={((s.id === active) ? "true" : "false")}
                onClick={() => onSelect?.(s.id)}
              >
                <Text size="sm" weight="semibold">
                  {s.date}
                </Text>
                <Text size="xs">
                  {s.tag}
                </Text>
              </Button>
            ))}
          </Box>
          {(steps ?? []).map((s, i) => (
            <Fragment key={s.id}>
              {(s.id === active) ? (
                <StoryPanel s={s} index={i} total={(steps?.length ?? 0)} labels={labels} />
              ) : null}
            </Fragment>
          ))}
        </SectionFrame>
      ) : (
        <SectionFrame
          id={id}
          title={title}
          lead={lead}
          heading={{ badge }}
          tone={tone}
          layout="split"
          media={{ show: true }}
          aside={(
            <Stack gap="4" align="stretch">
              <MediaFrame alt={title} ratio="4/3" />
              <Group gap="4" align="center" justify="between">
                <Button
                  variant="outline"
                  size="sm"
                  layout="icon"
                  type="button"
                  aria-label={labels.previous}
                  data-testid="story-arrow"
                >
                  <LucideIcon name="chevron-left" size={16} />
                </Button>
                <StoryDots steps={steps} active={active} />
                <Button
                  variant="outline"
                  size="sm"
                  layout="icon"
                  type="button"
                  aria-label={labels.next}
                  data-testid="story-arrow"
                >
                  <LucideIcon name="chevron-right" size={16} />
                </Button>
              </Group>
            </Stack>
          )}
        />
      )}
    </>
  );
}
