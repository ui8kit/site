    block / commerce

    Store Locations

    The stores with address, hours, phone and services.

    Install

    npx ui8kit add store-locations

    Pulls in: badge, box, group, inline, link, list, lucide-icon, media-frame, section-frame, stack, text, title, utils

    Code

    Svelte

    Point the CLI at this runtime once, then add the item:

    npx ui8kit init --yes --framework svelte --dir src --registry-url https://registry.buildy.tw/svelte --strict-cdn
    npx ui8kit add store-locations

export function StoreLocations({
  id,
  title,
  lead = "",

  kinds,

  return (
    <SectionFrame id={id} title={title} lead={lead} tone={tone} density="default">

            <Stack key={s.id}
              gap="4"

                (layout as string) === "list" ? "border-b border-border pb-4"
                : "rounded bg-card inset-ring inset-ring-border p-6"
              }

            >
              <Stack gap="2">
                <Badge variant="outline" size="default" className="w-fit">

              >
                <ListItem className="flex items-center gap-2">
                  <LucideIcon name="map-pin" size={16} />
                  <Inline>
                    {s.address}
                  </Inline>

                  <Inline>
                    {s.hours}
                  </Inline>
                </ListItem>
                <ListItem className="flex items-center gap-2">
                  <LucideIcon name="phone" size={16} />
                  <Link href={tpl("tel:{0}", s.phone)} variant="default">
                    {s.phone}
                  </Link>
                </ListItem>
              </List>
              {((layout as string) === "cards") ? (
                <>
                  <Text size="sm" className="text-muted-foreground">
                    {s.note}
