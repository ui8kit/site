    Usp Strip

    Free delivery, returns, warranty and help in one band.

    Install

    npx ui8kit add usp-strip

    Pulls in: box, feature-grid, grid, inline, list, lucide-icon, section-frame, stack, text, utils

    Preview

    Free shipping over $75

    Two-hour windows, seven days a week.

    Thirty-day returns

    Send it back for free if it does not fit your desk.

    Two-year warranty

    Repairs or a new one, whichever is faster.

    Help seven days a week

    A person answers within the hour.

export type UspStripProps = {
  id: string;
  title?: string;
  perks: Feature[];
  layout?: "strip" | "cards" | "bar";
  tone?: "default" | "muted";
  density?: "default" | "compact";
};

export function UspStrip({
  id,
  title = "",
  perks,
  layout = "strip",
  tone = "default",
  density = "default",
}: UspStripProps) {
  return (
    <SectionFrame id={id} title={title} tone={tone} density={density}>
      {((layout as string) === "bar") ? (
        <List tag="ul" className="flex flex-wrap items-center justify-center gap-x-8 gap-y-4" data-testid="items">
          {(perks ?? []).map((p) => (
            <ListItem key={p.id} className="flex items-center gap-2 text-sm font-medium" data-testid="item">
              <LucideIcon name={p.icon} size={16} />
              <Inline>
                {p.title}
              </Inline>
            </ListItem>
          ))}
        </List>
      ) : (
        <Grid
          cols="1-4"
          className={cn("gap-4", (((layout as string) === "strip") ? "rounded bg-muted p-6" : ""))}
          data-testid="items"
        >
          {(perks ?? []).map((p) => (
            <Stack key={p.id}
              gap="4"
              className={
                (layout as string) === "cards" ? "rounded bg-card inset-ring inset-ring-border p-6"
                : ""
              }
              data-testid="item"

              <Box className="flex size-10 shrink-0 items-center justify-center rounded bg-muted">

                  {p.text}
                </Text>
              </Stack>
            </Stack>
          ))}
        </Grid>
      )}
    </SectionFrame>
  );
}
