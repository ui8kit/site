    Post Split

    Tags, reading figures and an author card with share and save over a wide cover; the author, figures, tags and actions beside the cover; or the facts and related reading beside it.

    Install

    npx ui8kit add post-split

    Pulls in: media-frame, nav-bar, post-actions, post-author, post-card, post-outline, post-stats, post-tags, section-frame, utils

    Code

    Svelte

    Point the CLI at this runtime once, then add the item:

    npx ui8kit init --yes --framework svelte --dir src --registry-url https://registry.buildy.tw/svelte --strict-cdn
    npx ui8kit add post-split

/**
 * A post header with the author, the reading figures, the tags and the actions. Axes: layout (center: tags,
 * figures and an author card with share and save over a wide cover; split: beside the cover; outline: facts and
 * related reading beside the cover), mediaSide, author (line, card, bio), cover (wide, square), actions (pair,
 * wide, icons), tone.
 */
import { type ReactNode } from "react";
import { MediaFrame } from "../components/media-frame";
import { PostActions } from "../components/post-actions";
import { PostAuthor } from "../components/post-author";
import { PostOutline } from "../components/post-outline";
import { PostStats } from "../components/post-stats";
import { PostTags } from "../components/post-tags";
import { SectionFrame } from "../components/section-frame";
import type { PostExtras, PostSplitLabels } from "./post-split.types";
import type { Post } from "./post-card.types";
import type { NavLink } from "./nav-bar.types";

export type PostSplitProps = {
  id: string;
  post: Post;
  labels: PostSplitLabels;
  extras: PostExtras;
  level?: 1 | 2;
  layout?: "center" | "split" | "outline";
  mediaSide?: "end" | "start";
  author?: "line" | "card" | "bio";
  cover?: "wide" | "square";
  actions?: "pair" | "wide" | "icons";
  tone?: "default" | "muted";
};

export function PostSplit({
  id,
  post,
  labels,
  extras,
  level = 1,
  layout = "split",
  mediaSide = "end",
  author = "line",
  cover = "wide",
  actions = "pair",
  tone = "default",
}: PostSplitProps) {
  return (
    <>
      {((layout as string) === "center") ? (
        <SectionFrame
          id={id}
          title={post.title}
          lead={post.excerpt}
          heading={{ badge: post.category, level }}
          tone={tone}
          media={{ show: true }}
          align="center"
          aside={(
            <MediaFrame
              alt={post.title}
              ratio={
                (layout as string) === "center" ? "21/9"
                : (cover as string) === "square" ? "1/1"
                : "4/3"
              }
            />
          )}
        >
          <PostTags tags={extras.tags} align="center" />
          <PostStats post={post} extras={extras} labels={labels} align="center" />
          <PostAuthor
            post={post}
            extras={extras}
            labels={labels}
            mode="card"
            share={true}
            className="mx-auto w-full max-w-md"
          />
        </SectionFrame>
      ) : (
        <SectionFrame
          id={id}
          title={post.title}
          lead={post.excerpt}
          heading={{ badge: post.category, level }}
          tone={tone}
          media={{ show: true, side: mediaSide }}
          layout="split"
          aside={(
            <MediaFrame
              alt={post.title}
              ratio={
                (layout as string) === "center" ? "21/9"
                : (cover as string) === "square" ? "1/1"
                : "4/3"
              }
            />
          )}
        >
          {((layout as string) === "outline") ? (
            <PostOutline post={post} extras={extras} labels={labels} />
          ) : (
            <>
              <PostAuthor post={post} extras={extras} labels={labels} mode={author} />
              {((author as string) !== "line") ? (
                <PostStats post={post} extras={extras} labels={labels} />
              ) : null}
              <PostTags tags={extras.tags} align="start" />
            </>
          )}
          <PostActions labels={labels} mode={actions} />
        </SectionFrame>
      )}
    </>
  );
}
