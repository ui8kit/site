    Faq List

    Questions as native disclosures.

    Install

    npx ui8kit add faq-list

    Pulls in: block, disclosure, grid, search-field, section-frame, stack, switcher, switcher-item, text, utils

    Code

    Svelte

    Point the CLI at this runtime once, then add the item:

    npx ui8kit init --yes --framework svelte --dir src --registry-url https://registry.buildy.tw/svelte --strict-cdn
    npx ui8kit add faq-list

import { type ReactNode } from "react";
import { Block } from "../primitives/block/block";
import { Disclosure } from "../primitives/disclosure/disclosure";
import { Summary } from "../primitives/disclosure/summary";
import { Grid } from "../primitives/grid/grid";
import { Stack } from "../primitives/stack/stack";
import { Text } from "../primitives/text/text";
import { SearchField } from "../components/search-field";
import { SectionFrame } from "../components/section-frame";
import { Switcher } from "../components/switcher";
import { SwitcherItem } from "../components/switcher-item";
import { tpl } from "../utils";
import type { Faq, FaqTab, FaqSearch, FaqFilter } from "./faq-list.types";

export type FaqListProps = {
  id: string;
  title: string;
  items: Faq[];
  search?: FaqSearch;
  filter?: FaqFilter;
  layout?: "accordion" | "chat";
  cols?: "1" | "2";
  multiple?: boolean;
  tone?: "default" | "muted";
  density?: "roomy" | "default" | "compact";
  onQuery?: (text: string) => void;
  onCategory?: (value: string) => void;
};

export function FaqList({
  id,
  title,
  items,
  search,
  filter,
  layout = "accordion",
  cols = "1",
  multiple = false,
  tone = "default",
  density = "default",
  onQuery,
  onCategory,
}: FaqListProps) {
  const inCategory = items.filter((i) => !filter?.category || i["category"] === filter?.category);
  const shown = inCategory.filter((i) => !search?.query || String(i["question"]).toLowerCase().includes(String(search?.query).toLowerCase()));
  return (
    <SectionFrame
      id={id}
      title={title}
      tone={tone}
      density={density}
      measure={(((layout as string) === "chat") ? "2xl" : "default")}
    >
      {(search?.mode === "field") ? (
        <SearchField
          label={(search?.label ?? "")}
          placeholder={search?.label}
          value={(search?.query ?? "")}
          onValue={(arg: string) => onQuery?.(arg)}
        />
      ) : null}
      {(filter?.mode === "categories") ? (
        <Switcher label={filter?.label}>
          {(filter?.tabs ?? []).map((tab) => (
            <SwitcherItem key={tab.id} active={(tab.value === filter?.category)} onClick={() => onCategory?.(tab.value)}>
              {tab.title}
            </SwitcherItem>
          ))}
        </Switcher>
      ) : null}
      {((layout as string) === "chat") ? (
        <Stack gap="4" align="stretch" data-testid="faq">
          {(shown ?? []).map((item) => (
            <Stack key={item.id} gap="2" align="stretch" data-testid="faq-item">
              <Block tag="div" className="ml-auto max-w-xs sm:max-w-md rounded bg-primary px-4 py-4 text-primary-foreground">
                <Text size="sm" weight="semibold">
                  {item.question}
                </Text>
              </Block>
              <Block tag="div" className="mr-auto max-w-xs sm:max-w-md rounded bg-muted px-4 py-4">
                <Text size="sm">
                  {item.answer}
                </Text>
              </Block>
            </Stack>
          ))}
        </Stack>
      ) : (
        <Grid cols={(((cols as string) === "2") ? "1-2" : "1")} className="items-start gap-x-8" data-testid="faq">
          {(shown ?? []).map((item) => (
            <Disclosure key={item.id} variant="item" name={(multiple ? undefined : tpl("{0}-faq", (id ?? "faq")))} data-testid="faq-item">
              <Summary variant="item">
                {item.question}
              </Summary>
              <Text size="sm" className="pb-4 text-muted-foreground">
                {item.answer}
              </Text>
            </Disclosure>
          ))}
        </Grid>
      )}
    </SectionFrame>
  );
}
