    Order Summary

    Subtotal, shipping, tax and the total.

    Install

    npx ui8kit add order-summary

    Pulls in: box, button, inline, list, section-frame, stack, text, utils

    Preview

    Order summary

    • Subtotal$288
    • Shipping$12
    • Tax$24
    • Total$324

    Taxes are calculated at checkout.

    Code

    Point the CLI at this runtime once, then add the item:

    npx ui8kit init --yes --framework svelte --dir src --registry-url https://registry.buildy.tw/svelte --strict-cdn
    npx ui8kit add order-summary

  rows: SummaryRow[];
  labels: SummaryLabels;

export function OrderSummary({
  id,
  title,
  rows,
  labels,
  layout = "card",
  cta = "show",
  note = "show",
  tone = "default",
  density = "default",
  landmarkLabel,
}: OrderSummaryProps) {
  return (
    <SectionFrame id={id} title={title} tone={tone} density={density} measure="md" landmarkLabel={landmarkLabel}>
      <Box
        className={cn(
          (((layout as string) === "card") ? "rounded bg-card p-6 inset-ring inset-ring-border" : "")
        )}
        data-testid="summary"
      >
        <Stack gap="4" align="stretch">
          <List tag="ul" className="flex flex-col gap-4">
            {(rows ?? []).map((row) => (
              <ListItem key={row.id}
                className={cn(
                  "flex items-center justify-between gap-4",

                    row.strong ? "border-t border-border pt-4 text-base font-bold"
                    : "text-sm text-muted-foreground"
                  )
                )}
                data-testid="item"
              >
                <Inline>
                  {row.label}
                </Inline>
                <Inline>
                  {fmtMoney(row.amount, row.currency)}
                </Inline>
              </ListItem>
            ))}

            <Button variant="default" size="default" className="w-full" data-testid="checkout">

            </Text>

      </Box>
