    Order Summary

    Subtotal, shipping, tax and the total.

    Install

    npx ui8kit add order-summary

    Pulls in: box, button, inline, list, section-frame, stack, text, utils

    Svelte

    Point the CLI at this runtime once, then add the item:

    npx ui8kit init --yes --framework svelte --dir src --registry-url https://registry.buildy.tw/svelte --strict-cdn
    npx ui8kit add order-summary

/**
 * The totals of an order: subtotal, shipping, tax and the total, with a checkout button and a note. Axes: layout
 * (card, plain), cta (show, hide), note (show, hide), density, tone.
 */
import { type ReactNode } from "react";
import { Box } from "../primitives/box/box";
import { Button } from "../primitives/button/button";
import { Inline } from "../primitives/inline/inline";
import { List } from "../primitives/list/list";
import { ListItem } from "../primitives/list/list-item";
import { Stack } from "../primitives/stack/stack";
import { Text } from "../primitives/text/text";
import { SectionFrame } from "../components/section-frame";
import { cn, fmtMoney } from "../utils";
import type { SummaryRow, SummaryLabels } from "./order-summary.types";

export type OrderSummaryProps = {
  id: string;
  title: string;
  rows: SummaryRow[];
  labels: SummaryLabels;
  layout?: "card" | "plain";
  cta?: "show" | "hide";
  note?: "show" | "hide";
  tone?: "default" | "muted";
  density?: "default" | "compact";
};

export function OrderSummary({
  id,
  title,
  rows,
  labels,
  layout = "card",
  cta = "show",
  note = "show",
  tone = "default",
  density = "default",
}: OrderSummaryProps) {
  return (
    <SectionFrame id={id} title={title} tone={tone} density={density} measure="md">
      <Box
        className={cn(
          (((layout as string) === "card") ? "rounded bg-card p-6 inset-ring inset-ring-border" : "")
        )}
        data-testid="summary"
      >
        <Stack gap="4" align="stretch">
          <List tag="ul" className="flex flex-col gap-4">
            {(rows ?? []).map((row) => (
              <ListItem key={row.id}
                className={cn(
                  "flex items-center justify-between gap-4",
                  (
                    row.strong ? "border-t border-border pt-4 text-base font-bold"
                    : "text-sm text-muted-foreground"
                  )
                )}
                data-testid="item"
              >
                <Inline>
                  {row.label}
                </Inline>
                <Inline>
                  {fmtMoney(row.amount, row.currency)}
                </Inline>
              </ListItem>
            ))}
          </List>
          {((cta as string) === "show") ? (
            <Button variant="default" size="default" className="w-full" data-testid="checkout">
              {labels.checkout}
            </Button>
          ) : null}
          {((note as string) === "show") ? (
            <Text size="xs" className="text-muted-foreground">
              {labels.note}
            </Text>
          ) : null}
        </Stack>
      </Box>
    </SectionFrame>
  );
}
