    Bento Grid

    Tiles of different sizes in one grid; the large tiles hold the main features.

    Install

    npx ui8kit add bento-grid

    Pulls in: card, grid, lucide-icon, media-frame, section-frame, text, title, utils

    Preview

    Everything in one place

    Fast by default

    Pages ship little script and load in under a second.

    Ready blocks

    Compose a page from tested blocks and variants.

    Accessible

    Landmarks, labels and focus order are checked.

    Light and dark

    Every block follows the theme tokens.

    One source of data

    Pages read collections, never copies.

    Private by default

    Your data stays in your collections.

    Export anywhere

    Take pages and data with you.

/**
 * Tiles of different sizes in one grid: the large and wide tiles hold the important features. Sizes come from

 * the large tiles), tone, density.

import { type ReactNode } from "react";
import { Card } from "../primitives/card/card";
import { CardContent } from "../primitives/card/card-content";
import { Grid } from "../primitives/grid/grid";
import { LucideIcon } from "../primitives/lucide-icon/lucide-icon";
import { Text } from "../primitives/text/text";
import { Title } from "../primitives/title/title";
import { MediaFrame } from "../components/media-frame";
import { SectionFrame } from "../components/section-frame";
import { at, cn } from "../utils";
import type { Tile } from "./bento-grid.types";

const SPAN: Record<string, string> = {
  "large": "md:col-span-2 md:row-span-2",
  "wide": "md:col-span-2",
  "small": "",
};

export type BentoGridProps = {
  id: string;
  title: string;
  tiles: Tile[];
  layout?: "bento" | "even";
  media?: "none" | "image";
  tone?: "default" | "muted";
  density?: "default" | "compact";
};

export function BentoGrid({
  id,
  title,
  tiles,
  layout = "bento",
  media = "none",
  tone = "default",
  density = "default",
}: BentoGridProps) {
  return (
    <SectionFrame id={id} title={title} tone={tone} density={density}>
      <Grid cols="1-4" className="auto-rows-fr gap-4" data-testid="items">
        {(tiles ?? []).map((tile) => (
          <Card key={tile.id}
            variant="default"
            className={cn(
              "h-full overflow-hidden",
              (((density as string) === "compact") ? "gap-4 p-4" : ""),
              (((layout as string) === "bento") ? at(SPAN, tile.size) : "")
            )}
            data-testid="item"
            data-size={(((layout as string) === "bento") ? tile.size : "even")}
          >
            <CardContent className="flex h-full flex-col gap-4">
              <LucideIcon name={tile.icon} size={24} />
              <Title as={3} size="xl" weight="bold">
                {tile.title}
              </Title>
              <Text size="sm" className="text-muted-foreground">
                {tile.text}
              </Text>
              {(((media as string) === "image") && ((tile.size === "large") || ((layout as string) === "even"))) ? (
                <MediaFrame alt={tile.title} ratio="4/3" className="mt-auto" />
              ) : null}
            </CardContent>
          </Card>
        ))}
      </Grid>
    </SectionFrame>
  );
}
