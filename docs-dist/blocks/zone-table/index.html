    Zone Table

    Delivery zones and their prices as a table or as cards.

    Install

    npx ui8kit add zone-table

    Pulls in: badge, grid, inline, list, section-frame, stack, text, title, utils, zone-rows

    Code

    Svelte

    Point the CLI at this runtime once, then add the item:

    npx ui8kit init --yes --framework svelte --dir src --registry-url https://registry.buildy.tw/svelte --strict-cdn
    npx ui8kit add zone-table

/**
 * Delivery zones and their prices as a table or as cards: the zone and its areas, the price, the order that
 * ships free, the minimum order, the time and an express badge. Axes: layout (table, cards), tone, density.
 */
import { type ReactNode } from "react";
import { Badge } from "../primitives/badge/badge";
import { Grid } from "../primitives/grid/grid";
import { Inline } from "../primitives/inline/inline";
import { List } from "../primitives/list/list";
import { ListItem } from "../primitives/list/list-item";
import { Stack } from "../primitives/stack/stack";
import { Text } from "../primitives/text/text";
import { Title } from "../primitives/title/title";
import { SectionFrame } from "../components/section-frame";
import { ZoneRows } from "../components/zone-rows";
import { fmtMoney } from "../utils";
import type { Zone, ZoneLabels } from "./zone-table.types";

export type ZoneTableProps = {
  id: string;
  title: string;
  lead?: string;
  zones: Zone[];
  labels: ZoneLabels;
  layout?: "table" | "cards";
  tone?: "default" | "muted";
  density?: "default" | "compact";
};

export function ZoneTable({
  id,
  title,
  lead = "",
  zones,
  labels,
  layout = "table",
  tone = "default",
  density = "default",
}: ZoneTableProps) {
  return (
    <SectionFrame id={id} title={title} lead={lead} tone={tone} density={density}>
      {((layout as string) === "table") ? (
        <ZoneRows zones={zones} labels={labels} />
      ) : (
        <Grid cols="1-3" className="gap-4" data-testid="items">
          {(zones ?? []).map((z) => (
            <Stack key={z.id} gap="4" className="rounded bg-card inset-ring inset-ring-border p-6" data-testid="item">
              <Stack gap="2">
                <Title as={3} size="lg" weight="bold">
                  {z.name}
                </Title>
                <Text size="sm" className="text-muted-foreground">
                  {z.area}
                </Text>
              </Stack>
              <Text size="3xl" weight="bold">
                {fmtMoney(z.price, z.currency)}
              </Text>
              <List tag="ul" className="flex flex-col gap-2 border-t border-border pt-4">
                <ListItem className="flex items-center justify-between gap-4 text-sm">
                  <Inline className="text-muted-foreground">
                    {labels.freeFrom}
                  </Inline>
                  <Inline className="font-medium">
                    {fmtMoney(z.freeFrom, z.currency)}
                  </Inline>
                </ListItem>
                <ListItem className="flex items-center justify-between gap-4 text-sm">
                  <Inline className="text-muted-foreground">
                    {labels.minOrder}
                  </Inline>
                  <Inline className="font-medium">
                    {fmtMoney(z.minOrder, z.currency)}
                  </Inline>
                </ListItem>
                <ListItem className="flex items-center justify-between gap-4 text-sm">
                  <Inline className="text-muted-foreground">
                    {labels.time}
                  </Inline>
                  <Inline className="font-medium">
                    {z.time}
                  </Inline>
                </ListItem>
              </List>
              {z.express ? (
                <Badge variant="secondary" size="default" className="w-fit">
                  {z.express}
                </Badge>
              ) : null}
            </Stack>
          ))}
        </Grid>
      )}
    </SectionFrame>
  );
}
