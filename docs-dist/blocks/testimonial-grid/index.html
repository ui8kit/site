  mark?: "none" | "icon";
};

/**
 * Customer quotes as cards, plain figures or social posts, in a grid, a masonry, a scroll strip, one featured
 * quote or one narrow column. Axes: layout (grid, single, stack, masonry, strip), columns, item kind (card,
 * plain, social), item avatar, item rating, item mark (a company glyph), density, tone. A carousel is the
 * Carousel port contract around the single layout.
 */
import { Fragment, type ReactNode } from "react";
import { Block } from "../primitives/block/block";
import { Grid } from "../primitives/grid/grid";
import { QuoteCard } from "../components/quote-card";
import { SectionFrame } from "../components/section-frame";
import type { Quote, QuoteLabels, QuoteItem } from "./testimonial-grid.types";

export type TestimonialGridProps = {
  id: string;
  title: string;
  quotes: Quote[];
  labels: QuoteLabels;
  layout?: "grid" | "single" | "stack" | "masonry" | "strip";
  cols?: "1-2" | "1-3";
  item?: QuoteItem;
  tone?: "default" | "muted";
  density?: "roomy" | "default" | "compact";
  landmarkLabel?: string;
};

export function TestimonialGrid({
  id,
  title,
  quotes,
  labels,
  layout = "grid",
  cols = "1-3",
  item,
  tone = "default",
  density = "default",
  landmarkLabel,
}: TestimonialGridProps) {
  return (
    <SectionFrame
      id={id}
      title={title}
      tone={tone}
      density={density}
      align={(((layout as string) === "single") ? "center" : "start")}
      measure={(((layout as string) === "grid") ? "default" : "2xl")}
      landmarkLabel={landmarkLabel}
    >
      {(((layout as string) === "masonry") || ((layout as string) === "strip")) ? (
        <Block
          tag="div"
          className={
            (layout as string) === "masonry" ? "columns-1 gap-4 md:columns-2 xl:columns-3"
            : "flex snap-x gap-4 overflow-x-auto pb-2"
          }
          data-testid="items"
        >
          {(quotes ?? []).map((q) => (
            <QuoteCard key={q.id} q={q} labels={labels} item={item} layout={layout} />
          ))}
        </Block>
      ) : (
        <Grid cols={(((layout as string) === "grid") ? cols : "1")} className="gap-4" data-testid="items">
          {(quotes ?? []).map((q, i) => (
            <Fragment key={q.id}>
              {(((layout as string) !== "single") || (i === 0)) ? (
                <QuoteCard q={q} labels={labels} item={item} layout={layout} />
              ) : null}
            </Fragment>
          ))}
        </Grid>
      )}
    </SectionFrame>
  );
}
