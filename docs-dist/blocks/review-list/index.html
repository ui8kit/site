    Review List

    Customer reviews with ratings.

    Install

    npx ui8kit add review-list

    Pulls in: block, grid, group, media-frame, rating, section-frame, stack, text, title, utils

    Code

    Svelte

    Point the CLI at this runtime once, then add the item:

    npx ui8kit init --yes --framework svelte --dir src --registry-url https://registry.buildy.tw/svelte --strict-cdn
    npx ui8kit add review-list

export type ReviewListProps = {
  id: string;
  title: string;
  reviews: Review[];
  labels: ReviewLabels;
  layout?: "list" | "grid";
  item?: "card" | "plain";
  avatar?: "none" | "image";

  density?: "default" | "roomy";

export function ReviewList({
  id,
  title,
  reviews,
  labels,
  layout = "list",
  item = "card",
  avatar = "none",
  tone = "default",
  density = "default",
}: ReviewListProps) {
  return (
    <SectionFrame
      id={id}
      title={title}
      tone={tone}
      density={density}
      measure={(((layout as string) === "list") ? "2xl" : "default")}
    >
      <Grid cols={(((layout as string) === "grid") ? "1-3" : "1")} className="gap-4" data-testid="items">
        {(reviews ?? []).map((r) => (
          <Block key={r.id}
            tag="article"
            className={cn(
              "flex h-full flex-col gap-4",
              (
                (item as string) === "card" ? "rounded bg-card p-6 inset-ring inset-ring-border"
                : "border-b border-border pb-4"
              )
            )}
            data-testid="item"
          >
            <Rating value={r.rating} label={tpl(labels.ratingLabel, r.rating)} size={16} />
            <Title as={3} size="base" weight="bold">
              {r.title}
            </Title>
            <Text size="sm" className="text-muted-foreground">
              {r.text}
            </Text>
            <Group gap="4" align="center" className="mt-auto">
              {((avatar as string) === "image") ? (
                <MediaFrame alt={r.author} ratio="1/1" className="size-10 shrink-0 rounded-full" />
              ) : null}
              <Stack gap="0">
                <Text size="sm" weight="semibold">
                  {r.author}
                </Text>
                <Text size="xs" className="text-muted-foreground">
                  {r.date}
                </Text>
              </Stack>
            </Group>
          </Block>
        ))}
      </Grid>
    </SectionFrame>
  );
}
