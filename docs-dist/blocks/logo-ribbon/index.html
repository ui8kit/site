/** A heading over one centered row of logo slots (LogoSlot); the row and slot sizes come from the default preset. */
import { type ReactNode } from "react";
import { Group } from "../primitives/group/group";
import { Stack } from "../primitives/stack/stack";
import { Title } from "../primitives/title/title";
import { LogoSlot } from "../components/logo-slot";
import { SectionFrame } from "../components/section-frame";
import { defaults } from "../presets/defaults";
import { cn } from "../utils";
import type { Logo } from "./logo-cloud.types";

export type LogoRibbonProps = {
  id: string;
  title: string;
  logos: Logo[];
  tone?: "default" | "muted";
  density?: "roomy" | "default" | "compact";
};

export function LogoRibbon({ id, title, logos, tone = "default", density = "default" }: LogoRibbonProps) {
  return (
    <SectionFrame id={id} title="" tone={tone} density={density} measure="fill" align="center" className="py-0">
      <Stack gap="6" align="center" data-testid="logos-block">
        <Title as={2} size="xl" weight="semibold" className="leading-7" id={(id + "-title")}>
          {title}
        </Title>
        <Group
          gap="0"
          justify="center"
          align="start"
          className={cn("flex w-full flex-wrap", defaults.logos.row)}
          data-testid="items"
        >
          {(logos ?? []).map((logo) => (
            <LogoSlot key={logo.id} name={logo.name} icon={logo.icon} />
          ))}
        </Group>
      </Stack>
    </SectionFrame>
  );
}
