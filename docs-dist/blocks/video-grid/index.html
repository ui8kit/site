    Video Grid

    Customer videos as thumbnails with a play button. Playing is a port contract (VideoEmbed).

    Install

    npx ui8kit add video-grid

    Pulls in: box, button, figure-block, grid, lucide-icon, media-frame, section-frame, testimonial-grid, text, utils

    Preview

    Hear it from customers

    Code

    Point the CLI at this runtime once, then add the item:

    npx ui8kit init --yes --framework svelte --dir src --registry-url https://registry.buildy.tw/svelte --strict-cdn
    npx ui8kit add video-grid

import { Box } from "../primitives/box/box";

export type VideoGridProps = {
  id: string;
  title: string;
  quotes: Quote[];
  playLabel: string;
  byLine: string;
  cols?: "1-2" | "1-3";
  caption?: "below" | "overlay";
  tone?: "default" | "muted";
  density?: "default" | "compact";
  landmarkLabel?: string;
  onPlay?: (id: string) => void;

export function VideoGrid({
  id,
  title,
  quotes,
  playLabel,
  byLine,
  cols = "1-3",
  caption = "below",
  tone = "default",
  density = "default",
  landmarkLabel,
  onPlay,
}: VideoGridProps) {
  return (
    <SectionFrame id={id} title={title} tone={tone} density={density} landmarkLabel={landmarkLabel}>
      <Grid cols={cols} className="gap-4" data-testid="items">
        {(quotes ?? []).map((q) => (
          <FigureBlock key={q.id} className="m-0 flex flex-col gap-4" testid="item">
            <Box className="relative">
              <MediaFrame alt={q.author} />
              <Button
                variant="default"
                size="default"
                type="button"
                className="absolute top-1/2 left-1/2 size-12 -translate-x-1/2 -translate-y-1/2 rounded p-0"
                aria-label={tpl(playLabel, q.author)}
                data-testid="play"
                onClick={() => onPlay?.(q.id)}
              >
                <LucideIcon name="play" size={20} />
              </Button>
              {((caption as string) === "overlay") ? (
                <Box className="absolute inset-x-0 bottom-0 rounded bg-background p-4" data-testid="caption">
                  <Text size="sm" weight="semibold">
                    {q.author}
                  </Text>
                  <Text size="xs" className="text-muted-foreground">
                    {tpl(byLine, q.role, q.company)}
                  </Text>
                </Box>
              ) : null}
            </Box>
            {((caption as string) === "below") ? (
              <Text tag="figcaption" data-testid="caption">
                <Text size="sm" weight="semibold">
                  {q.author}
                </Text>
                <Text size="xs" className="text-muted-foreground">
                  {tpl(byLine, q.role, q.company)}
                </Text>
              </Text>
            ) : null}
          </FigureBlock>
        ))}
      </Grid>
    </SectionFrame>
  );
}
