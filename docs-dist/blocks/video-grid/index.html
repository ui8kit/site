    Video Grid

    Customer videos as thumbnails with a play button. Playing is a port contract (VideoEmbed).

    Install

    npx ui8kit add video-grid

    Pulls in: block, box, button, grid, lucide-icon, media-frame, section-frame, testimonial-grid, text, utils

    Preview

    Hear it from customers

/**
 * Video testimonials as thumbnails: a picture, a play button, the name and the company. The player is a port

import { type ReactNode } from "react";
import { Block } from "../primitives/block/block";
import { Box } from "../primitives/box/box";
import { Button } from "../primitives/button/button";
import { Grid } from "../primitives/grid/grid";
import { LucideIcon } from "../primitives/lucide-icon/lucide-icon";
import { Text } from "../primitives/text/text";
import { MediaFrame } from "../components/media-frame";
import { SectionFrame } from "../components/section-frame";
import { tpl } from "../utils";
import type { Quote } from "./testimonial-grid.types";

export type VideoGridProps = {
  id: string;
  title: string;
  quotes: Quote[];
  playLabel: string;
  byLine: string;
  cols?: "1-2" | "1-3";
  caption?: "below" | "overlay";
  tone?: "default" | "muted";
  density?: "default" | "compact";
  onPlay?: (id: string) => void;
};

export function VideoGrid({
  id,
  title,
  quotes,
  playLabel,
  byLine,
  cols = "1-3",
  caption = "below",
  tone = "default",
  density = "default",
  onPlay,
}: VideoGridProps) {
  return (
    <SectionFrame id={id} title={title} tone={tone} density={density}>
      <Grid cols={cols} className="gap-4" data-testid="items">
        {(quotes ?? []).map((q) => (
          <Block key={q.id} tag="figure" className="m-0 flex flex-col gap-4" data-testid="item">
            <Box className="relative">
              <MediaFrame alt={q.author} ratio="16/9" />
              <Button
                variant="default"
                size="default"
                type="button"
                className="absolute top-1/2 left-1/2 size-12 -translate-x-1/2 -translate-y-1/2 rounded p-0"
                aria-label={tpl(playLabel, q.author)}
                data-testid="play"
                onClick={() => onPlay?.(q.id)}
              >
                <LucideIcon name="play" size={20} />
              </Button>
              {((caption as string) === "overlay") ? (
                <Box className="absolute inset-x-0 bottom-0 rounded bg-background p-4" data-testid="caption">
                  <Text size="sm" weight="semibold">
                    {q.author}
                  </Text>
                  <Text size="xs" className="text-muted-foreground">
                    {tpl(byLine, q.role, q.company)}
                  </Text>
                </Box>
              ) : null}
            </Box>
            {((caption as string) === "below") ? (
              <Block tag="figcaption" data-testid="caption">
                <Text size="sm" weight="semibold">
                  {q.author}
                </Text>
                <Text size="xs" className="text-muted-foreground">
                  {tpl(byLine, q.role, q.company)}
                </Text>
              </Block>
            ) : null}
          </Block>
        ))}
      </Grid>
    </SectionFrame>
  );
}
