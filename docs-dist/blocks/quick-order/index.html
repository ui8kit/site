    Quick Order

    Compact rows with a one-click order button and a phone field.

    Install

    npx ui8kit add quick-order

    Pulls in: button, compose, dish-showcase, empty-state, field, group, inline, input, kit-box, kit-button, kit-input, kit-text, list, list-item, lucide-icon, section-frame, stack, text, utils

 * the dish, under a phone field that the one click uses. Axes: phone (show, hide), tone.
 */
import { type ReactNode } from "react";
import { Button, Group, Inline, Input, List, ListItem, Stack, Text } from "../compose";
import { LucideIcon } from "../icons/LucideIcon";
import { EmptyState } from "../components/empty-state";
import { Field } from "../components/field";
import { SectionFrame } from "../components/section-frame";
import { fmtMoney, tpl } from "../utils";
import type { Dish, FoodLabels } from "./dish-showcase.types";

export type QuickOrderProps = {
  id: string;
  title: string;
  lead?: string;
  dishes: Dish[];
  empty?: string;
  labels: FoodLabels;
  phone?: "show" | "hide";
  tone?: "default" | "muted";
  landmarkLabel?: string;
};

export function QuickOrder({
  id,
  title,
  lead = "",
  dishes,
  empty = "",
  labels,
  phone = "show",
  tone = "default",
  landmarkLabel,
}: QuickOrderProps) {
  return (
    <SectionFrame id={id} title={title} lead={lead} tone={tone} landmarkLabel={landmarkLabel}>
      {((dishes?.length ?? 0) === 0) ? (
        <EmptyState icon="zap" title={empty} />
      ) : (
        <Stack gap="4" align="stretch" className="mx-auto w-full max-w-3xl">
          {((phone as string) === "show") ? (
            <Field label={labels.phone} htmlFor={tpl("{0}-phone", id)}>
              <Input type="tel" name={tpl("{0}-phone", id)} id={tpl("{0}-phone", id)} />
            </Field>
          ) : null}
          <List tag="ul" className="flex flex-col" data-testid="items">
            {(dishes ?? []).map((d) => (
              <ListItem key={d.id} className="flex flex-wrap items-center gap-4 border-b border-border py-4" data-testid="item">
                <Stack gap="0" className="min-w-0 flex-1">
                  <Group gap="2" align="baseline">
                    <Text size="base" weight="semibold">
                      {d.name}
                    </Text>
                    <Text size="xs" className="text-muted-foreground">
                      {d.weight}
                    </Text>
                  </Group>
                  <Text size="sm" className="text-muted-foreground">
                    {d.summary}
                  </Text>
                </Stack>
                <Inline className="text-base font-bold">
                  {fmtMoney(d.price, d.currency)}
                </Inline>
                <Button variant="default" size="sm" type="button" aria-label={tpl(labels.orderOf, d.name)} data-testid="order">
                  <LucideIcon name="zap" size={16} />
                  {labels.order}
                </Button>
              </ListItem>
            ))}
          </List>
        </Stack>
      )}
    </SectionFrame>
  );
}
