    Hero Classic

    Heading, lead, one or two actions and optional media.

    Install

    npx ui8kit add hero-classic

    Pulls in: action-group, backdrop, box, button, code-window, cta-actions, grid, group, input, label, media-frame, section-frame, topic-chips, utils

templ heroClassicFill1(p HeroClassicProps, vTitle any, vMedia any, vMediaKind any) {
	if uiutils.T(uiutils.And(uiutils.Eq(vMediaKind, "code"), uiutils.Get(vMedia, "lines"), uiutils.Get(vMedia, "codeTitle"), uiutils.Get(vMedia, "prompt"))) {
		@CodeWindow(CodeWindowProps{Title: uiutils.Get(vMedia, "codeTitle"), Prompt: uiutils.Get(vMedia, "prompt"), Lines: uiutils.Get(vMedia, "lines")})
	}
	if uiutils.Eq(vMediaKind, "cluster") {
		@grid.Grid(grid.GridProps{Cols: "3", Class: "items-end gap-4", Attrs: uiutils.Attrs("data-testid", "cluster")}) {
			@box.Box(box.BoxProps{Class: uiutils.S(uiutils.Cnv("w-full", ""))}) {
				@MediaFrame(MediaFrameProps{Alt: vTitle, Ratio: "3/4"})
			}
			@box.Box(box.BoxProps{Class: uiutils.S(uiutils.Cnv("w-full", "-translate-y-4"))}) {
				@MediaFrame(MediaFrameProps{Alt: vTitle, Ratio: "4/3"})
			}
			@box.Box(box.BoxProps{Class: uiutils.S(uiutils.Cnv("w-full", ""))}) {
				@MediaFrame(MediaFrameProps{Alt: vTitle, Ratio: "3/4"})
			}
		}
	}
	if uiutils.T(uiutils.Or(uiutils.Eq(vMediaKind, "image"))) {
		@MediaFrame(MediaFrameProps{Alt: vTitle, Ratio: "4/3"})
	}
}
