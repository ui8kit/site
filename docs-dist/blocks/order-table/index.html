/**
 * The order history as a table or as cards: number, date, items, total and a status pill, with an optional
 * pager. Axes: layout (table, cards), pager (show, hide), tone, density.
 */
import { type ReactNode } from "react";
import { Card } from "../primitives/card/card";
import { CardContent } from "../primitives/card/card-content";
import { Stack } from "../primitives/stack/stack";
import { Text } from "../primitives/text/text";
import { OrderRows } from "../components/order-rows";
import { Pager } from "../components/pager";
import { Pill } from "../components/pill";
import { SectionFrame } from "../components/section-frame";
import { at, fmtMoney, tpl } from "../utils";
import type { Order, OrderLabels } from "./order-table.types";

export type OrderTableProps = {
  id: string;
  title: string;
  orders: Order[];
  statuses: Record<string, string>;
  labels: OrderLabels;
  layout?: "table" | "cards";
  pager?: "show" | "hide";
  tone?: "default" | "muted";
  density?: "default" | "compact";
};

export function OrderTable({
  id,
  title,
  orders,
  statuses,
  labels,
  layout = "table",
  pager = "show",
  tone = "default",
  density = "default",
}: OrderTableProps) {
  return (
    <SectionFrame id={id} title={title} tone={tone} density={density}>
      <Stack gap="4" align="stretch">
        {((layout as string) === "table") ? (
          <OrderRows orders={orders} statuses={statuses} labels={labels} density={density} />
        ) : (
          <Stack gap="4" align="stretch" data-testid="order-cards">
            {(orders ?? []).map((o) => (
              <Card key={o.id} variant="default" data-testid="item">
                <CardContent className="flex flex-wrap items-center justify-between gap-4">
                  <Stack gap="0">
                    <Text size="sm" weight="semibold">
                      {o.number}
                    </Text>
                    <Text size="xs" className="text-muted-foreground">
                      {o.date}
                    </Text>
                  </Stack>
                  <Text size="sm">
                    {tpl(labels.itemsCount, o.items)}
                  </Text>
                  <Text size="sm" weight="bold">
                    {fmtMoney(o.total, o.currency)}
                  </Text>
                  <Pill
                    tone={
                      o.status === "paid" ? "green"
                      : o.status === "shipped" ? "blue"
                      : o.status === "pending" ? "amber"
                      : "gray"
                    }
                  >
                    {at(statuses, o.status)}
                  </Pill>
                </CardContent>
              </Card>
            ))}
          </Stack>
        )}
        {((pager as string) === "show") ? (
          <Pager
            page={1}
            pages={3}
            summary={tpl(labels.summary, 1, 4, 12)}
            previousLabel={labels.previous}
            nextLabel={labels.next}
          />
        ) : null}
      </Stack>
    </SectionFrame>
  );
}
