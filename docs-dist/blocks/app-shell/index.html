    App Shell

    The frame of a signed-in app with a first screen: a sidebar, a top bar, or three panes.

    Install

    npx ui8kit add app-shell

    Pulls in: activity-feed, app-main, block, chart-panel, container, dashboard-overview, grid, list, metric-row, nav-bar, panel, sidebar-nav, stack, text, utils

export function AppShell({
  id,
  title,
  nav,
  user,
  search,
  metrics,
  chart,
  activity,
  layout = "sidebar",
  density = "default",
}: AppShellProps) {
  return (
    <Block
      tag="section"
      className="bg-background"
      aria-labelledby={tpl("{0}-title", id)}
      data-testid="app-shell"
      data-layout={layout}
    >
      {((layout as string) === "top") ? (
        <Stack gap="6" align="stretch">
          <NavBar
            label={nav.label}
            brand={nav.brand}
            links={nav.links}
            current={nav.current}
            search={search}
            actions={{ mode: "none" }}
            layout="search"
            over="none"
          />
          <Container className={cn("mx-auto w-full max-w-6xl px-4", "pb-8")} data-slot="container">
            <AppMain id={id} title={title} metrics={metrics} chart={chart} cols="1-4" activity={activity} density={density} />
          </Container>
        </Stack>
      ) : (
        <Container className={cn("mx-auto w-full max-w-6xl px-4", "py-8")} data-slot="container">
          {((layout as string) === "sidebar") ? (
            <Grid cols="1" className="items-start gap-6 md:grid-cols-[16rem_minmax(0,1fr)]">
              <SidebarNav
                label={nav.label}
                brand={nav.brand}
                links={nav.links}
                current={nav.current}
                userName={user.name}
                userEmail={user.email}
                width="full"
                user="show"
                density={density}
              />
              <AppMain id={id} title={title} metrics={metrics} chart={chart} cols="1-2" activity={activity} density={density} />
            </Grid>
          ) : (
            <Grid cols="1-3" className="items-start gap-4">
              <SidebarNav
                label={nav.label}
                brand={nav.brand}
                links={nav.links}
                current={nav.current}
                userName={user.name}
                userEmail={user.email}
                width="compact"
                user="show"
                density={density}
              />
              <Panel title={activity.title} description={activity.text} id={tpl("{0}-list", id)}>
                <List className="m-0 flex list-none flex-col gap-4 p-0" data-testid="items">
                  {(activity.events ?? []).map((e) => (
                    <ListItem key={e.id} className="flex flex-col gap-2" data-testid="item">
                      <Text size="sm" weight="semibold">
                        {e.title}
                      </Text>
                      <Text size="xs" className="text-muted-foreground">
                        {e.time}
                      </Text>
                    </ListItem>
                  ))}
                </List>
              </Panel>
              <AppMain id={id} title={title} metrics={metrics} chart={chart} cols="1-2" density="compact" />
            </Grid>
          )}

      )}
    </Block>
  );
}
