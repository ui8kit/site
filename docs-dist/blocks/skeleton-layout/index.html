    Skeleton Layout

    Grey blocks in the shape of what is coming: cards, a list or an article.

    Install

    npx ui8kit add skeleton-layout

    Pulls in: block, grid, group, section-frame, skeleton, stack, utils

/**
 * The loading state of a page: grey blocks in the shape of the content that is coming (cards, a list of rows or
 * an article). Axes: shape (cards, list, article), tone. The label tells assistive technology that the content
 * is loading. The shimmer itself belongs to the Skeleton part.
 */
import { type ReactNode } from "react";
import { Block } from "../primitives/block/block";
import { Grid } from "../primitives/grid/grid";
import { Group } from "../primitives/group/group";
import { Skeleton } from "../primitives/skeleton/skeleton";
import { Stack } from "../primitives/stack/stack";
import { SectionFrame } from "../components/section-frame";
import type { Line } from "./skeleton-layout.types";

export type SkeletonLayoutProps = {
  id: string;
  title: string;
  label: string;
  lines: Line[];
  shape?: "cards" | "list" | "article";
  tone?: "default" | "muted";
};

export function SkeletonLayout({
  id,
  title,
  label,
  lines,
  shape = "cards",
  tone = "default",
}: SkeletonLayoutProps) {
  return (
    <SectionFrame id={id} title={title} tone={tone} density="default">
      <Block tag="div" className="w-full" role="status" aria-label={label} data-testid="skeleton" data-shape={shape}>
        {((shape as string) === "cards") ? (
          <Grid cols="1-3" className="gap-4">
            {(lines ?? []).map((l) => (
              <Stack key={l.id} gap="4" className="rounded bg-card inset-ring inset-ring-border p-4" data-testid="item">
                <Skeleton className="aspect-video w-full" />
                <Skeleton className="h-4 w-2/3" />
                <Skeleton className="h-4 w-full" />
              </Stack>
            ))}
          </Grid>
        ) : ((shape as string) === "list") ? (
          <Stack gap="4" align="stretch">
            {(lines ?? []).map((l) => (
              <Group key={l.id} gap="4" align="center" data-testid="item">
                <Skeleton className="size-10 shrink-0 rounded-full" />
                <Stack gap="2" className="flex-1">
                  <Skeleton className="h-4 w-1/3" />
                  <Skeleton className="h-4 w-full" />
                </Stack>
              </Group>
            ))}
          </Stack>
        ) : (
          <Stack gap="4" align="stretch" className="mx-auto max-w-2xl">
            <Skeleton className="h-8 w-3/4" />
            {(lines ?? []).map((l) => (
              <Stack key={l.id} gap="2" data-testid="item">
                <Skeleton className="h-4 w-full" />
                <Skeleton className="h-4 w-5/6" />
              </Stack>
            ))}
          </Stack>
        )}
      </Block>
    </SectionFrame>
  );
}
