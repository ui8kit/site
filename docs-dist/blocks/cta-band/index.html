    Cta Band

    A closing call to action.

    Install

    npx ui8kit add cta-band

    Pulls in: block, card, container, cta-actions, cta-banner, media-frame, quote-proof, section-frame, stack, text, title, utils

    npx ui8kit init --yes --framework svelte --dir src --registry-url https://registry.buildy.tw/svelte --strict-cdn

export type CtaBandProps = {
  id: string;
  title: string;
  text: string;
  actions: ActionSet;
  quote?: CtaQuote;
  layout?: "card" | "band" | "split" | "row" | "banner";
  tone?: "default" | "muted";
  proof?: "none" | "quote";
  density?: "roomy" | "default" | "compact";

export function CtaBand({
  id,
  title,
  text,
  actions,
  quote,
  layout = "card",
  tone = "muted",
  proof = "none",
  density = "default",
}: CtaBandProps) {
  const center = ((layout as string) === "band") || ((layout as string) === "banner");
  return (
    <>
      {((layout as string) === "banner") ? (
        <CtaBanner id={id} title={title} text={text} cta={actions.cta} href={actions.href} tone={tone} />
      ) : ((layout as string) === "row") ? (
        <Block
          tag="section"
          className="py-8"
          aria-labelledby={tpl("{0}-title", id)}
          data-testid="section"
          data-tone={tone}
          data-density={density}
        >
          <Container className="mx-auto w-full max-w-6xl px-4" data-slot="container">
            <Card variant="muted">
              <CardContent className="flex flex-col gap-4 md:flex-row md:items-center md:justify-between">
                <Stack gap="2" className="min-w-0">
                  <Title as={2} size="2xl" weight="bold" id={tpl("{0}-title", id)}>
                    {title}
                  </Title>
                  <Text size="base" className="text-muted-foreground">
                    {text}
                  </Text>
                </Stack>
                {((proof as string) === "quote") ? (
                  <QuoteProof quote={quote?.text} quoteBy={quote?.by} center={center} />
                ) : null}
                <CtaActions id={id} actions={actions} center={center} />
              </CardContent>
            </Card>
          </Container>
        </Block>
      ) : (
        <SectionFrame
          id={id}
          title={title}
          lead={text}
          layout={(((layout as string) === "split") ? "split" : "stack")}
          align={(center ? "center" : "start")}
          boxed={((layout as string) === "card")}
          media={{ show: ((layout as string) === "split") }}
          tone={tone}
          density={density}
          aside={(
            <MediaFrame alt={title} ratio="4/3" />
          )}
        >
          {((proof as string) === "quote") ? (
            <QuoteProof quote={quote?.text} quoteBy={quote?.by} center={center} />
          ) : null}
          <CtaActions id={id} actions={actions} center={center} />
        </SectionFrame>
      )}
    </>
  );
}
