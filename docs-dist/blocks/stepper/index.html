    Stepper

    A multi-step form header with the current step.

    Install

    npx ui8kit add stepper

    Pulls in: button, group, iconbadge, list, lucide-icon, section-frame, stack, steps-list, text, title, utils

  current?: number;
  stepLabel: string;
  back: string;
  next: string;
  layout?: "horizontal" | "vertical";
  marker?: "number" | "icon";
  summary?: "show" | "hide";

export function Stepper({
  id,
  title,
  steps,

                variant={(((i + 1) < (current + 1)) ? "primary" : "outline")}
