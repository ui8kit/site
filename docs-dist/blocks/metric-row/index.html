/**

import { Fragment, type ReactNode } from "react";
import { Block } from "../primitives/block/block";
import { Grid } from "../primitives/grid/grid";
import { Group } from "../primitives/group/group";
import { LucideIcon } from "../primitives/lucide-icon/lucide-icon";
import { Stack } from "../primitives/stack/stack";
import { Text } from "../primitives/text/text";
import { SectionFrame } from "../components/section-frame";
import { StatTile } from "../components/stat-tile";
import type { Metric } from "./metric-row.types";

export type MetricRowProps = {
  id: string;
  title: string;
  metrics: Metric[];
  cols?: "2" | "4";
  item?: "tile" | "compact";
  tone?: "default" | "muted";
  density?: "default" | "compact";
};

export function MetricRow({
  id,
  title,
  metrics,
  cols = "4",
  item = "tile",
  tone = "default",
  density = "default",
}: MetricRowProps) {
  return (
    <SectionFrame id={id} title={title} tone={tone} density={density}>
      <Grid cols={(((cols as string) === "2") ? "1-2" : "1-4")} className="gap-4" data-testid="items">
        {(metrics ?? []).map((m) => (
          <Fragment key={m.id}>
            {((item as string) === "tile") ? (
              <Block tag="div" data-testid="item">
                <StatTile label={m.label} value={m.value} icon={m.icon} hint={m.hint} />
              </Block>
            ) : (
              <Stack gap="0" className="rounded bg-card p-4 inset-ring inset-ring-border" data-testid="item">
                <Group gap="2" align="center" className="text-muted-foreground">
                  <LucideIcon name={m.icon} size={16} />
                  <Text size="xs">
                    {m.label}
                  </Text>
                </Group>
                <Text size="2xl" weight="bold">
                  {m.value}
                </Text>
                <Text size="xs" className="text-muted-foreground">
                  {m.hint}
                </Text>
              </Stack>
            )}
          </Fragment>
        ))}
      </Grid>
    </SectionFrame>
  );
}
