    block / marketing

    Page Trail

    Breadcrumbs over the badge, the heading and the lead; with one large figure, a note card or four key facts at the end.

    Install

    npx ui8kit add page-trail

    Pulls in: block, box, container, nav-bar, page-aside, page-crumbs, section-header, stack, stats-band, utils

    Code

    Svelte

    Point the CLI at this runtime once, then add the item:

    npx ui8kit init --yes --framework svelte --dir src --registry-url https://registry.buildy.tw/svelte --strict-cdn
    npx ui8kit add page-trail

  id: string;

  lead?: string;

export function PageTrail({
  id,
  title,
  lead = "",
  badge = "",
  label,
  crumbs,
  extras,
  aside = "none",
  level = 1,
  tone = "default",
}: PageTrailProps) {
  const titleId = id + "-title";
  return (
    <Block
      tag="section"
      className={cn("py-8", (((tone as string) === "muted") ? "bg-muted" : ""))}
      aria-labelledby={titleId}
      data-testid="section"
      data-tone={tone}
      data-aside={aside}
    >
      <Container className={cn("mx-auto w-full max-w-6xl px-4", "flex flex-col gap-6")} data-slot="container">
        <PageCrumbs label={label} crumbs={crumbs} />
        <Box
          className={
            (aside as string) === "facts" ? "grid grid-cols-1 items-center gap-8 lg:grid-cols-2"
            : "flex flex-col gap-6 md:flex-row md:items-end md:justify-between"
          }
        >
          <Stack gap="4" align="start" className="min-w-0">
            <SectionHeader titleId={titleId} title={title} heading={{ level, badge }} lead={lead} />
          </Stack>
          {((aside as string) !== "none") ? (
            <PageAside extras={extras} mode={aside} />
          ) : null}
        </Box>
      </Container>
    </Block>
  );
}
