export function PricingCard({
  plan,
  labels,
  current,
  emphasis = "auto",
  density = "default",
  billing = "monthly",
  footer,
  onSelect,
}: PricingCardProps) {
  const stand = plan.highlighted && ((emphasis as string) === "auto");
  const yearlyOn = ((billing as string) === "annual") && (plan.yearly > 0);
  const amount = yearlyOn ? plan.yearly : plan.price;
  return (
    <Card
      variant="default"
      className="flex h-full flex-col"
      data-highlighted={(stand ? "true" : "false")}
      data-current={(current ? "true" : "false")}
    >
      <CardHeader className="flex flex-col gap-2">
        <Box className="flex min-h-6 items-center" data-slot="badge-slot">
          {stand ? (
            <Badge variant="default" size="default">
              {labels.popular}
            </Badge>
          ) : null}
        </Box>
        <CardTitle as={3}>
          {plan.name}
        </CardTitle>
        <CardDescription>
          {plan.tagline}
        </CardDescription>
      </CardHeader>
      <CardContent className={cn("flex flex-1 flex-col", (((density as string) === "compact") ? "gap-2" : "gap-4"))}>
        <Group gap="2" align="baseline">
          <Title as={2} size="4xl" weight="bold">
            {(amount === 0) ? (
              <>
                {labels.free}
              </>
            ) : (
              <>
                {fmtMoney(amount, plan.currency)}
              </>
            )}
          </Title>
          {(amount > 0) ? (
            <Text size="sm" className="text-muted-foreground">
              {(yearlyOn ? labels.perYear : labels.perMonth)}
            </Text>
          ) : null}
        </Group>
        <Text size="sm" weight="medium">
          {tpl(labels.seats, plan.seats)}
        </Text>
        <Text size="sm" className="text-muted-foreground">
          {tpl(labels.storage, plan.storage)}
        </Text>
        <List tag="ul" className="flex flex-col gap-2 text-sm">
          {(plan.features ?? []).map((feature) => (
            <ListItem key={feature} className="flex gap-2">
              {feature}
            </ListItem>
          ))}
        </List>
      </CardContent>
      <CardFooter className="flex flex-wrap items-center gap-2">
        <Button
          variant={(stand ? "default" : "outline")}
          size="default"
          data-testid="plan-select"
          onClick={() => onSelect?.(plan.id)}
        >
          {labels.cta}
        </Button>
        {footer?.(plan)}
      </CardFooter>
    </Card>
  );
}
