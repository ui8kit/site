const stack: object[] = [];   // topmost dismissable layer handles Escape first (menu inside off-canvas sidebar, …)

/** Escape and pointer-down outside the node call onDismiss. Returns the cleanup. */
export function dismissable(node: HTMLElement, opts: () => { onDismiss: (reason: DismissReason) => void; ignore?: () => (Element | null | undefined)[]; outside?: boolean }): () => void {
  {
    const layer = {};
    stack.push(layer);
    const { onDismiss, ignore = () => [], outside = true } = opts();
    const onKey = (e: KeyboardEvent) => {
      if (e.key !== "Escape" || e.defaultPrevented || stack[stack.length - 1] !== layer) return;
      e.preventDefault(); e.stopPropagation();
      onDismiss("escape");
    };
    const onDown = (e: PointerEvent) => {
      if (!outside || stack[stack.length - 1] !== layer) return;
      const t = e.target as Node | null;
      if (!t || node.contains(t) || ignore().some((el) => el?.contains(t))) return;
      onDismiss("outside");
    };
    document.addEventListener("keydown", onKey, true);
    document.addEventListener("pointerdown", onDown, true);
    return () => {
      document.removeEventListener("keydown", onKey, true);
      document.removeEventListener("pointerdown", onDown, true);
      const i = stack.indexOf(layer); if (i >= 0) stack.splice(i, 1);
    };
  }
}

/** Handles ArrowUp/Down/Left/Right, Home, End. Returns true when the key was consumed. */
export function moveFocus(e: KeyboardEvent, items: HTMLElement[], { orientation = "vertical", loop = true, activate = false }: RovingOptions = {}): boolean {
  if (!items.length || e.altKey || e.ctrlKey || e.metaKey) return false;
  const prev = orientation !== "horizontal" ? ["ArrowUp"] : [];
  const next = orientation !== "horizontal" ? ["ArrowDown"] : [];
  if (orientation !== "vertical") { prev.push("ArrowLeft"); next.push("ArrowRight"); }
  const cur = items.indexOf(document.activeElement as HTMLElement);
  let idx = -1;
  if (next.includes(e.key)) idx = cur < 0 ? 0 : cur + 1;
  else if (prev.includes(e.key)) idx = cur < 0 ? items.length - 1 : cur - 1;
  else if (e.key === "Home") idx = 0;
  else if (e.key === "End") idx = items.length - 1;
  else return false;
  if (idx >= items.length) idx = loop ? 0 : items.length - 1;
  if (idx < 0) idx = loop ? items.length - 1 : 0;
  e.preventDefault();
  const el = items[idx];
  el.focus();
  if (activate) el.click();
  return true;
}

/** type-ahead: focus the next item whose text starts with the typed characters (reset after 600 ms) */
export function typeahead() {
  let buf = "", t: ReturnType<typeof setTimeout> | undefined;
  return (e: KeyboardEvent, items: HTMLElement[]): boolean => {
    if (e.key.length !== 1 || e.ctrlKey || e.metaKey || e.altKey || e.key === " ") return false;
    buf += e.key.toLowerCase();
    clearTimeout(t); t = setTimeout(() => (buf = ""), 600);
    const start = Math.max(0, items.indexOf(document.activeElement as HTMLElement) + (buf.length === 1 ? 1 : 0));
    const ordered = [...items.slice(start), ...items.slice(0, start)];
    const hit = ordered.find((el) => (el.textContent ?? "").trim().toLowerCase().startsWith(buf));
    if (hit) { hit.focus(); e.preventDefault(); return true; }
    return false;
  };
}

/** the page numbers a pager shows: the first, the last and the neighbours of the current page; null is a gap */
export function pageWindow(page: number, pages: number): (number | null)[] {
  const keep = new Set([1, pages, page - 1, page, page + 1].filter((n) => n >= 1 && n <= pages));
  const out: (number | null)[] = [];
  let prev = 0;
  for (const n of [...keep].sort((a, b) => a - b)) {
    if (n - prev > 1) out.push(null);
    out.push(n);
    prev = n;
  }
  return out;
}

/** a responsive column variant of Grid capped by the number of items, so no column stays empty: ("1-4", 3) -> "1-3", ("2-6", 4) -> "2-4" */
export function fitCols(cols: string, n: number): string {
  const m = /^(\d+)-(\d+)$/.exec(cols);
  if (!m) return cols;
  const lo = Number(m[1]), hi = Number(m[2]);
  if (n >= hi) return cols;
  const top = Math.max(lo, n);
  const known = ["1-2", "1-3", "1-4", "2-3", "2-4", "2-6", "3-6"];
  for (let h = top; h > lo; h--) if (known.includes(`${lo}-${h}`)) return `${lo}-${h}`;
  return String(Math.min(lo, Math.max(n, 1)));
}

/**
 * A shadcn/cva-style variant table, without the dependency (same idea as `cva` from class-variance-authority).
 *
 *   const badgeVariants = variants({
 *     base: "inline-flex items-center",
 *     variants: { size: { sm: "h-5 px-2", lg: "h-8 px-3" } },
 *     defaultVariants: { size: "sm" },
 *     compoundVariants: [{ size: "lg", shape: "rounded", class: "rounded-lg" }],
 *   });
 *   badgeVariants({ size: "lg" }); // "inline-flex items-center h-8 px-3"
 *
 * Classes come in this order: base, each variant in table order, then the matching compound variants.
 * tailwind-merge (`cn`) resolves conflicts, the later class wins. A compound value may list alternatives: "32|lg".
 */
export type VariantTable = {
  base?: string;
  variants: Record<string, Record<string, string>>;
  defaultVariants: Record<string, string>;
  compoundVariants?: Array<{ class: string } & Record<string, string>>;
};

/** the props a table accepts: one optional value per variant */
export type VariantSelection<T extends VariantTable> = {
  [K in keyof T["variants"]]?: (keyof T["variants"][K] & string) | undefined;
};

/** a variant function: call it with the selected values, read `.table` for the types */
export type VariantFn<T extends VariantTable> = ((selection?: VariantSelection<T>) => string) & { table: T };

/** the values of one variant of a table: `VariantKey<typeof badgeVariants, "size">` is `"sm" | "lg"` */
export type VariantKey<F extends { table: VariantTable }, K extends keyof F["table"]["variants"]> = Exclude<
  keyof F["table"]["variants"][K] & string,
  ""
>;

export function variants<T extends VariantTable>(table: T): VariantFn<T> {
  const pick = (selection: Record<string, string | undefined>, key: string): string =>
    (selection[key] ?? "").trim() || (table.defaultVariants[key] ?? "").trim();
  const fn = (selection: VariantSelection<T> = {}): string => {
    const sel = selection as Record<string, string | undefined>;
    const parts: string[] = [table.base ?? ""];
    for (const [key, choices] of Object.entries(table.variants)) {
      const choice = pick(sel, key);
      const cls = choices[choice];
      // a typo in development is an error; in production the unknown value is dropped
      if (cls === undefined && isDevEnv()) throw new Error(`[variants] unknown value "${choice}" for "${key}"`);
      parts.push(cls ?? "");
    }
    for (const compound of table.compoundVariants ?? []) {
      const matches = Object.entries(compound).every(([k, want]) => k === "class" || want.split("|").includes(pick(sel, k)));
      if (matches) parts.push(compound.class);
    }
    return cn(...parts.map((p) => p.trim()).filter(Boolean));
  };
  return Object.assign(fn, { table });
}

export function moveFocus(e: KeyboardEvent, items: HTMLElement[], { orientation = "vertical", loop = true, activate = false }: RovingOptions = {}): boolean {
  if (!items.length || e.altKey || e.ctrlKey || e.metaKey) return false;
  const prev = orientation !== "horizontal" ? ["ArrowUp"] : [];
  const next = orientation !== "horizontal" ? ["ArrowDown"] : [];

  let idx = -1;
  if (next.includes(e.key)) idx = cur < 0 ? 0 : cur + 1;
  else if (prev.includes(e.key)) idx = cur < 0 ? items.length - 1 : cur - 1;
  else if (e.key === "Home") idx = 0;
  else if (e.key === "End") idx = items.length - 1;
  else return false;
  if (idx >= items.length) idx = loop ? 0 : items.length - 1;
  if (idx < 0) idx = loop ? items.length - 1 : 0;

 * A shadcn/cva-style variant table, without the dependency (same idea as `cva` from class-variance-authority).
 *
 *   const badgeVariants = variants({
 *     base: "inline-flex items-center",
 *     variants: { size: { sm: "h-5 px-2", lg: "h-8 px-3" } },
 *     defaultVariants: { size: "sm" },
 *     compoundVariants: [{ size: "lg", shape: "rounded", class: "rounded-lg" }],
 *   });
 *   badgeVariants({ size: "lg" }); // "inline-flex items-center h-8 px-3"
 *
 * Classes come in this order: base, each variant in table order, then the matching compound variants.
 * tailwind-merge (`cn`) resolves conflicts, the later class wins. A compound value may list alternatives: "32|lg".
 */
export type VariantTable = {
  base?: string;
  variants: Record<string, Record<string, string>>;
  defaultVariants: Record<string, string>;
  compoundVariants?: Array<{ class: string } & Record<string, string>>;
};

/** the props a table accepts: one optional value per variant */
export type VariantSelection<T extends VariantTable> = {
  [K in keyof T["variants"]]?: (keyof T["variants"][K] & string) | undefined;
};

/** a variant function: call it with the selected values, read `.table` for the types */
export type VariantFn<T extends VariantTable> = ((selection?: VariantSelection<T>) => string) & { table: T };

/** the values of one variant of a table: `VariantKey<typeof badgeVariants, "size">` is `"sm" | "lg"` */
export type VariantKey<F extends { table: VariantTable }, K extends keyof F["table"]["variants"]> = Exclude<
  keyof F["table"]["variants"][K] & string,
  ""
>;

export function variants<T extends VariantTable>(table: T): VariantFn<T> {
  const pick = (selection: Record<string, string | undefined>, key: string): string =>
    (selection[key] ?? "").trim() || (table.defaultVariants[key] ?? "").trim();
  const fn = (selection: VariantSelection<T> = {}): string => {
    const sel = selection as Record<string, string | undefined>;
    const parts: string[] = [table.base ?? ""];
    for (const [key, choices] of Object.entries(table.variants)) {
      const choice = pick(sel, key);
      const cls = choices[choice];
      // a typo in development is an error; in production the unknown value is dropped
      if (cls === undefined && isDevEnv()) throw new Error(`[variants] unknown value "${choice}" for "${key}"`);
      parts.push(cls ?? "");
    }
    for (const compound of table.compoundVariants ?? []) {
      const matches = Object.entries(compound).every(([k, want]) => k === "class" || want.split("|").includes(pick(sel, k)));
      if (matches) parts.push(compound.class);
    }
    return cn(...parts.map((p) => p.trim()).filter(Boolean));
  };
  return Object.assign(fn, { table });
}

		if v := strings.TrimSpace(S(sel[k])); v != "" {
			return v
		}
		return strings.TrimSpace(r.Defaults[k])
	}

	for _, k := range r.Keys {
		parts = append(parts, r.ByKey[k][pick(k)])

		ok := true
		for k, want := range c {
			if k == "class" {
				continue
			}
			hit := false
			for _, w := range strings.Split(want, "|") {
				if w == pick(k) {
					hit = true
				}
			}
			ok = ok && hit
		}
		if ok {
			parts = append(parts, c["class"])
		}
	}
	keep := []string{}
	for _, p := range parts {
		if p = strings.TrimSpace(p); p != "" {
			keep = append(keep, p)
		}
	}
	return mergeTailwind(keep...)
}

		choices, ok := v.ByKey[key]
		if !ok {
			continue
		}

		if choice == "" {
			choice = strings.TrimSpace(v.Defaults[key])
		}

			if len(choices) > 0 {

			}
			continue

				parts = append(parts, cls)
			}
		}

		// the DOM as literal class names.

        return c or str(defaults.get(k, "")).strip()

    parts = [str(r["base"]).strip()] if str(r.get("base", "")).strip() else []
    for key in r.get("keys", []):
        choices = (r.get("byKey") or {}).get(key)

            continue

        if not choice:
            if choices:

    public static function merge(array $classes): string
    {
        $tokens = [];
        foreach ($classes as $chunk) {
            foreach (preg_split('/\s+/', trim((string) $chunk)) ?: [] as $t) {
                if ($t !== '') {
                    $tokens[] = $t;
                }
            }
        }
        $kept = [];
        foreach ($tokens as $token) {
            [$prefix, $key, $removes] = self::classKey($token);
            if ($key !== '') {
                $drop = [$key => true];
                foreach ($removes as $g) {
                    $drop[$prefix . $g] = true;
                }
                $kept = array_values(array_filter($kept, fn ($it) => !isset($drop[$it[1]])));
            }
            $kept[] = [$token, $key];
        }
        return implode(' ', array_map(fn ($it) => $it[0], $kept));
    }

    private static function classKey(string $token): array
    {
        [$prefix, $core] = self::splitVariant($token);
        if ($core === '') {
            return ['', '', []];
        }
        $groups = self::utilityGroups($core);
        if (!$groups) {
            return [$prefix, '=' . $token, []];
        }
        return [$prefix, $prefix . $groups[0], array_slice($groups, 1)];
    }

    private static function splitVariant(string $token): array
    {
        $depth = 0;
        $cut = -1;
        $n = strlen($token);
        for ($i = 0; $i < $n; $i++) {
            $c = $token[$i];
            if ($c === '[') {
                $depth++;
            } elseif ($c === ']') {
                if ($depth > 0) {
                    $depth--;
                }
            } elseif ($c === ':' && $depth === 0) {
                $cut = $i;
            }
        }
        if ($cut === -1) {
            return ['', self::noBang($token)];
        }
        return [substr($token, 0, $cut + 1), self::noBang(substr($token, $cut + 1))];
    }

    private static function noBang(string $s): string
    {
        return str_starts_with($s, '!') ? substr($s, 1) : $s;
    }

    private static function utilityGroups(string $core): array

    private static function spaceAxis(string $core, string $kind, string $n): array
    {
        if (str_starts_with($core, $kind . 'x-')) {
            return ["$n-x", "$n-r", "$n-l", "$n-s", "$n-e"];
        }
        if (str_starts_with($core, $kind . 'y-')) {
            return ["$n-y", "$n-t", "$n-b"];
        }
        foreach (['t', 'r', 'b', 'l'] as $s) {
            if (str_starts_with($core, $kind . $s . '-')) {
                return ["$n-$s"];
            }
        }
        return [];
    }

    private static function isTextSize(string $core): bool
    {
        if (in_array($core, ['text-xs', 'text-sm', 'text-base', 'text-lg', 'text-xl', 'text-2xl', 'text-3xl', 'text-4xl', 'text-5xl', 'text-6xl', 'text-7xl', 'text-8xl', 'text-9xl'], true)) {
            return true;
        }
        if (str_starts_with($core, 'text-[')) {
            $rest = substr($core, 6);
            return $rest !== '' && (ctype_digit($rest[0]) || $rest[0] === '.');
        }
        return false;
    }

    private static function isNumber(string $s): bool
    {
        if (str_starts_with($s, '[')) {
            return true;
        }
        return $s !== '' && strspn($s, '0123456789.') === strlen($s);
    }

    /** the class string of a recipe table for a selection (organisms: base, each key, the matching compounds) */
    public function recipe(string $path, array $sel): string
    {
        $r = $this->json($path);
        $pick = function (string $k) use ($r, $sel): string {
            $v = trim(self::s($sel[$k] ?? null));
            return $v !== '' ? $v : trim((string) ($r['defaults'][$k] ?? ''));
        };
        $parts = [(string) ($r['base'] ?? '')];
        foreach ($r['keys'] ?? [] as $k) {
            $parts[] = (string) ($r['byKey'][$k][$pick($k)] ?? '');
        }
        foreach ($r['compound'] ?? [] as $c) {
            if (self::compoundHits($c, $pick)) {
                $parts[] = (string) ($c['class'] ?? '');
            }
        }
        return CnMerge::merge(array_filter(array_map('trim', $parts), fn ($p) => $p !== ''));

    private static function compoundHits(array $c, callable $pick): bool
    {
        foreach ($c as $k => $want) {
            if ($k === 'class') {
                continue;
            }
            if (!in_array($pick((string) $k), explode('|', (string) $want), true)) {
                return false;
            }
        }
        return true;
    }

    public static function attrHtml(mixed $attrs): string

    /** the brick class contract: recipe base, the selected variants (defaults for empty), compounds, then the tail */
    public function compose(string $path, array $selection, array $extra): string
    {
        $r = $this->json($path);
        $parts = [];
        if (trim((string) ($r['base'] ?? '')) !== '') {
            $parts[] = trim((string) $r['base']);
        }
        $pick = function (string $k) use ($r, $selection): string {
            $c = trim(self::s($selection[$k] ?? null));
            return $c !== '' ? $c : trim((string) ($r['defaults'][$k] ?? ''));
        };
        foreach ($r['keys'] ?? [] as $key) {
            $choices = $r['byKey'][$key] ?? null;
            if ($choices === null) {
                continue;
            }
            $choice = $pick($key);
            if ($choice === '') {
                if ($choices) {
                    throw new \RuntimeException("[compose] missing default for key $key");
                }
                continue;
            }
            if (isset($choices[$choice]) && trim((string) $choices[$choice]) !== '') {
                $parts[] = (string) $choices[$choice];
            }
        }
        foreach ($r['compound'] ?? [] as $c) {
            if (self::compoundHits($c, $pick) && trim((string) ($c['class'] ?? '')) !== '') {
                $parts[] = (string) $c['class'];
            }
        }
        return self::cn([...$parts, ...$extra]);
    }

    /** the tag of a primitive: the requested one when the allow-list has it, else the fallback */
    public static function resolveTag(mixed $tag, string $fallback, array $allowed): string
    {
        $t = strtolower(trim(self::s($tag)));
        return $t !== '' && in_array($t, $allowed, true) ? $t : $fallback;

    /** the inner markup of a Lucide glyph: its nodes with attributes in name order (circle for an unknown name) */
    public function lucide(mixed $name): string
    {
        $icons = $this->json('ui/lucide-icon/lucide-icons.json')['icons'];
        $nodes = $icons[self::s($name)] ?? $icons['circle'];

            }
            $out .= '></' . $tag . '>';
        }
        return $out;
    }
