/**
 * A sidebar section heading with optional right actions. Set `expanded` and it becomes a disclosure button
 * (aria-expanded, the toggle event).
 */
import { type ReactNode } from "react";
import { Box } from "../primitives/box/box";
import { ButtonPlain } from "../primitives/button-plain/button-plain";
import { Inline } from "../primitives/inline/inline";
import { cn, variants } from "../utils";

/** SectionHead: the classes of each value (shadcn/cva style). */
export const sectionHeadVariants = variants({
  base: "flex items-center justify-between pr-4 pl-4",
  variants: {
    size: {
      default: "h-8 py-2 text-xs font-semibold",
      lg: "h-12 py-2 text-sm font-bold",
    },
  },
  defaultVariants: { size: "default" },
});

export type SectionHeadProps = {
  size?: "default" | "lg";
  expanded?: boolean;
  controls?: string;
  className?: string;
  children?: ReactNode;
  right?: ReactNode;
  onToggle?: () => void;
};

export function SectionHead({
  size = "default",
  expanded,
  controls,
  className,
  children,
  right,
  onToggle,
}: SectionHeadProps) {
  const cls =
    cn(
      sectionHeadVariants({ size: size }),
      "rounded text-foreground in-data-[nav=collapsed]:hidden",
      className,
    );
  return (
    <>
      {(expanded != null) ? (
        <ButtonPlain
          type="button"
          className={cn(cls, "w-full text-left")}
          aria-expanded={expanded}
          aria-controls={controls}
          data-slot="section-head"
          data-expanded={expanded}
          onClick={() => onToggle?.()}
        >
          <Inline data-slot="section-head-title">
            {children}
          </Inline>
          {right ? (
            <Inline className="flex items-center gap-2" data-slot="section-head-right">
              {right}
            </Inline>
          ) : null}
        </ButtonPlain>
      ) : (
        <Box className={cls} data-slot="section-head">
          <Inline data-slot="section-head-title">
            {children}
          </Inline>
          {right ? (
            <Inline className="flex items-center gap-2" data-slot="section-head-right">
              {right}
            </Inline>
          ) : null}
        </Box>
      )}
    </>
  );
}
