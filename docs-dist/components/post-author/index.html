/**
 * The author of a post: the portrait, the name and role, and the date or the share and save buttons; as a line,
 * a card, or a card with the bio.
 */
import { type ReactNode } from "react";
import { Box } from "../primitives/box/box";
import { Button } from "../primitives/button/button";
import { Group } from "../primitives/group/group";
import { LucideIcon } from "../primitives/lucide-icon/lucide-icon";
import { Stack } from "../primitives/stack/stack";
import { Text } from "../primitives/text/text";
import { MediaFrame } from "../components/media-frame";
import { cn } from "../utils";
import type { Post } from "../blocks/post-card.types";
import type { PostExtras, PostSplitLabels } from "../blocks/post-split.types";

export type PostAuthorProps = {
  post: Post;
  extras: PostExtras;
  labels: PostSplitLabels;
  mode?: "line" | "card" | "bio";
  share?: boolean;
  className?: string;
};

export function PostAuthor({
  post,
  extras,
  labels,
  mode = "line",
  share = false,
  className,
}: PostAuthorProps) {
  return (
    <Stack
      gap="4"
      align="stretch"
      className={cn(
        (((mode as string) === "line") ? "" : "rounded bg-card inset-ring inset-ring-border p-4"),
        className
      )}
      data-testid="post-author"
    >
      <Group gap="4" align="center">
        <MediaFrame alt={post.author} ratio="1/1" className="size-10 shrink-0 rounded-full" />
        <Stack gap="0" align="start" className="min-w-0 flex-1 text-start">
          <Text size="sm" weight="semibold">
            {post.author}
          </Text>
          <Text size="xs" className="text-muted-foreground">
            {extras.role}
          </Text>
        </Stack>
        {share ? (
          <Group gap="2">
            <Button variant="ghost" size="default" href="#/" className="gap-2" data-testid="post-share">
              <LucideIcon name="share-2" size={16} />
              {labels.share}
            </Button>
            <Button variant="ghost" size="default" href="#/" className="gap-2" data-testid="post-save">
              <LucideIcon name="bookmark" size={16} />
              {labels.save}
            </Button>
          </Group>
        ) : (
          <Box className="text-muted-foreground">
            <Group gap="2" align="center">
              <LucideIcon name="calendar" size={16} />
              <Text size="sm">
                {post.date}
              </Text>
            </Group>
          </Box>
        )}
      </Group>
      {((mode as string) === "bio") ? (
        <Text size="sm" className="text-muted-foreground">
          {extras.bio}
        </Text>
      ) : null}
    </Stack>
  );
}
