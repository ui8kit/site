/**
 * The row inside NavItem: the icon and the label on the left, the right slot (or an empty 20 by 20 frame) on the
 * right.
 */
import { type ReactNode } from "react";
import { Inline } from "../primitives/inline/inline";

export type NavItemRowProps = {
  inner?: string;
  reserveRight?: boolean;
  children?: ReactNode;
  icon?: ReactNode;
  right?: ReactNode;
};

export function NavItemRow({ inner, reserveRight = false, children, icon, right }: NavItemRowProps) {
  return (
    <Inline className={inner} data-slot="nav-item-content">
      <Inline
        className={"flex min-w-0 flex-1 items-center gap-4 in-data-[nav=collapsed]:flex-none in-data-[nav=collapsed]:justify-center"}
        data-slot="nav-item-left"
      >
        {icon}
        <Inline className="min-w-0 flex-1 truncate in-data-[nav=collapsed]:hidden" data-slot="nav-item-label">
          {children}
        </Inline>
      </Inline>
      {right ? (
        <Inline className="flex items-center gap-2 in-data-[nav=collapsed]:hidden" data-slot="nav-item-right">
          {right}
        </Inline>
      ) : reserveRight ? (
        <Inline className="size-5 shrink-0 in-data-[nav=collapsed]:hidden" data-slot="nav-item-right" />
      ) : null}
    </Inline>
  );
}
