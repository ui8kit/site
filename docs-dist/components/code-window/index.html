/**
 * A terminal or code window: a title bar with three round marks, then the lines. A `command` line starts with
 * the prompt, an `output` line is muted. Static: the typing animation belongs to the port (TypingText contract).
 */
import { type ReactNode } from "react";
import { Box } from "../primitives/box/box";
import { Inline } from "../primitives/inline/inline";
import { List } from "../primitives/list/list";
import { ListItem } from "../primitives/list/list-item";
import { Text } from "../primitives/text/text";
import { cn } from "../utils";
import type { CodeLine } from "./code-window.types";

export type CodeWindowProps = {
  title: string;
  prompt: string;
  lines: CodeLine[];
  className?: string;
};

export function CodeWindow({ title, prompt, lines, className }: CodeWindowProps) {
  return (
    <Box
      className={cn(
        "w-full overflow-hidden rounded bg-card text-left inset-ring inset-ring-border",
        className
      )}
      role="group"
      aria-label={title}
      data-slot="code-window"
    >
      <Box className="flex items-center border-b border-border px-4 py-2">
        <Text size="xs" className="text-muted-foreground">
          {title}
        </Text>
      </Box>
      <List className="m-0 flex list-none flex-col gap-2 p-4 font-mono text-sm">
        {(lines ?? []).map((line) => (
          <ListItem key={line.id}
            className={cn(
              "flex gap-2",
              ((line.kind === "command") ? "text-foreground" : "text-muted-foreground")
            )}
            data-kind={line.kind}
          >
            {(line.kind === "command") ? (
              <Inline className="select-none text-muted-foreground">
                {prompt}
              </Inline>
            ) : null}
            <Inline>
              {line.text}
            </Inline>
          </ListItem>
        ))}
      </List>
    </Box>
  );
}

package blocks

import (
	"github.com/codegen-x/ui8kit/packages/templ/ui/ui/box"
	"github.com/codegen-x/ui8kit/packages/templ/ui/ui/inline"
	"github.com/codegen-x/ui8kit/packages/templ/ui/ui/list"
	"github.com/codegen-x/ui8kit/packages/templ/ui/ui/text"
	uiutils "github.com/codegen-x/ui8kit/packages/templ/ui/utils"
)

// CodeWindowProps are the props of CodeWindow; the values are the JSON shapes of the TypeScript props.
type CodeWindowProps struct {
	Title any `json:"title"`
	Prompt any `json:"prompt"`
	Lines any `json:"lines"`
	Class string `json:"className"`
}

// A terminal or code window: a title bar with three round marks, then the lines. A `command` line starts with the prompt, an `output` line is muted. Static: the typing animation belongs to the port (TypingText contract).
templ CodeWindow(p CodeWindowProps) {
	{{ vTitle := p.Title }}
	{{ vPrompt := p.Prompt }}
	{{ vLines := p.Lines }}
	@box.Box(box.BoxProps{Class: uiutils.S(uiutils.Cnv("w-full overflow-hidden rounded bg-card text-left inset-ring inset-ring-border", p.Class)), Attrs: uiutils.Attrs("role", "group", "aria-label", vTitle, "data-slot", "code-window")}) {
		@box.Box(box.BoxProps{Class: "flex items-center border-b border-border px-4 py-2"}) {
			@text.Text(text.TextProps{Size: "xs", Class: "text-muted-foreground"}) {
				{ uiutils.Text(vTitle) }
			}
		}
		@list.List(list.ListProps{Class: "m-0 flex list-none flex-col gap-2 p-4 font-mono text-sm"}) {
			for _, vLine := range uiutils.L(vLines) {
				@list.ListItem(list.ListItemProps{Class: uiutils.S(uiutils.Cnv("flex gap-2", uiutils.Cond(uiutils.T(uiutils.Eq(uiutils.Get(vLine, "kind"), "command")), "text-foreground", "text-muted-foreground"))), Attrs: uiutils.Attrs("data-kind", uiutils.Get(vLine, "kind"))}) {
					if uiutils.Eq(uiutils.Get(vLine, "kind"), "command") {
						@inline.Inline(inline.InlineProps{Class: "select-none text-muted-foreground"}) {
							{ uiutils.Text(vPrompt) }
						}
					}
					@inline.Inline(inline.InlineProps{}) {
						{ uiutils.Text(uiutils.Get(vLine, "text")) }
					}
				}
			}
		}
	}
}
