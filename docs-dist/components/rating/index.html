/**
 * One to five stars (a review's grade, the average of a product). The value is read as the label; the stars
 * themselves are decoration.
 */
import { type ReactNode } from "react";
import { Box } from "../primitives/box/box";
import { LucideIcon } from "../primitives/lucide-icon/lucide-icon";
import { cn } from "../utils";

export type RatingProps = {
  value: number;
  label: string;
  max?: number;
  size?: number;
  className?: string;
};

export function Rating({ value, label, max = 5, size = 16, className }: RatingProps) {
  const full = Math.round(Math.min(max, Math.max(0, value)));
  return (
    <Box
      className={cn("inline-flex items-center gap-0", className)}
      role="img"

      ))}

  );
}

package blocks

import (
	"github.com/codegen-x/ui8kit/packages/templ/ui/ui/box"
	lucideicon "github.com/codegen-x/ui8kit/packages/templ/ui/ui/lucide-icon"
	uiutils "github.com/codegen-x/ui8kit/packages/templ/ui/utils"
)

// RatingProps are the props of Rating; the values are the JSON shapes of the TypeScript props.
type RatingProps struct {
	Value any `json:"value"`
	Label any `json:"label"`
	Max any `json:"max"`
	Size any `json:"size"`
	Class string `json:"className"`
}

// One to five stars (a review's grade, the average of a product). The value is read as the label; the stars themselves are decoration.
templ Rating(p RatingProps) {
	{{ vValue := p.Value }}
	{{ vLabel := p.Label }}
	{{ vMax := uiutils.Def(p.Max, float64(5)) }}
	{{ vSize := uiutils.Def(p.Size, float64(16)) }}
	{{ vFull := any(uiutils.Round(uiutils.Min(vMax, uiutils.Max(float64(0), vValue)))) }}
	@box.Box(box.BoxProps{Class: uiutils.S(uiutils.Cnv("inline-flex items-center gap-0", p.Class)), Attrs: uiutils.Attrs("role", "img", "aria-label", vLabel, "data-slot", "rating", "data-value", vFull)}) {
		for _, vI := range uiutils.L(uiutils.Range(vMax)) {
			@lucideicon.LucideIcon(lucideicon.LucideIconProps{Name: "star", Size: uiutils.I(vSize), Class: uiutils.S(uiutils.Cond(uiutils.T(uiutils.Lt(vI, vFull)), "fill-foreground text-foreground", "text-border"))})
		}
	}
}
