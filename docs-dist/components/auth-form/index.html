 * The form of AuthCard: the fields of the mode (sign-in, sign-up, reset, magic link), remember me and the forgot
 * link, the terms, the submit button.
 */
import { type ReactNode } from "react";
import { Button } from "../primitives/button/button";
import { Checkbox } from "../primitives/checkbox/checkbox";
import { Form } from "../primitives/form/form";
import { Group } from "../primitives/group/group";
import { Input } from "../primitives/input/input";

import { Field } from "../components/field";
import { cn, tpl } from "../utils";
import type { AuthLabels } from "../blocks/auth-card.types";

export type AuthFormProps = {
  id: string;
  mode?: "signin" | "signup" | "reset" | "magic";
  labels: AuthLabels;
  submit: string;
};

export function AuthForm({ id, mode = "signin", labels, submit }: AuthFormProps) {
  return (
    <Form className="gap-4" data-testid="auth-form">
      {((mode as string) === "signup") ? (
        <Field label={labels.name} htmlFor={tpl("{0}-name", id)}>
          <Input type="text" name="name" autoComplete="name" id={tpl("{0}-name", id)} data-testid="field" />
        </Field>
      ) : null}
      <Field label={labels.email} htmlFor={tpl("{0}-email", id)}>
        <Input type="email" name="email" autoComplete="email" id={tpl("{0}-email", id)} data-testid="field" />
      </Field>
      {(((mode as string) !== "reset") && ((mode as string) !== "magic")) ? (
        <Field label={labels.password} htmlFor={tpl("{0}-password", id)}>
          <Input
            type="password"
            name="password"
            autoComplete={(((mode as string) === "signup") ? "new-password" : "current-password")}
            id={tpl("{0}-password", id)}
            data-testid="field"
          />
        </Field>
      ) : null}
      {((mode as string) === "signin") ? (
        <Group justify="between" align="center" className="flex-wrap gap-2">
          <Group gap="2" align="center" className={cn("w-fit", "")}>
            <Checkbox name="remember" id={tpl("{0}-remember", id)} />
            <Label htmlFor={tpl("{0}-remember", id)} className="text-sm">
              {labels.remember}
            </Label>
          </Group>
          <Link href="#/" variant="default" className="text-sm">
            {labels.forgot}
          </Link>
        </Group>
      ) : null}
      {((mode as string) === "signup") ? (
        <Group gap="2" align="center" className={cn("w-fit", "")}>
          <Checkbox name="terms" id={tpl("{0}-terms", id)} />
          <Label htmlFor={tpl("{0}-terms", id)} className="text-sm">
            {labels.terms}
          </Label>
        </Group>
      ) : null}
      <Button variant="default" size="default" type="button" className="w-full" data-testid="submit">
        {submit}
      </Button>
    </Form>
  );
}
