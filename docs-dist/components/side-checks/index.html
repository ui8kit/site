/**
 * Extras to tick: a fieldset named by its legend, one checkbox per option with its note and price. Static
 * markup.
 */
import { type ReactNode } from "react";
import { Checkbox, Fieldset, Inline, Label, Legend } from "../compose";
import { fmtMoney, tpl } from "../utils";
import type { DishOption } from "./option-chips.types";

export type SideChecksProps = {
  name: string;
  legend: string;
  options: DishOption[];
};

export function SideChecks({ name, legend, options }: SideChecksProps) {
  return (
    <Fieldset className="flex flex-col gap-2" data-testid="sides">
      <Legend size="sm" className="mb-2">
        {legend}
      </Legend>
      {(options ?? []).map((o) => (
        <Label key={o.id} className={"flex items-center justify-between gap-4 rounded px-4 py-2 text-sm inset-ring inset-ring-border has-checked:bg-muted"}>

            {tpl("{0} · {1}", o.note, fmtMoney(o.price, o.currency))}
          </Inline>
        </Label>
      ))}
    </Fieldset>
  );
}

import (
	"github.com/ui8kit/codegen/generated/templ/ui/box"

)

// SideChecksProps are the props of SideChecks; the values are the JSON shapes of the TypeScript props.
type SideChecksProps struct {
	Name any `json:"name"`
	Legend any `json:"legend"`
	Options any `json:"options"`
}

// Extras to tick: a fieldset named by its legend, one checkbox per option with its note and price. Static markup.
templ SideChecks(p SideChecksProps) {
	{{ vName := p.Name }}
	{{ vLegend := p.Legend }}
	{{ vOptions := p.Options }}
	@compose.Fieldset(compose.FieldsetProps{BoxProps: box.BoxProps{Class: "flex flex-col gap-2", Attrs: uiutils.Attrs("data-testid", "sides")}}) {
		@compose.Legend(compose.LegendProps{Size: "sm", TextProps: text.TextProps{Class: "mb-2"}}) {
			{ uiutils.Text(vLegend) }
		}
		for _, vO := range uiutils.L(vOptions) {
			@compose.Label(compose.LabelProps{LabelProps: label.LabelProps{Class: "flex items-center justify-between gap-4 rounded px-4 py-2 text-sm inset-ring inset-ring-border has-checked:bg-muted"}}) {
				@compose.Inline(compose.InlineProps{TextProps: text.TextProps{Class: "flex items-center gap-2"}}) {
					@compose.Checkbox(compose.CheckboxProps{CheckboxProps: checkbox.CheckboxProps{Name: uiutils.S(uiutils.Tpl("{0}-{1}", vName, uiutils.Get(vO, "id"))), Value: uiutils.SP("on")}})
					@compose.Inline(compose.InlineProps{TextProps: text.TextProps{}}) {
						{ uiutils.Text(uiutils.Get(vO, "name")) }
					}
				}
				@compose.Inline(compose.InlineProps{TextProps: text.TextProps{Class: "text-muted-foreground"}}) {
					{ uiutils.Text(uiutils.Tpl("{0} · {1}", uiutils.Get(vO, "note"), uiutils.Money(uiutils.Get(vO, "price"), uiutils.Get(vO, "currency")))) }
				}
			}
		}
	}
}
