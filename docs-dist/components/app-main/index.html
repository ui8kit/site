/**
 * The main column of AppShell: the heading, the metric tiles, and the chart panel (beside the activity panel

import { type ReactNode } from "react";
import { Block } from "../primitives/block/block";
import { Grid } from "../primitives/grid/grid";
import { Stack } from "../primitives/stack/stack";
import { Title } from "../primitives/title/title";
import { BarChart } from "../components/bar-chart";
import { Panel } from "../components/panel";
import { StatTile } from "../components/stat-tile";
import { Timeline } from "../components/timeline";
import { tpl } from "../utils";
import type { Metric } from "../blocks/metric-row.types";
import type { Point } from "../blocks/chart-panel.types";
import type { Event } from "../blocks/activity-feed.types";
import type { OverviewChart, OverviewActivity } from "../blocks/dashboard-overview.types";

export type AppMainProps = {
  id: string;
  title: string;
  metrics: Metric[];
  chart: OverviewChart;
  activity?: OverviewActivity;
  cols?: "1-2" | "1-4";
  density?: "default" | "compact";
};

export function AppMain({
  id,
  title,
  metrics,
  chart,
  activity,
  cols = "1-2",
  density = "default",
}: AppMainProps) {
  return (
    <Stack
      gap={(((density as string) === "compact") ? "4" : "6")}
      align="stretch"
      className="min-w-0"
      data-testid="app-main"
    >
      <Title as={2} size="2xl" weight="bold" id={tpl("{0}-title", id)}>
        {title}
      </Title>
      <Grid cols={cols} className="gap-4" data-testid="items">
        {(metrics ?? []).map((m) => (
          <Block key={m.id} tag="div" data-testid="item">
            <StatTile label={m.label} value={m.value} icon={m.icon} hint={m.hint} />
          </Block>
        ))}
      </Grid>
      {activity ? (
        <Grid cols="1-2" className="items-stretch gap-4">
          <Panel title={chart.title} description={chart.text} id={tpl("{0}-chart", id)} className="h-full">
            <BarChart bars={chart.series} label={chart.summary} height={144} />
          </Panel>
          <Panel title={activity?.title} description={activity?.text} id={tpl("{0}-feed", id)} className="h-full">
            <Timeline items={activity?.events} />
          </Panel>
        </Grid>
      ) : (
        <Panel title={chart.title} description={chart.text} id={tpl("{0}-chart", id)} className="h-full">
          <BarChart bars={chart.series} label={chart.summary} height={144} />
        </Panel>
      )}
    </Stack>
  );
}
