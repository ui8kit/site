});

export type IconTabProps = {
  active?: boolean;
  label: string;
  className?: string;
  children?: ReactNode;
  onClick?: () => void;
};

export function IconTab({ active = false, label, className, children, onClick }: IconTabProps) {
  return (
    <ButtonPlain
