/**
 * A vertical list of events (an order's history): a dot on a line, the title, an optional text and the time.
 * Newest first or oldest first is the caller's order.
 */
import { type ReactNode } from "react";
import { Block } from "../primitives/block/block";
import { Box } from "../primitives/box/box";
import { Inline } from "../primitives/inline/inline";
import { List } from "../primitives/list/list";
import { ListItem } from "../primitives/list/list-item";
import { Text } from "../primitives/text/text";
import { at, cn } from "../utils";
import type { TimelineItem } from "./timeline.types";

const TONE_DOT: Record<string, string> = {
  "gray": "bg-muted-foreground",
  "green": "bg-emerald-500",
  "blue": "bg-blue-500",
  "amber": "bg-amber-500",
  "red": "bg-red-500",
};

export type TimelineProps = {
  items: TimelineItem[];
  className?: string;
};

export function Timeline({ items, className }: TimelineProps) {
  return (
    <Block className={cn("min-w-0", className)} data-slot="timeline">
      <List className="m-0 flex list-none flex-col p-0">
        {(items ?? []).map((item, i) => (
          <ListItem key={item.id} className="relative flex min-w-0 gap-4 pb-4 last:pb-0" data-event={item.id}>
            <Box className="relative flex w-3 shrink-0 justify-center">
              {(i < ((items?.length ?? 0) - 1)) ? (
                <Inline className="absolute top-4 -bottom-0 w-px bg-border" />
              ) : null}
              <Inline className={cn("mt-1.5 size-2.5 rounded-full", at(TONE_DOT, (item.tone ?? "gray")))} />
            </Box>
            <Box className="flex min-w-0 flex-1 flex-col gap-0">
              <Box className="flex flex-wrap items-baseline justify-between gap-x-4">
                <Text className="m-0 text-sm font-bold text-foreground">
                  {item.title}
                </Text>
                <Text className="m-0 shrink-0 text-xs font-medium text-muted-foreground">
                  {item.time}
                </Text>
              </Box>
              {item.text ? (
                <Text className="m-0 text-xs font-medium text-muted-foreground">
                  {item.text}
                </Text>
              ) : null}
            </Box>
          </ListItem>
        ))}
      </List>
    </Block>
  );
}
