  return (
    <Block className={cn("min-w-0", className)} data-slot="meter-row">
      {href ? (
        <Link
          href={href}
          variant="block"
          className="flex min-w-0 flex-col gap-2 rounded p-2 text-foreground no-underline hover:bg-muted"

              {caption}
            </Text>
          </Box>
          <Progress value={(value == null ? undefined : String(value))} max="100" aria-label={label} />
          {hint ? (
            <Text className="m-0 text-xs font-medium text-muted-foreground">
              {hint}
            </Text>
          ) : null}
        </Link>
      ) : (
        <Box className="flex min-w-0 flex-col gap-2 p-2">
          <Box className="flex items-baseline justify-between gap-4">
            <Text className="m-0 min-w-0 truncate text-sm font-bold text-foreground">

            <Text className="m-0 text-xs font-medium text-muted-foreground">
