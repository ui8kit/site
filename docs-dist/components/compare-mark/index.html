/**
 * One cell value of ComparisonTable: yes shows a check, no a dash (both with the hidden word for screen
 * readers), any other value is shown as text.
 */
import { type ReactNode } from "react";
import { Inline } from "../primitives/inline/inline";
import { LucideIcon } from "../primitives/lucide-icon/lucide-icon";

export type CompareMarkProps = {
  value: string;
  yes: string;
  no: string;
};

export function CompareMark({ value, yes, no }: CompareMarkProps) {
  return (
    <>
      {((value as string) === "yes") ? (
        <Inline className="inline-flex items-center gap-2" data-mark="yes">
          <LucideIcon name="check" size={16} className="text-foreground" />
          <Inline className="sr-only">
            {yes}
          </Inline>
        </Inline>
      ) : null}
      {((value as string) === "no") ? (
        <Inline className="inline-flex items-center gap-2 text-muted-foreground" data-mark="no">
          <LucideIcon name="minus" size={16} />
          <Inline className="sr-only">
            {no}
          </Inline>
        </Inline>
      ) : null}
      {(((value as string) !== "yes") && ((value as string) !== "no")) ? (
        <Inline>
          {value}
        </Inline>
      ) : null}
    </>
  );
}
