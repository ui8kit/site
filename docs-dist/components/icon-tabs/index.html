/**
 * A row of icon-only tabs (a file tree pane: Files, Search, Apps). Put IconTab in it. Arrow keys: attach
 * `moveFocus` from the utils to the keydown of the row (orientation horizontal, activate).
 */
import { type ReactNode } from "react";
import { Block } from "../primitives/block/block";
import { cn } from "../utils";

export type IconTabsProps = {
  label?: string;
  className?: string;
  children?: ReactNode;
};

export function IconTabs({ label, className, children }: IconTabsProps) {
  return (
    <Block
      className={cn("flex items-center gap-2", className)}
      role="tablist"
      aria-label={label}
      data-slot="icon-tabs"
    >
      {children}
    </Block>
  );
}
