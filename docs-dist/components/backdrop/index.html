/**
 * Something behind the children: a faint full-size picture (image), or nothing (none: the children as they are,
 * no wrapper).
 */
import { type ReactNode } from "react";
import { Block } from "../primitives/block/block";
import { MediaFrame } from "../components/media-frame";

export type BackdropProps = {
  kind?: "none" | "image";
  alt?: string;
  children?: ReactNode;
};

export function Backdrop({ kind = "none", alt = "", children }: BackdropProps) {
  return (
    <>
      {((kind as string) === "image") ? (
        <Block tag="div" className="relative isolate overflow-hidden" data-testid="backdrop">
          <MediaFrame alt={alt} ratio="21/9" className="absolute inset-0 -z-10 size-full rounded-none opacity-30" />
          {children}
        </Block>
      ) : (
        <>
          {children}
        </>
      )}
    </>
  );
}
