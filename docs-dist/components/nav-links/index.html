/**
 * The links of a top bar in a row; the current one carries aria-current. side: all, or the first or second half
 * of the list.
 */
import { Fragment, type ReactNode } from "react";
import { Group } from "../primitives/group/group";
import { Link } from "../primitives/link/link";
import type { NavLink } from "../blocks/nav-bar.types";

export type NavLinksProps = {
  links: NavLink[];
  current?: string;
  side?: "all" | "left" | "right";
};

export function NavLinks({ links, current = "", side = "all" }: NavLinksProps) {
  const count = links?.length ?? 0;
  return (
    <Group gap="4" align="center" className="flex-wrap">
      {(links ?? []).map((l, i) => (
        <Fragment key={l.id}>
          {(((side as string) === "all") || (((i * 2) < count) === ((side as string) === "left"))) ? (
            <Link
              href={(l.href ?? "#/")}
              variant="default"
              aria-current={((l.id === current) ? "page" : undefined)}
              className="text-sm"
              data-testid="nav-link"
            >
              {l.name}
            </Link>
          ) : null}
        </Fragment>
      ))}
    </Group>
  );
}
