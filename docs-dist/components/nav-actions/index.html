/**
 * The buttons at the end of a top bar: login, the call to action and the theme button (it only reports the
 * press). No wrapper: they sit in the row of the bar.
 */
import { type ReactNode } from "react";
import { Button } from "../primitives/button/button";

export type NavActionsProps = {
  login?: string;
  loginHref?: string;
  cta?: string;
  ctaHref?: string;
  actions?: "none" | "cta";
  tone?: "default" | "inverse";
  themeLabel?: string;
  onTheme?: () => void;
};

export function NavActions({
  login,
  loginHref = "#/login",
  cta,
  ctaHref,
  actions = "cta",
  tone = "default",
  themeLabel,
  onTheme,
}: NavActionsProps) {
  return (
    <>
      {login ? (
        <Button
          variant="ghost"
          size="sm"
          href={loginHref}
          className={(((tone as string) === "inverse") ? "text-background hover:bg-background/10" : "")}
          data-testid="nav-login"
        >
          {login}
        </Button>
      ) : null}
      {(((actions as string) === "cta") && cta) ? (
        <Button
          variant={(((tone as string) === "inverse") ? "secondary" : "default")}
          size="sm"
          href={ctaHref}
          data-testid="header-cta"
        >
          {cta}
        </Button>
      ) : null}
      {themeLabel ? (
        <Button variant="outline" size="sm" data-testid="theme-toggle" onClick={() => onTheme?.()}>
          {themeLabel}
        </Button>
      ) : null}
    </>
  );
}
