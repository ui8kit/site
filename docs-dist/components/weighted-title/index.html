/** A display title, one span per word; word i takes weights[i] (from the preset). The words wrap like text. */
import { Fragment, type ReactNode } from "react";
import { Inline } from "../primitives/inline/inline";
import { Title } from "../primitives/title/title";
import { defaults } from "../presets/defaults";
import { cn } from "../utils";

export type WeightedTitleProps = {
  id: string;
  title: string;
  weights?: readonly string[];
  level?: 1 | 2;
};

export function WeightedTitle({ id, title, weights, level = 1 }: WeightedTitleProps) {
  return (
    <Title
      as={level}
      size="5xl"
      weight="semibold"
      className={cn("text-center tracking-tight text-background", defaults.hero.title)}
      id={id}
    >
      {((title ?? "").split(" ") ?? []).map((word, i) => (
        <Fragment key={i}>
          <Inline className={weights?.[i]}>
            {word}
          </Inline>
          {" "}
        </Fragment>
      ))}
    </Title>
  );
}

    // Code generated by xgen. DO NOT EDIT.
// Source: xgen/specs/organisms/weighted-title.json

package blocks

import (
	"github.com/codegen-x/ui8kit/packages/templ/ui/ui/inline"
	"github.com/codegen-x/ui8kit/packages/templ/ui/ui/title"
	uiutils "github.com/codegen-x/ui8kit/packages/templ/ui/utils"
)

// WeightedTitleProps are the props of WeightedTitle; the values are the JSON shapes of the TypeScript props.
type WeightedTitleProps struct {
	Id any `json:"id"`
	Title any `json:"title"`
	Weights any `json:"weights"`
	Level any `json:"level"`
}

// A display title, one span per word; word i takes weights[i] (from the preset). The words wrap like text.
templ WeightedTitle(p WeightedTitleProps) {
	{{ vId := p.Id }}
	{{ vTitle := p.Title }}
	{{ vWeights := p.Weights }}
	{{ vLevel := uiutils.Def(p.Level, float64(1)) }}
	@title.Title(title.TitleProps{As: uiutils.HeadingLevel(uiutils.I(vLevel)), Size: "5xl", Weight: "semibold", Class: uiutils.S(uiutils.Cnv("text-center tracking-tight text-background", uiutils.Get(presetDefaults, "hero", "title"))), Attrs: uiutils.Attrs("id", vId)}) {
		for vI, vWord := range uiutils.L(uiutils.Words(vTitle)) {
			@inline.Inline(inline.InlineProps{Class: uiutils.S(uiutils.Nth(vWeights, float64(vI)))}) {
				{ uiutils.Text(vWord) }
			}
			{ uiutils.Text(" ") }
		}
	}
}

    A display title, one span per word; word i takes weights[i] (from the preset). The words wrap like text.

    Props of WeightedTitle
    PropTypeDefault Description
    id (required)string-id of the heading (the band is labelled by it)
    title (required)string-the words, separated by spaces
    weights readonly string[]-a font-weight class per word
    level 1 | 21
