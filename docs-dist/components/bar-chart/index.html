/**
 * A row of vertical bars (sales by day, orders by week) with a caption under each bar. The chart is one image
 * for assistive technology: `label` carries the summary, the bar values are in the captions and titles.
 */
import { type ReactNode } from "react";
import { Block } from "../primitives/block/block";
import { Box } from "../primitives/box/box";
import { Text } from "../primitives/text/text";
import { cn } from "../utils";
import type { Bar } from "./bar-chart.types";

export type BarChartProps = {
  bars: Bar[];
  label: string;
  max?: number;
  height?: number;
  className?: string;
};

export function BarChart({ bars, label, max, height = 160, className }: BarChartProps) {
  const top = Math.max(1, (max ?? bars.reduce((m, i) => Math.max(m, i.value), 0)));
  return (
    <Block className={cn("min-w-0", className)} data-slot="bar-chart">
      <Box className="flex min-w-0 items-end gap-2" role="img" aria-label={label} style={{ height: `${height}px` }}>
        {(bars ?? []).map((b) => (
          <Box key={b.id} className="flex h-full min-w-0 flex-1 items-end" title={(b.title ?? b.label)}>
            <Box
              className={cn("w-full rounded", (b.muted ? "bg-muted-foreground" : "bg-primary"))}
              style={{ height: `${Math.max(2, Math.round(((b.value / top) * 100)))}%` }}
              data-bar={b.id}
            />
          </Box>
        ))}
      </Box>
      <Box className="mt-2 flex min-w-0 gap-2" aria-hidden="true">
        {(bars ?? []).map((b) => (
          <Box key={b.id} className="relative h-4 min-w-0 flex-1">
            <Text className={"absolute left-1/2 top-0 m-0 -translate-x-1/2 whitespace-nowrap text-xs font-medium text-muted-foreground"}>
              {b.label}
            </Text>
          </Box>
        ))}
      </Box>
    </Block>
  );
}

    Names and types come from the xgen specs of the source. Standard HTML attributes (id, role, tabIndex, aria-*) pass through to the root element; the class prop is className in React.

    A row of vertical bars (sales by day, orders by week) with a caption under each bar. The chart is one image for assistive technology: `label` carries the summary, the bar values are in the captions and titles.

    Props of BarChart
    PropTypeDefaultDescription
    bars (required)Bar[]-id, caption, value; `title` is the tooltip, `muted` dims a period without data
    label (required)string-the summary read to assistive technology
    max number-the value of a full-height bar; the largest value by default
    height number160height of the plot area in px
