/**
 * One or two calls to action side by side: a filled primary button and an outline secondary one. Links, not
 * handlers: the page decides what they open.
 */
import { type ReactNode } from "react";
import { Button } from "../primitives/button/button";
import { Group } from "../primitives/group/group";

export type ActionGroupProps = {
  primary: string;
  primaryHref?: string;
  secondary?: string;
  secondaryHref?: string;
  align?: "start" | "center";
};

export function ActionGroup({
  primary,
  primaryHref,
  secondary,
  secondaryHref,
  align = "start",
}: ActionGroupProps) {
  return (
    <Group
      gap="4"
      align="center"
      justify={(((align as string) === "center") ? "center" : "start")}
      className="flex-wrap"
      data-slot="action-group"
    >
      <Button variant="default" size="default" href={primaryHref} data-testid="action-primary">
        {primary}
      </Button>
      {secondary ? (
        <Button variant="outline" size="default" href={secondaryHref} data-testid="action-secondary">
          {secondary}
        </Button>
      ) : null}
    </Group>
  );
}
