/**
 * A centered row of topic pills (static links). tone inverse: on a dark band with an icon per topic; default:
 * outlined.
 */
import { Fragment, type ReactNode } from "react";
import { Button } from "../primitives/button/button";
import { Group } from "../primitives/group/group";
import { LucideIcon } from "../primitives/lucide-icon/lucide-icon";
import type { Topic } from "./topic-chips.types";

export type TopicChipsProps = {
  topics: Topic[];
  tone?: "default" | "inverse";
};

export function TopicChips({ topics, tone = "default" }: TopicChipsProps) {
  return (
    <Group gap="2" align="center" justify="center" className="flex-wrap" data-testid="hero-topics">
      {(topics ?? []).map((topic) => (
        <Fragment key={topic.id}>
          {((tone as string) === "inverse") ? (
            <Button
              href="#/"
              variant="ghost"
              size="sm"
              className="gap-2 text-background inset-ring-background/30 hover:bg-background/10"
              data-testid="hero-topic"
            >
              {topic.icon ? (
                <LucideIcon name={topic.icon} size={16} />
              ) : null}
              {topic.name}
            </Button>
          ) : (
            <Button href="#/" variant="outline" size="sm" data-testid="hero-topic">
              {topic.name}
            </Button>
          )}
        </Fragment>
      ))}
    </Group>
  );
}
