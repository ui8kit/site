import { type ReactNode } from "react";

import { cn } from "../utils";
import type { ShellNav } from "../blocks/dashboard-shell.types";

export type ShellTopbarProps = {
  nav: ShellNav;
  brand?: "show" | "hide";
  links?: "show" | "hide";
  search?: "show" | "hide";
};

export function ShellTopbar({ nav, brand = "show", links = "hide", search = "show" }: ShellTopbarProps) {
  return (
    <Block
      tag="header"
      className="flex h-16 shrink-0 items-center gap-6 border-b border-border bg-background px-6"
      data-testid="topbar"
    >
      {((brand as string) === "show") ? (
        <Group gap="2" align="center" className="shrink-0" data-testid="brand">
          <LucideIcon name="layers" size={24} />

            {nav.brand}
          </Text>
        </Group>

        <Block tag="nav" className="hidden lg:block" aria-label={nav.label}>

            {(nav.links ?? []).map((l) => (
              <ListItem key={l.id}>
                <Link
                  href="#/"
                  variant="default"
                  size="sm"
                  aria-current={((l.id === nav.current) ? "page" : undefined)}
                  className={cn(
                    "flex items-center gap-4 rounded py-2",
                    (true ? "px-4" : "justify-center px-2"),
                    ((l.id === nav.current) ? "bg-muted font-medium" : "")
                  )}
                  data-testid="item"
                >
                  <LucideIcon name={l.icon} size={16} />
                  <Inline className={(true ? "" : "sr-only")}>
                    {l.name}
                  </Inline>
                </Link>
              </ListItem>
            ))}
          </List>
        </Block>
      ) : null}
      {((search as string) === "show") ? (
        <>
          {((links as string) === "show") ? (
            <Box className="ml-auto hidden w-64 shrink-0 xl:block">
              <SearchPalette label={nav.search} shortcut={nav.shortcut} />
            </Box>
          ) : (
            <SearchPalette label={nav.search} shortcut={nav.shortcut} />
          )}
        </>
      ) : null}
      <Group
        gap="4"

          : "ml-auto shrink-0"

        data-testid="user"

        <LucideIcon name="bell" size={20} className="text-muted-foreground" />
        <MediaFrame alt={nav.userName} ratio="1/1" className="size-8 shrink-0 rounded-full" />
      </Group>
    </Block>
  );
