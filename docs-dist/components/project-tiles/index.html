/**
 * The projects of ProjectGrid as tiles: cards (picture, featured badge, category and year, title, summary, tags,
 * two links), compact cards in four columns, captions over the pictures, or a bento where the first project
 * takes a big tile.
 */
import { type ReactNode } from "react";
import { Badge } from "../primitives/badge/badge";
import { Block } from "../primitives/block/block";
import { Box } from "../primitives/box/box";
import { Group } from "../primitives/group/group";
import { Link } from "../primitives/link/link";
import { LucideIcon } from "../primitives/lucide-icon/lucide-icon";
import { Stack } from "../primitives/stack/stack";
import { Text } from "../primitives/text/text";
import { Title } from "../primitives/title/title";
import { MediaFrame } from "../components/media-frame";
import { cn, tpl } from "../utils";
import type { Project, ProjectLabels } from "../blocks/project-grid.types";

export type ProjectTilesProps = {
  projects: Project[];
  labels: ProjectLabels;
  layout?: "cards" | "compact" | "overlay" | "bento" | "list";
};

export function ProjectTiles({ projects, labels, layout = "cards" }: ProjectTilesProps) {
  const over = ((layout as string) === "overlay") ? true : ((layout as string) === "bento");
  return (
    <Box
      className={
        (layout as string) === "compact" ? "grid grid-cols-1 gap-4 sm:grid-cols-2 lg:grid-cols-4"
        : "grid grid-cols-1 gap-4 sm:grid-cols-2 lg:grid-cols-3"
      }
      data-testid="items"
    >
      {(projects ?? []).map((p, i) => (
        <Block key={p.id}
          tag="article"
          className={cn(
            "flex flex-col gap-4",
            (over ? "relative" : "rounded bg-card inset-ring inset-ring-border p-4"),
            ((((layout as string) === "bento") && (i === 0)) ? "sm:col-span-2 lg:row-span-2" : "")
          )}
          data-testid="item"
        >
          <Box className={(over ? "h-full" : "relative")}>
            <MediaFrame
              alt={p.title}
              ratio={(((layout as string) === "compact") ? "1/1" : (over ? "3/4" : "4/3"))}
              className={(over ? "h-full" : "")}
            />
            {(p.featured && ((layout as string) !== "compact")) ? (
              <Badge variant="default" size="default" className="absolute left-4 top-4" data-testid="featured">
                {labels.featured}
              </Badge>
            ) : null}
          </Box>
          <Stack
            gap="2"
            align="stretch"
            className={(over ? "absolute right-4 bottom-4 left-4 rounded bg-background p-4" : "")}
          >
            <Text size="xs" className="text-muted-foreground">
              {tpl("{0}, {1}", p.category, p.year)}
            </Text>
            <Title as={3} size={(((layout as string) === "compact") ? "sm" : "lg")} weight="semibold">
              {p.title}
            </Title>
            {((layout as string) !== "compact") ? (
              <Text size="sm" className="text-muted-foreground">
                {p.summary}
              </Text>
            ) : null}
            {((layout as string) === "cards") ? (
              <>
                <Group gap="2" className="flex-wrap">
                  {(p.tags ?? []).map((k) => (
                    <Badge key={k} variant="secondary" size="sm">
                      {k}
                    </Badge>
                  ))}
                </Group>
                <Group gap="4" className="pt-2">
                  <Link href="#/work" variant="default" className="inline-flex items-center gap-2 text-sm font-medium">
                    {labels.case}
                    <LucideIcon name="arrow-right" size={16} />
                  </Link>
                  <Link href="#/work" variant="default" className="inline-flex items-center gap-2 text-sm font-medium">
                    {labels.view}
                    <LucideIcon name="external-link" size={16} />
                  </Link>
                </Group>
              </>
            ) : null}
          </Stack>
        </Block>
      ))}
    </Box>
  );
}
