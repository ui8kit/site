export type HeroSplitMediaProps = {
  content: HeroContent;
  alt: string;
  media?: "picture" | "card" | "grid" | "plans";
};

export function HeroSplitMedia({ content, alt, media = "picture" }: HeroSplitMediaProps) {
  return (
    <>
      {((media as string) === "picture") ? (

      {((media as string) === "card") ? (

          {(content.features ?? []).map((f, i) => (
            <Fragment key={f.id}>
              {(i < 4) ? (
                <Block tag="figure" className="flex flex-col gap-2" data-testid="hero-tile">
                  <MediaFrame alt={f.title} ratio="1/1" />

          ))}
        </Box>
      ) : null}
      {((media as string) === "plans") ? (
        <Box className="relative" data-testid="hero-media">
          <MediaFrame alt={alt} ratio="4/3" />
          {(content.plans ?? []).map((p, i) => (
            <Fragment key={p.id}>
              {(i < 2) ? (
                <Stack
                  gap="2"
                  className={cn(
                    "absolute w-48 rounded bg-card p-4 shadow-lg",
                    ((i === 0) ? "top-8 left-4" : "right-4 bottom-8")

                    </Text>
                  </Group>
                </Stack>
              ) : null}
            </Fragment>
          ))}
        </Box>
      ) : null}
    </>
  );
}
