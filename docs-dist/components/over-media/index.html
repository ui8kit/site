/**
 * Puts its children over a faded picture (a transparent top bar over a hero image). on false: the children
 * alone, no wrapper.
 */
import { type ReactNode } from "react";
import { Block } from "../primitives/block/block";
import { MediaFrame } from "../components/media-frame";

export type OverMediaProps = {
  on?: boolean;
  alt: string;
  children?: ReactNode;
};

export function OverMedia({ on = false, alt, children }: OverMediaProps) {
  return (
    <>
      {on ? (
        <Block tag="div" className="relative isolate overflow-hidden" data-testid="over">
          <MediaFrame alt={alt} ratio="21/9" className="absolute inset-0 -z-10 size-full rounded-none opacity-40" />
          <Block tag="div" className={"[&_header]:border-0 [&_header]:bg-transparent"}>
            {children}
          </Block>
        </Block>
      ) : (
        <>
          {children}
        </>
      )}
    </>
  );
}
