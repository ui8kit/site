/** The body of a SectionFrame: inside a card when `boxed`, else the children as they are (no wrapper). */
import { type ReactNode } from "react";
import { Card } from "../primitives/card/card";

export type SectionBoxProps = {
  boxed?: boolean;
  children?: ReactNode;
};

export function SectionBox({ boxed = false, children }: SectionBoxProps) {
  return (
    <>
      {boxed ? (
        <Card variant="muted">
          {children}
        </Card>
      ) : (
        <>
          {children}
        </>
      )}
    </>
  );
}
