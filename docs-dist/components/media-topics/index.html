import { Fragment, type ReactNode } from "react";

import { tpl } from "../utils";
import type { MediaExtras, MediaListLabels } from "../blocks/media-list.types";

export type MediaTopicsProps = {
  extras: MediaExtras;
  labels: MediaListLabels;
};

export function MediaTopics({ extras, labels }: MediaTopicsProps) {
  return (
    <Stack gap="4" align="stretch">

        {labels.list}

                </Text>
                <Text size="xs" className="text-muted-foreground">
                  {tpl(labels.count, c.count)}
                </Text>
              </Stack>
              <LucideIcon name="arrow-right" size={16} />
            </Link>
          ) : null}
        </Fragment>
      ))}
    </Stack>
  );
