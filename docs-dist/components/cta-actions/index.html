/**
 * The actions of a call to action: one or two links (ActionGroup), or an email field with a button (actions
 * form).
 */
import { type ReactNode } from "react";
import { Button } from "../primitives/button/button";
import { Form } from "../primitives/form/form";
import { Group } from "../primitives/group/group";
import { Input } from "../primitives/input/input";
import { Label } from "../primitives/label/label";
import { ActionGroup } from "../components/action-group";
import { cn, tpl } from "../utils";
import type { ActionSet } from "./cta-actions.types";

export type CtaActionsProps = {
  id: string;
  actions: ActionSet;
  center?: boolean;
};

export function CtaActions({ id, actions, center = false }: CtaActionsProps) {
  return (
    <>
      {(actions.mode === "form") ? (
        <Form method="dialog" className={cn("w-full max-w-md", (center ? "mx-auto" : ""))} data-testid="email-form">
          <Label htmlFor={tpl("{0}-email", id)} className="sr-only">
            {actions.emailLabel}
          </Label>
          <Group gap="2" align="start" className="w-full flex-wrap">
            <Input type="email" name="email" className="min-w-0 flex-1" id={tpl("{0}-email", id)} data-testid="field" />
            <Button variant="default" size="default" type="submit" data-testid="submit">
              {actions.cta}
            </Button>
          </Group>
        </Form>
      ) : (
        <ActionGroup
          primary={actions.cta}
          primaryHref={actions.href}
          secondary={((actions.mode === "two") ? actions.secondary : undefined)}
          secondaryHref={actions.secondaryHref}
          align={(center ? "center" : "start")}
        />
      )}
    </>
  );
}
