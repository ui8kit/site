/**
 * The toolbar above a table: search and filter controls on the left (children), the actions on the right
 * (`actions`). One tab stop group (role toolbar).
 */
import { type ReactNode } from "react";
import { Block } from "../primitives/block/block";
import { Inline } from "../primitives/inline/inline";
import { cn } from "../utils";

export type TableToolbarProps = {
  label: string;
  className?: string;
  children?: ReactNode;
  actions?: ReactNode;
};

export function TableToolbar({ label, className, children, actions }: TableToolbarProps) {
  return (
    <Block
      className={cn("flex flex-wrap items-center justify-between gap-2", className)}
      role="toolbar"
      aria-label={label}
      data-slot="table-toolbar"
    >
      <Inline className="flex min-w-0 flex-wrap items-center gap-2">
        {children}
      </Inline>
      <Inline className="flex flex-wrap items-center gap-2">
        {actions}
      </Inline>
    </Block>
  );
}
