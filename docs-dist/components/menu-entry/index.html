/**
 * One entry of a menu: a link to its address, or, without one, its content as text in the same box (the magazine
 * routes nothing, so it shows text). current marks the page the entry stands for (aria-current) either way.
 */
import { type ReactNode } from "react";
import { Box, Inline, Link } from "../compose";
import { urlOf } from "../utils";

export type MenuEntryProps = {
  href?: string;
  current?: boolean;
  variant?: "default" | "muted" | "block";
  size?: "default" | "sm";
  testid?: string;
  block?: boolean;
  className?: string;
  children?: ReactNode;
};

export function MenuEntry({
  href,
  current = false,
  variant = "default",
  size = "default",
  testid,
  block = false,
  className,
  children,
}: MenuEntryProps) {
  return (
    <>
      {href ? (
        <Link
          href={urlOf(href)}
          variant={variant}
          size={size}
          aria-current={(current ? "page" : undefined)}
          className={className}
          data-testid={testid}
        >
          {children}
        </Link>
      ) : block ? (
        <Box className={className} aria-current={(current ? "page" : undefined)} data-testid={testid}>
          {children}
        </Box>
      ) : (
        <Inline className={className} aria-current={(current ? "page" : undefined)} data-testid={testid}>
          {children}
        </Inline>
      )}
    </>
  );
}
