      ) : (
        <Group gap="2">
          <Button variant="default" size="default" href="#/" className="gap-2 flex-1" data-testid="action-primary">
            <LucideIcon name="book-open" size={16} />
            {labels.read}
          </Button>
          {((mode as string) === "icons") ? (
            <>
              <Button variant="outline" size="default" layout="icon" aria-label={labels.share} data-testid="post-share">
