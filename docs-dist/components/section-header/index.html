/**
 * The head of a section: an optional badge, the heading (h1 or h2) with an optional link opposite it, and an
 * optional lead. No wrapper: its parts are children of the section stack.
 */
import { type ReactNode } from "react";
import { Badge } from "../primitives/badge/badge";
import { Group } from "../primitives/group/group";
import { Link } from "../primitives/link/link";
import { Text } from "../primitives/text/text";
import { Title } from "../primitives/title/title";
import { defaults } from "../presets/defaults";
import { cn } from "../utils";
import type { SectionHeading, HeadLink } from "./section-frame.types";

export type SectionHeaderProps = {
  titleId: string;
  title?: string;
  heading?: SectionHeading;
  action?: HeadLink;
  lead?: string;
  center?: boolean;
};

export function SectionHeader({
  titleId,
  title = "",
  heading,
  action,
  lead,
  center = false,
}: SectionHeaderProps) {
  const level = heading?.level ?? 2;
  const scale = heading?.scale ?? "default";
  const linkHref = action?.href ?? "#/";
  return (
    <>
      {heading?.badge ? (
        <Badge variant="outline" size="lg" className={(center ? "mx-auto w-fit" : "w-fit")} data-testid="section-badge">
          {heading?.badge}
        </Badge>
      ) : null}
      {(((title as string) !== "") && action?.label) ? (
        <Group gap="4" align="center" justify="between" className="w-full flex-wrap">
          <Title
            as={level}
            size={((level === 1) ? "5xl" : "xl")}
            weight="semibold"
            className={
              (scale as string) === "display" ? "text-6xl tracking-tight md:text-8xl"
              : level === 1 ? "leading-tight"
              : "leading-7"
            }
            id={titleId}
          >
            {title}
          </Title>
          <Link
            href={linkHref}
            variant="default"
            className={cn(defaults.text.tiny, "shrink-0 font-medium leading-5")}
            data-testid="section-action"
          >
            {action?.label}
          </Link>
        </Group>
      ) : null}
      {(((title as string) !== "") && !action?.label) ? (
        <Title
          as={level}
          size={((level === 1) ? "5xl" : "xl")}
          weight="semibold"
          className={
            (scale as string) === "display" ? "text-6xl tracking-tight md:text-8xl"
            : level === 1 ? "leading-tight"
            : "leading-7"
          }
          id={titleId}
        >
          {title}
        </Title>
      ) : null}
      {(((title as string) === "") && action?.label) ? (
        <Group gap="4" align="center" justify="end" className="w-full">
