import { type ReactNode } from "react";
import { Badge, Group, Link, Text, Title } from "../compose";
import { defaults } from "../presets/defaults";

  titleId: string;

  action?: HeadLink;

  center?: boolean;

export function SectionHeader({
  titleId,
  title = "",
  heading,
  action,
  lead,
  center = false,
}: SectionHeaderProps) {
  const level = heading?.level ?? 2;
  const scale = heading?.scale ?? "default";
  const hasAction = action?.label && action?.href;
  return (
    <>
      {heading?.badge ? (
        <Badge variant="outline" size="lg" className={(center ? "mx-auto w-fit" : "w-fit")} data-testid="section-badge">
          {heading?.badge}
        </Badge>

      {(((title as string) !== "") && hasAction) ? (

              : level === 1 ? "leading-tight"
              : "leading-7"
            }
            id={titleId}
          >
            {title}
          </Title>
          <Link
            href={urlOf(action?.href)}
            variant="default"

            {action?.label}
