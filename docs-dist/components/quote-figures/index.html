/** Three figures in a row above the quotes of QuoteSplit. */
import { Fragment, type ReactNode } from "react";
import { Box } from "../primitives/box/box";
import { Stack } from "../primitives/stack/stack";
import { Text } from "../primitives/text/text";
import type { QuoteExtras } from "../blocks/quote-split.types";
import type { QuoteLabels } from "../blocks/testimonial-grid.types";

export type QuoteFiguresProps = {
  extras: QuoteExtras;
};

export function QuoteFigures({ extras }: QuoteFiguresProps) {
  return (
    <Box className="grid grid-cols-3 gap-4 border-b border-border pb-8" data-testid="quote-figures">
      {(extras.stats ?? []).map((s, i) => (
        <Fragment key={s.id}>
          {(i < 3) ? (
            <Stack gap="0" data-testid="quote-stat">
              <Text size="3xl" weight="bold">
                {s.value}
              </Text>
              <Text size="xs" className="text-muted-foreground">
                {s.label}
              </Text>
            </Stack>
          ) : null}
        </Fragment>
      ))}
    </Box>
  );
}
