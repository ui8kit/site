/**
 * A menu item: an optional icon, the label and an optional trailing mark (a check for selected items). tabindex
 * -1: the menu moves focus with the arrow keys.
 */
import { type ReactNode } from "react";
import { ButtonPlain } from "../primitives/button-plain/button-plain";
import { Inline } from "../primitives/inline/inline";
import { cn } from "../utils";

export type DropdownItemProps = {
  selected?: boolean;
  className?: string;
  children?: ReactNode;
  icon?: ReactNode;
  right?: ReactNode;
  onClick?: () => void;
};

export function DropdownItem({
  selected = false,
  className,
  children,
  icon,
  right,
  onClick,
}: DropdownItemProps) {
  return (
    <ButtonPlain
      type="button"
      role="menuitem"
      className={cn("group flex h-9 w-full items-center px-2 text-left text-foreground", className)}
      tabIndex={-1}
      data-slot="dropdown-item"
      data-state={(selected ? "selected" : "default")}
      onClick={() => onClick?.()}
    >
      <Inline
        className={"flex h-7 min-w-0 flex-1 items-center rounded p-1 group-hover:bg-muted group-focus-visible:bg-muted " +
          "group-data-[state=selected]:bg-muted"}
        data-slot="dropdown-item-content"
      >
        {icon ? (
          <Inline className="flex w-7 shrink-0 items-center pr-2" data-slot="dropdown-item-icon">
            {icon}
          </Inline>
        ) : null}
        <Inline className={cn("text-xs", "min-w-0 flex-1 truncate px-1 font-bold")} data-slot="dropdown-item-label">
          {children}
        </Inline>
        {right ? (
          <Inline className="flex shrink-0 items-center pr-1" data-slot="dropdown-item-right">
            {right}
          </Inline>
        ) : null}
      </Inline>
    </ButtonPlain>
  );
}
