	Kind any `json:"kind"`

	Expanded any `json:"expanded"`
	Class string `json:"className"`
}

	{{ vLabel := p.Label }}
	{{ vSelected := uiutils.Def(p.Selected, false) }}
	{{ vExpanded := uiutils.Def(p.Expanded, false) }}
	@buttonplain.ButtonPlain(buttonplain.ButtonPlainProps{Type: "button", Role: "treeitem", Class: uiutils.S(uiutils.Cnv(recipeFileTreeItem.Of(map[string]any{"kind": vKind}), p.Class)), Attrs: uiutils.Attrs("aria-selected", vSelected, "tabindex", uiutils.Cond(uiutils.T(vSelected), float64(0), float64(-1)), "aria-expanded", uiutils.Cond(uiutils.T(uiutils.Eq(vKind, "folder")), vExpanded, nil), "data-slot", "file-tree-item", "data-kind", vKind, "data-state", uiutils.Cond(uiutils.T(vSelected), "selected", "default"))}) {
