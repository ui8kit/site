package blocks

import (
	"github.com/codegen-x/ui8kit/packages/templ/ui/ui/image"
	"github.com/codegen-x/ui8kit/packages/templ/ui/ui/inline"
	uiutils "github.com/codegen-x/ui8kit/packages/templ/ui/utils"
)

// AvatarProps are the props of Avatar; the values are the JSON shapes of the TypeScript props.
type AvatarProps struct {
	Src any `json:"src"`
	Alt any `json:"alt"`
	Size any `json:"size"`
	Class string `json:"className"`
}

// An image in a round frame (recipe: size xs to xl, 24 to 64 px). Avatars are always round.
templ Avatar(p AvatarProps) {
	{{ vSrc := p.Src }}
	{{ vAlt := uiutils.Def(p.Alt, "") }}
	{{ vSize := uiutils.Def(p.Size, "default") }}
	@inline.Inline(inline.InlineProps{Class: uiutils.S(uiutils.Cnv(recipeAvatar.Of(map[string]any{"size": vSize}), p.Class)), Attrs: uiutils.Attrs("data-slot", "avatar")}) {
		@image.Image(image.ImageProps{Src: uiutils.S(vSrc), Alt: uiutils.S(vAlt), Class: "size-full object-cover", Fit: "cover", Size: "auto", Loading: "eager", Decoding: "auto", Attrs: uiutils.Attrs("draggable", "false")})
	}
}
