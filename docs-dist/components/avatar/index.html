    Avatar component

    Install

    npx ui8kit add avatar

    Pulls in: image, inline, utils

    Svelte

    Point the CLI at this runtime once, then add the item:

    npx ui8kit init --yes --framework svelte --dir src --registry-url https://registry.buildy.tw/svelte --strict-cdn
    npx ui8kit add avatar

import { type ReactNode } from "react";

/** Avatar: the classes of each value (shadcn/cva style). */
export const avatarVariants = variants({
  base: "relative inline-flex shrink-0 overflow-hidden bg-muted rounded-full",
  variants: {
    size: {
      xs: "size-6",
      sm: "size-8",
      default: "size-10",
      lg: "size-12",
      xl: "size-16",
    },
  },
  defaultVariants: { size: "default" },
});

export type AvatarProps = {
  src: string;
  alt?: string;
  size?: "xs" | "sm" | "default" | "lg" | "xl";
  className?: string;
};

export function Avatar({ src, alt = "", size = "default", className }: AvatarProps) {
  return (
    <Inline className={cn(avatarVariants({ size: size }), className)} data-slot="avatar">

        decoding="auto"
