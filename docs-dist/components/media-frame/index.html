/**
 * A picture in a fixed ratio that follows the theme. `src` is the picture of the light theme, `darkSrc` the one
 * of the dark theme (without it the dark theme shows `src`). Without both it shows the built-in neutral
 * placeholders (base64 data URIs, one light and one dark), so a page can be laid out before it has pictures; the
 * placeholder stands in for a real `src` and is switched by the theme token. `ratio` comes from the recipe; the
 * corners are the plain theme radius.
 */
import { type ReactNode } from "react";
import { Box } from "../primitives/box/box";
import { Image } from "../primitives/image/image";
import { at, cn, variants } from "../utils";
import { PLACEHOLDER } from "./media-frame.assets";

/** MediaFrame: the classes of each value (shadcn/cva style). */
export const mediaFrameVariants = variants({
  base: "relative w-full overflow-hidden bg-muted text-muted-foreground inset-ring inset-ring-border rounded",
  variants: {
    ratio: {
      "16/9": "aspect-video",
      "4/3": "aspect-4/3",
      "1/1": "aspect-square",
      "3/4": "aspect-3/4",
      "21/9": "aspect-21/9",
    },
  },
  defaultVariants: { ratio: "16/9" },
});

export type MediaFrameProps = {
  alt: string;
  src?: string;
  darkSrc?: string;
  ratio?: "16/9" | "4/3" | "1/1" | "3/4" | "21/9";
  className?: string;
};

export function MediaFrame({ alt, src, darkSrc, ratio = "16/9", className }: MediaFrameProps) {
  const lightSrc = src ?? at(PLACEHOLDER, "light");
  const nightSrc = darkSrc ?? src ?? at(PLACEHOLDER, "dark");
  return (
    <Box
      className={cn(mediaFrameVariants({ ratio: ratio }), className)}
      data-slot="media-frame"
      data-placeholder={(src ? "false" : "true")}
    >
      <Image
        src={lightSrc}
        alt={alt}
        className="absolute inset-0 size-full object-cover dark:hidden"
        data-theme-src="light"
      />
      <Image
        src={nightSrc}
        alt={alt}
        className="absolute inset-0 hidden size-full object-cover dark:block"
        data-theme-src="dark"
      />
    </Box>
  );
}

    Props

    Names and types come from the xgen specs of the source. Standard HTML attributes (id, role, tabIndex, aria-*) pass through to the root element; the class prop is className in React.

    A picture in a fixed ratio that follows the theme. `src` is the picture of the light theme, `darkSrc` the one of the dark theme (without it the dark theme shows `src`). Without both it shows the built-in neutral placeholders (base64 data URIs, one light and one dark), so a page can be laid out before it has pictures; the placeholder stands in for a real `src` and is switched by the theme token. `ratio` comes from the recipe; the corners are the plain theme radius.
