 * Delivery windows by day: each day lists its windows as radio chips; closed windows are disabled and struck

 */
import { type ReactNode } from "react";
import { Box } from "../primitives/box/box";
import { Group } from "../primitives/group/group";

import { Text } from "../primitives/text/text";
import { cn, tpl } from "../utils";
import type { Day } from "./slot-grid.types";

export type SlotGridProps = {
  days: Day[];
  name: string;
  item?: "card" | "plain";
};

export function SlotGrid({ days, name, item = "card" }: SlotGridProps) {
  return (
    <Stack gap="4" align="stretch" data-testid="days">
      {(days ?? []).map((d) => (
        <Stack key={d.id}
          gap="4"
          className={
            (item as string) === "card" ? "rounded bg-card inset-ring inset-ring-border p-4"
            : "border-b border-border pb-4"
          }

        >
          <Group gap="4" align="baseline" justify="between" className="flex-wrap">
            <Text size="sm" weight="semibold">

              {d.note}
            </Text>
          </Group>
          <Box className="grid grid-cols-2 gap-2 sm:grid-cols-3">
            {(d.slots ?? []).map((s) => (
              <Label key={s}
                className={"flex items-center gap-2 rounded px-4 py-2 text-sm inset-ring inset-ring-border has-checked:bg-muted " +
                  "has-checked:inset-ring-foreground"}
              >
                <Radio name={name} value={tpl("{0} {1}", d.id, s)} />
                <Inline>
                  {s}
                </Inline>
              </Label>
            ))}
            {(d.closed ?? []).map((s) => (
              <Label key={s}
                className={cn(
                  "flex items-center gap-2 rounded px-4 py-2 text-sm inset-ring inset-ring-border has-checked:bg-muted " +
                    "has-checked:inset-ring-foreground",
                  "text-muted-foreground line-through"

                <Inline>

}
