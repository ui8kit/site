type MediaContactsProps struct {
	Extras any `json:"extras"`
	Labels any `json:"labels"`
}

// The ways to get help of MediaList as wide outline buttons, then three support figures as rows in a card.
templ MediaContacts(p MediaContactsProps) {
	{{ vExtras := p.Extras }}
	{{ vLabels := p.Labels }}
	@stack.Stack(stack.StackProps{Gap: "6", Align: "stretch"}) {
		@stack.Stack(stack.StackProps{Gap: "2", Align: "stretch"}) {
			@text.Text(text.TextProps{Size: "sm", Weight: "semibold"}) {
				{ uiutils.Text(uiutils.Get(vLabels, "list")) }
			}
			for _, vC := range uiutils.L(uiutils.Get(vExtras, "channels")) {
				@button.Button(button.ButtonProps{Variant: "outline", Size: "default", Href: "#/", Class: "w-full gap-2", Attrs: uiutils.Attrs("data-testid", "media-channel")}) {
					@lucideicon.LucideIcon(lucideicon.LucideIconProps{Name: uiutils.S(uiutils.Get(vC, "icon")), Size: 16})
					{ uiutils.Text(uiutils.Get(vC, "title")) }
				}
			}
		}
		@stack.Stack(stack.StackProps{Gap: "2", Align: "stretch", Class: "rounded bg-card inset-ring inset-ring-border p-4"}) {
			@text.Text(text.TextProps{Size: "sm", Weight: "semibold"}) {
				{ uiutils.Text(uiutils.Get(vLabels, "figures")) }
			}
			for vI, vS := range uiutils.L(uiutils.Get(vExtras, "stats")) {
				if uiutils.Lt(float64(vI), float64(3)) {
					@group.Group(group.GroupProps{Gap: "4", Align: "center", Justify: "between", Attrs: uiutils.Attrs("data-testid", "media-stat")}) {

							{ uiutils.Text(uiutils.Get(vS, "value")) }
						}
					}
				}
			}
		}
	}
}
