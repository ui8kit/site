/**
 * The head row of a hero band: skip link, brand, menu links (the current one carries aria-current), login and a
 * call to action.
 */
import { type ReactNode } from "react";
import { Block } from "../primitives/block/block";
import { Button } from "../primitives/button/button";
import { Group } from "../primitives/group/group";
import { LinkPlain } from "../primitives/link-plain/link-plain";
import { Link } from "../primitives/link/link";
import { defaults } from "../presets/defaults";
import { cn } from "../utils";
import type { NavLink } from "../blocks/nav-bar.types";

export type HeroHeadProps = {
  brand?: string;
  links?: NavLink[];
  current?: string;
  login?: string;
  loginHref?: string;
  cta?: string;
  href?: string;
  skip?: string;
  navLabel?: string;
};

export function HeroHead({
  brand,
  links,
  current = "",
  login,
  loginHref = "#/login",
  cta,
  href,
  skip,
  navLabel,
}: HeroHeadProps) {
  return (
    <Block
      tag="header"
      className={cn("flex w-full flex-wrap items-center justify-between pr-4", defaults.hero.head)}
      data-testid="hero-head"
    >
      {skip ? (
        <LinkPlain
          href="#main-content"
          className={"sr-only focus:not-sr-only focus:fixed focus:left-2 focus:top-2 focus:z-50 focus:rounded focus:bg-background " +
            "focus:px-4 focus:py-2 focus:text-sm focus:text-foreground focus:inset-ring focus:inset-ring-border"}
          data-testid="skip-link"
        >
          {skip}
        </LinkPlain>
      ) : null}
      <Link
        href="#/"
        variant="default"
        className={cn(defaults.text.small, "font-medium leading-6 text-background")}
        data-testid="brand"
      >
        {brand}
      </Link>
      <Block tag="nav" aria-label={navLabel} data-testid="site-nav">
        <Group gap="8" align="center" className="flex-wrap">
          {(links ?? []).map((l) => (
            <Link key={l.id}
              href={(l.href ?? "#/")}
              variant="default"
              aria-current={((l.id === current) ? "page" : undefined)}
              className="text-xs font-medium leading-5 text-background"
              data-testid="nav-link"
            >
              {l.name}
            </Link>
          ))}
        </Group>
      </Block>
      <Group gap="2" align="center">
        {login ? (
          <Button
            href={loginHref}
            variant="ghost"
            size="sm"
            className="text-background hover:bg-background/10"
            data-testid="nav-login"
          >
            {login}
          </Button>
        ) : null}
        {cta ? (
          <Button href={(href ?? "#/")} variant="secondary" size="sm" data-testid="header-cta">
            {cta}
          </Button>
        ) : null}
      </Group>
    </Block>
  );
}
