/** One dot per step of StoryTabs; the dot of the active step is filled. */
import { type ReactNode } from "react";

package blocks

import (
	"github.com/codegen-x/ui8kit/packages/templ/ui/ui/group"
	"github.com/codegen-x/ui8kit/packages/templ/ui/ui/inline"
	uiutils "github.com/codegen-x/ui8kit/packages/templ/ui/utils"
)

// StoryDotsProps are the props of StoryDots; the values are the JSON shapes of the TypeScript props.
type StoryDotsProps struct {
	Steps any `json:"steps"`
	Active any `json:"active"`
}

// One dot per step of StoryTabs; the dot of the active step is filled.
templ StoryDots(p StoryDotsProps) {
	{{ vSteps := p.Steps }}
	{{ vActive := p.Active }}
	@group.Group(group.GroupProps{Gap: "2", Align: "center"}) {
		for _, vS := range uiutils.L(vSteps) {
			@inline.Inline(inline.InlineProps{Class: uiutils.S(uiutils.Cnv("size-2 rounded-full", uiutils.Cond(uiutils.T(uiutils.Eq(uiutils.Get(vS, "id"), vActive)), "bg-primary", "bg-border"))), Attrs: uiutils.Attrs("data-testid", "story-dot")})
		}
	}
}
