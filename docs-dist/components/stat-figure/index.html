/** One key number of StatsBand: a card (StatTile), or the value over its label, optionally under a Lucide icon. */
import { type ReactNode } from "react";
import { LucideIcon } from "../primitives/lucide-icon/lucide-icon";
import { Stack } from "../primitives/stack/stack";
import { Text } from "../primitives/text/text";
import { StatTile } from "../components/stat-tile";
import { defaults } from "../presets/defaults";
import { cn } from "../utils";
import type { Stat } from "../blocks/stats-band.types";

export type StatFigureProps = {
  stat: Stat;
  item?: "plain" | "card" | "icon";
  align?: "start" | "center";
};

export function StatFigure({ stat, item = "plain", align = "center" }: StatFigureProps) {
  return (
    <>
      {((item as string) === "card") ? (
        <StatTile label={stat.label} value={stat.value} icon={stat.icon} hint={stat.hint} />
      ) : (
        <Stack
          gap="6"
          align={(((align as string) === "center") ? "center" : "start")}
          className={cn((((align as string) === "center") ? "text-center" : ""), defaults.stats.column)}
          data-testid="item"
        >
          {((item as string) === "icon") ? (
            <LucideIcon name={stat.icon} size={24} />
          ) : null}
          <Stack gap="2" align={(((align as string) === "center") ? "center" : "start")}>
            <Text size="5xl" weight="medium" className={cn("flex items-center", defaults.stats.figure)}>
              {stat.value}
            </Text>
            <Text size="sm" weight="semibold" className="leading-5 text-muted-foreground">
              {stat.label}
            </Text>
          </Stack>
        </Stack>
      )}
    </>
  );
}
