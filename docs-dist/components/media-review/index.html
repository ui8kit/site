/** The featured review of MediaList in a muted card: a label with the stars, the text in quotes and the author. */
import { Fragment, type ReactNode } from "react";
import { Group } from "../primitives/group/group";
import { Stack } from "../primitives/stack/stack";
import { Text } from "../primitives/text/text";
import { Rating } from "../components/rating";
import { tpl } from "../utils";
import type { MediaExtras, MediaListLabels } from "../blocks/media-list.types";

export type MediaReviewProps = {
  extras: MediaExtras;
  labels: MediaListLabels;
};

export function MediaReview({ extras, labels }: MediaReviewProps) {
  return (
    <>
      {(extras.reviews ?? []).map((r, i) => (
        <Fragment key={r.id}>
          {(i < 1) ? (
            <Stack gap="4" align="stretch" className="rounded bg-muted p-6" data-testid="media-review">
              <Group gap="4" align="center" justify="between" className="flex-wrap">
                <Text size="sm" weight="medium">
                  {labels.featured}
                </Text>
                <Rating value={r.rating} label={tpl(labels.rating, r.rating)} size={16} />
              </Group>
              <Text size="base">
                {tpl("“{0}”", r.text)}
              </Text>
              <Text size="sm" className="text-muted-foreground">
                {tpl("{0} · {1}", r.author, r.date)}
              </Text>
            </Stack>
          ) : null}
        </Fragment>
      ))}
    </>
  );
}
