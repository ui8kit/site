/** Empty list or nothing found: a round icon, a title, a hint and an optional action. */
import { type ReactNode } from "react";
import { Box } from "../primitives/box/box";
import { IconBadge } from "../primitives/iconbadge/iconbadge";
import { LucideIcon } from "../primitives/lucide-icon/lucide-icon";
import { Text } from "../primitives/text/text";
import { cn } from "../utils";

export type EmptyStateProps = {
  icon?: string;
  title: string;
  text?: string;
  className?: string;
  action?: ReactNode;
};

export function EmptyState({ icon = "inbox", title, text, className, action }: EmptyStateProps) {
  return (
    <Box
      className={cn(
        "flex min-w-0 flex-col items-center gap-4 rounded bg-card px-6 py-14 text-center inset-ring inset-ring-border",
        className
      )}
      role="status"
    >
      <IconBadge size="lg">
        <LucideIcon name={icon} size={20} />
      </IconBadge>
      <Text className="m-0 text-sm font-bold text-foreground">
        {title}
      </Text>
      {text ? (
        <Text className="m-0 max-w-80 text-xs font-medium text-muted-foreground">
          {text}
        </Text>
      ) : null}
      {action}
    </Box>
  );
}

	uiutils "github.com/codegen-x/ui8kit/packages/templ/ui/utils"
)

// EmptyStateProps are the props of EmptyState; the values are the JSON shapes of the TypeScript props.
type EmptyStateProps struct {
	Icon any `json:"icon"`
	Title any `json:"title"`
	Text any `json:"text"`
	Class string `json:"className"`
	Action templ.Component `json:"-"`
}

// Empty list or nothing found: a round icon, a title, a hint and an optional action.
templ EmptyState(p EmptyStateProps) {
	{{ vIcon := uiutils.Def(p.Icon, "inbox") }}
	{{ vTitle := p.Title }}
	{{ vText := p.Text }}
	@box.Box(box.BoxProps{Class: uiutils.S(uiutils.Cnv("flex min-w-0 flex-col items-center gap-4 rounded bg-card px-6 py-14 text-center inset-ring inset-ring-border", p.Class)), Attrs: uiutils.Attrs("role", "status")}) {
		@iconbadge.IconBadge(iconbadge.IconBadgeProps{Size: "lg"}) {
			@lucideicon.LucideIcon(lucideicon.LucideIconProps{Name: uiutils.S(vIcon), Size: 20})
		}
		@text.Text(text.TextProps{Class: "m-0 text-sm font-bold text-foreground"}) {
			{ uiutils.Text(vTitle) }
		}
		if uiutils.T(vText) {
			@text.Text(text.TextProps{Class: "m-0 max-w-80 text-xs font-medium text-muted-foreground"}) {
				{ uiutils.Text(vText) }
			}
		}
		if p.Action != nil {
			@p.Action
		}
	}
