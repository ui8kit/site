/**
 * Input is a single-line native text field.
 * Supports types, validation attrs, and wireframe variant/size classes.
 */

import { forwardRef, type ReactNode, type InputHTMLAttributes } from "react";
import { cn, variants, type VariantKey } from "../../utils";
import { defaultIfEmpty } from "../../utils/expr";

/** Input: the classes of each value of each prop (shadcn/cva style). */
export const inputVariants = variants({
  base:
    "w-full min-w-0 bg-background text-foreground inset-ring inset-ring-input outline-none " +
    "placeholder:text-muted-foreground disabled:cursor-not-allowed disabled:opacity-50 " +
    "focus-visible:inset-ring-ring rounded",
  variants: {
    size: {
      sm: "h-8 px-3 text-base md:text-sm",
      default: "h-10 px-3 text-base md:text-sm",
      lg: "h-12 px-4 text-base",
    },
    variant: {
      default: "",
      ghost: "bg-muted inset-ring-transparent",
    },
  },
  defaultVariants: { size: "default", variant: "default" },
});

export type InputVariant = VariantKey<typeof inputVariants, "variant">;
export type InputSize = VariantKey<typeof inputVariants, "size">;

/**
 * Input renders a styled single-line form control.
 */
export type InputProps = Omit<
  InputHTMLAttributes<HTMLInputElement>,
  | "className" | "children" | "variant" | "size" | "type" | "disabled" | "required"
> & {
  variant?: InputVariant;
  size?: InputSize;
  type?: string;
  disabled?: boolean;
  required?: boolean;
  className?: string;
};

export const Input = forwardRef<HTMLInputElement, InputProps>(function Input(
  { variant, size, type, disabled, required, className, ...rest },
  ref
) {
  const cls = cn(inputVariants({ size, variant }), className);
  return (
    <input
      ref={ref}
      type={defaultIfEmpty(type, "text")}
      className={cls || undefined}
      disabled={disabled || undefined}
      required={required || undefined}
      {...rest}
    />
  );
});
Input.displayName = "Input";
