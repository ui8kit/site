    Kit Label

    Install

    npx ui8kit add kit-label

/**
 * Label names a form control.
 */

import { forwardRef, type ReactNode, type LabelHTMLAttributes } from "react";
import { labelClasses } from "./label.shared";
import { safeRest } from "../../utils/expr";

/**
 * Label renders a native label element.
 */
export type LabelProps = Omit<LabelHTMLAttributes<HTMLLabelElement>, "className" | "children"> & {
  className?: string;
  children?: ReactNode;
};

export const Label = forwardRef<HTMLLabelElement, LabelProps>(function Label(
  { className, children, ...rawRest },
  ref
) {
  const rest = safeRest(rawRest, { blank: ["for", "htmlfor"] });
  const cls = labelClasses({ className });
  return (
    <label ref={ref} className={cls || undefined} {...rest}>
      {children}
    </label>
  );
});
Label.displayName = "Label";

    // Code generated by @ui8kit/codegen. DO NOT EDIT.
/**
 * Label names a form control.
 */

import { splitProps, type JSX, type Component } from "solid-js";
import { labelClasses } from "./label.shared";
import { safeRest } from "../../utils/expr";

/**
 * Label renders a native label element.
 */
export type LabelProps = Omit<JSX.LabelHTMLAttributes<HTMLLabelElement>, "class" | "className" | "children"> & {
  class?: string;
  className?: string;
  children?: JSX.Element;
};

export const Label: Component<LabelProps> = (props) => {
  const [local, rest] = splitProps(props, ["class", "className", "children"]);
  const cls = () => labelClasses({ className: local.class ?? local.className });
  return (
    <label {...(cls() ? { class: cls() } : {})} {...safeRest(rest, { blank: ["for", "htmlfor"] })}>
      {local.children}
    </label>
  );
};

package label

import (
	uiutils "github.com/ui8kit/codegen/generated/templ/utils"
)

// LabelProps configures Label.
type LabelProps struct {
	// Class — token class names supplied by the consumer
	Class string
	// For — id of the labeled control
	For string
	Attrs templ.Attributes
}

// LabelClasses composes Label classes without rendering —
// the server-composition escape hatch for custom root elements (no asChild in templ).
func LabelClasses(p LabelProps) string {
	return uiutils.Cn(p.Class)
}

// Label renders the Label contract.
templ Label(p LabelProps) {
	<label
		if LabelClasses(p) != "" {
			class={ LabelClasses(p) }
		}
		if !uiutils.HasAttr(p.Attrs, "for") {
			if uiutils.IsSetStr(p.For) {
				for={ p.For }
			}
		}
		{ uiutils.RestAttrs(p.Attrs, false, "for")... }
	>
		{ children... }
	</label>
}
