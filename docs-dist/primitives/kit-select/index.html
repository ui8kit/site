  required?: boolean;
  disabled?: boolean;
  options?: SelectOptionItem[];
}>();

const attrs = useAttrs();
const callerClass = computed(() => attrs.class as string | undefined);
const restAttrs = computed(() => {
  const { class: _class, ...restEntries } = attrs;

		}

			}

			}
