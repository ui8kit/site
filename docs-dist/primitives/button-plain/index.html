import { forwardRef, type ReactNode, type ButtonHTMLAttributes } from "react";
import { cn } from "../../utils";
import { oneOfOr } from "../../utils/expr";

/**
 * ButtonPlain is a bare button element: no styling beyond Class. Patterns (tabs, tree rows, menu items) style it
 * themselves.
 */
export type ButtonPlainProps = Omit<
  ButtonHTMLAttributes<HTMLButtonElement>,
  | "className" | "children" | "type" | "disabled"
> & {
  type?: string;
  disabled?: boolean;
  className?: string;
  children?: ReactNode;
};

export const ButtonPlain = forwardRef<HTMLButtonElement, ButtonPlainProps>(function ButtonPlain(
  { type, disabled, className, children, ...rest },
  ref
) {
  const cls = cn((disabled ? "pointer-events-none opacity-50" : ""), className);
  return (
    <button
      ref={ref}
      type={oneOfOr(type, "button", "submit", "reset")}
      className={cls || undefined}
      disabled={disabled || undefined}
      {...rest}
    >
      {children}
    </button>
  );
});

ButtonPlain is a bare button element: no styling beyond Class. Patterns (tabs, tree rows, menu items) style it themselves.

package buttonplain

import (
	uiutils "github.com/codegen-x/ui8kit/packages/templ/ui/utils"
)

// ButtonPlainProps configures ButtonPlain.
type ButtonPlainProps struct {
	// Class — adds Tailwind utilities
	Class string
	// Type — button type: button, submit, reset (ignored when Href is set)
	Type string
	// Disabled — blocks interaction
	Disabled bool
	// ID — element id
	ID string
	// Role — ARIA role override
	Role string
	// TabIndex — tab order override
	TabIndex string
	// AriaLabel — accessible name for unlabeled controls
	AriaLabel string
	Attrs templ.Attributes
}

// ButtonPlainClasses composes ButtonPlain classes without rendering —
// the server-composition escape hatch for custom root elements.
func ButtonPlainClasses(p ButtonPlainProps) string {
	return uiutils.Compose(ButtonPlainVariants, map[string]string{}, uiutils.If(p.Disabled, "pointer-events-none opacity-50", ""), p.Class)
}

// ButtonPlain renders the ButtonPlain contract.
templ ButtonPlain(p ButtonPlainProps) {
	<button
		type={ uiutils.OneOfOr(p.Type, "button", "submit", "reset") }
		class={ ButtonPlainClasses(p) }
		disabled?={ p.Disabled }
		if uiutils.IsSetStr(p.ID) {
			id={ p.ID }
		}
		if uiutils.IsSetStr(p.Role) {
			role={ p.Role }
		}
		if uiutils.IsSetStr(p.TabIndex) {
			tabindex={ p.TabIndex }
		}
		if uiutils.IsSetStr(p.AriaLabel) {
			aria-label={ p.AriaLabel }
		}
		{ p.Attrs... }
	>
		{ children... }
	</button>
}
