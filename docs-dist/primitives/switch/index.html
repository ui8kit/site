/** Switch: the classes of each value of each prop (shadcn/cva style). */

  base:
    "relative shrink-0 appearance-none bg-muted inset-ring inset-ring-input checked:bg-primary " +
    "checked:inset-ring-primary switch-knob disabled:cursor-not-allowed disabled:opacity-50 rounded-full",
  variants: {
    size: {
      default: "h-5 w-9 [--knob:12px] [--pad:4px]",
      sm: "h-4 w-8 [--knob:12px] [--pad:2px]",
    },
  },
  defaultVariants: { size: "default" },
});

export type SwitchSize = VariantKey<typeof switchVariants, "size">;

/**
 * Switch renders input[type=checkbox] with switch role attrs.
 */
export type SwitchProps = Omit<
  InputHTMLAttributes<HTMLInputElement>,
  | "className" | "children" | "size" | "checked" | "disabled" | "required"
> & {
  size?: SwitchSize;
  checked?: boolean;
  disabled?: boolean;
  required?: boolean;
  className?: string;
};

export const Switch = forwardRef<HTMLInputElement, SwitchProps>(function Switch(
  { size, checked, disabled, required, className, ...rest },

      disabled={disabled || undefined}

/*
Switch is a toggle control (checkbox with switch ARIA semantics).
Use for on/off settings; wireframe uses native checkbox styling.
*/

package formswitch

import (
	uiutils "github.com/codegen-x/ui8kit/packages/templ/ui/utils"
)

// SwitchProps configures Switch.
type SwitchProps struct {
	// Size — sets density preset
	Size string
	// Class — adds Tailwind utilities
	Class string
	Name string
	Value string
	Checked bool
	Disabled bool
	Required bool
	ID string
	TabIndex string
	AriaLabel string
	Attrs templ.Attributes
}

// SwitchClasses composes Switch classes without rendering —
// the server-composition escape hatch for custom root elements.
func SwitchClasses(p SwitchProps) string {
	return uiutils.Compose(SwitchVariants, map[string]string{
		"size": p.Size,
	}, p.Class)
}

// Switch renders the Switch contract.
templ Switch(p SwitchProps) {
	<input
		if uiutils.IsSetStr(p.ID) {
			id={ p.ID }
		}
		type="checkbox"
		if uiutils.IsSetStr(p.Name) {
			name={ p.Name }
		}
		if uiutils.IsSetStr(p.Value) {
			value={ p.Value }
		}
		class={ SwitchClasses(p) }
		checked?={ p.Checked }
		disabled?={ p.Disabled }
		required?={ p.Required }
		if uiutils.IsSetStr(p.TabIndex) {
			tabindex={ p.TabIndex }
		}
		if uiutils.IsSetStr(p.AriaLabel) {
			aria-label={ p.AriaLabel }
		}
		role="switch"
		aria-checked={ uiutils.BoolStr(p.Checked) }
		{ p.Attrs... }
	/>
}
