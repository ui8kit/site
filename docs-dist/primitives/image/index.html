	uiutils "github.com/codegen-x/ui8kit/packages/templ/ui/utils"
)

// ImageProps configures Image.
type ImageProps struct {
	// Variant — sets appearance preset
	Variant string
	// Size — sets density preset
	Size string
	// Class — adds Tailwind utilities
	Class string
	Src string
	SrcSet string
	Sizes string
	// Alt — alternative text; ignored when Decorative
	Alt string
	Width string
	Height string
	// Fit — object-fit preset
	Fit string
	// Position — object-position preset
	Position string
	// Aspect — aspect-ratio preset
	Aspect string
	// Loading — eager or lazy (default lazy)
	Loading string
	// Decoding — sync, auto, or async (default async)
	Decoding string
	FetchPriority string
	// Decorative — marks the image as purely presentational
	Decorative bool
	ID string
	Attrs templ.Attributes
}

// ImageClasses composes Image classes without rendering —
// the server-composition escape hatch for custom root elements.
func ImageClasses(p ImageProps) string {
	return uiutils.Compose(ImageVariants, map[string]string{
		"aspect": p.Aspect,
		"fit": p.Fit,
		"position": p.Position,
		"size": p.Size,
		"variant": p.Variant,
	}, p.Class)
}

// Image renders the Image contract.
templ Image(p ImageProps) {
	<img
		if uiutils.IsSetStr(p.Src) {
			src={ p.Src }
		}
		if uiutils.IsSetStr(p.SrcSet) {
			srcset={ p.SrcSet }
		}
		if uiutils.IsSetStr(p.Sizes) {
			sizes={ p.Sizes }
		}
		alt={ uiutils.If(p.Decorative, "", p.Alt) }
		if uiutils.IsSetStr(p.Width) {
			width={ p.Width }
		}
		if uiutils.IsSetStr(p.Height) {
			height={ p.Height }
		}
		loading={ uiutils.OneOfOr(p.Loading, "lazy", "eager") }
		decoding={ uiutils.OneOfOr(p.Decoding, "async", "sync", "auto") }
		if uiutils.IsSetStr(p.FetchPriority) {
			fetchpriority={ p.FetchPriority }
		}
		class={ ImageClasses(p) }
		if uiutils.IsSetStr(p.ID) {
			id={ p.ID }
		}
		if p.Decorative {
			aria-hidden={ "true" }
		}
		{ p.Attrs... }
	/>
}

type PictureProps struct {

	Class string
	Attrs templ.Attributes
