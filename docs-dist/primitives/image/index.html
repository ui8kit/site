/** Image: the classes of each value of each prop (shadcn/cva style). */

  base: "block max-w-full",
  variants: {
    aspect: {
      auto: "aspect-auto",
      square: "aspect-square",
      video: "aspect-video",
    },
    fit: {
      contain: "object-contain",
      cover: "object-cover",
      fill: "object-fill",
      none: "object-none",
      "scale-down": "object-scale-down",
    },
    position: {
      bottom: "object-bottom",
      center: "object-center",
      left: "object-left",
      "left-bottom": "object-left-bottom",
      "left-top": "object-left-top",
      right: "object-right",
      "right-bottom": "object-right-bottom",
      "right-top": "object-right-top",
      top: "object-top",
    },
    size: {
      auto: "",
      default: "",
      full: "h-full w-full",
      xs: "size-6",
      sm: "size-8",
      md: "size-12",
      lg: "size-16",
      xl: "size-20",
    },
    variant: {
      default: "",
      logo: "object-contain",
    },
  },
  defaultVariants: { aspect: "auto", fit: "cover", position: "center", size: "auto", variant: "default" },
});

export type ImageVariant = VariantKey<typeof imageVariants, "variant">;
export type ImageSize = VariantKey<typeof imageVariants, "size">;
export type ImageFit = VariantKey<typeof imageVariants, "fit">;
export type ImagePosition = VariantKey<typeof imageVariants, "position">;
export type ImageAspect = VariantKey<typeof imageVariants, "aspect">;

/**
 * Image renders an accessible img element.
 */
export type ImageProps = Omit<
  ImgHTMLAttributes<HTMLImageElement>,
  | "className" | "children" | "variant" | "size" | "alt" | "fit" | "position" | "aspect" | "loading"
  | "decoding" | "decorative"
> & {
  variant?: ImageVariant;
  size?: ImageSize;
  alt?: string;
  fit?: ImageFit;
  position?: ImagePosition;
  aspect?: ImageAspect;
  loading?: string;
  decoding?: string;
  decorative?: boolean;
  className?: string;
};

export const Image = forwardRef<HTMLImageElement, ImageProps>(function Image(
  { variant, size, alt, fit, position, aspect, loading, decoding, decorative, className, ...rest },
  ref
) {
  const cls = cn(imageVariants({ aspect, fit, position, size, variant }), className);
  return (
    <img
      ref={ref}
      alt={((decorative ? "" : alt)) ?? ""}
      loading={oneOfOr(loading, "lazy", "eager")}
      decoding={oneOfOr(decoding, "async", "sync", "auto")}
      className={cls || undefined}
      aria-hidden={decorative ? "true" : undefined}
      {...rest}
    />
  );
});
Image.displayName = "Image";
