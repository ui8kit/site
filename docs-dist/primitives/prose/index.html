package compose

import "github.com/ui8kit/codegen/generated/templ/ui/box"

// ProseProps: Prose: Box <div> + recipe prose.
type ProseProps struct {
	Size string
	Width string
	box.BoxProps
}

func proseInner(p ProseProps) box.BoxProps {
	in := p.BoxProps
	in.Class = Join(Rc("prose", map[string]string{"size": p.Size, "width": p.Width}), p.BoxProps.Class)
	in.Tag = "div"
	return in
}

// Prose renders Box with the composition classes.
templ Prose(p ProseProps) {
	@box.Box(proseInner(p)) {
		{ children... }
	}
}
