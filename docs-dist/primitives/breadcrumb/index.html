import type { ComponentProps } from "react";
import { Box } from "../kit/ui/box/box";
import { Link } from "../kit/ui/link/link";
import { Text } from "../kit/ui/text/text";
import { cn } from "../kit/utils";

export type BreadcrumbItem = { label?: string; href?: string; current?: boolean; disabled?: boolean };
export type BreadcrumbProps = Omit<ComponentProps<typeof Box>, "tag" | "children"> & { items?: BreadcrumbItem[]; "aria-label"?: string; className?: string };

/** Breadcrumb: Box nav > ol > li, Link (href, not disabled) or Text span; aria-current="page" on the current item. */
export function Breadcrumb({ items, "aria-label": ariaLabel, className, ...rest }: BreadcrumbProps) {
  return (
    <Box tag="nav" className={cn("text-sm", className)} aria-label={ariaLabel || "Breadcrumb"} {...rest}>
      <Box tag="ol" className="flex flex-wrap items-center gap-2">
        {(items ?? []).map((it, i) => (
          <Box key={i} tag="li" className="inline-flex items-center">
            {it.href && !it.disabled
              ? <Link href={it.href} className={cn("text-sm", (it.current ? "font-medium" : it.disabled ? "text-muted-foreground" : "text-primary underline-offset-4 hover:underline"))} aria-current={it.current ? "page" : undefined}>{it.label}</Link>
              : <Text tag="span" className={cn("text-sm", (it.current ? "font-medium" : it.disabled ? "text-muted-foreground" : "text-primary underline-offset-4 hover:underline"))} aria-current={it.current ? "page" : undefined}>{it.label}</Text>}
          </Box>
        ))}
      </Box>
    </Box>
  );
}

import { For, splitProps, type ComponentProps } from "solid-js";
import { Box } from "../kit/ui/box/box.solid";
import { Link } from "../kit/ui/link/link.solid";
import { Text } from "../kit/ui/text/text.solid";
import { cn } from "../kit/utils";

export type BreadcrumbItem = { label?: string; href?: string; current?: boolean; disabled?: boolean };
export type BreadcrumbProps = Omit<ComponentProps<typeof Box>, "tag" | "children" | "class" | "className"> & { items?: BreadcrumbItem[]; "aria-label"?: string; class?: string; className?: string };

/** Breadcrumb: Box nav > ol > li, Link (href, not disabled) or Text span; aria-current="page" on the current item. */
export function Breadcrumb(props: BreadcrumbProps) {
  const [local, rest] = splitProps(props, ["items", "aria-label", "class", "className"]);
  return (
    <Box tag="nav" class={cn("text-sm", local.class ?? local.className)} aria-label={local["aria-label"] || "Breadcrumb"} {...rest}>
      <Box tag="ol" class="flex flex-wrap items-center gap-2">
        <For each={local.items ?? []}>{(it) => (
          <Box tag="li" class="inline-flex items-center">
            {it.href && !it.disabled
              ? <Link href={it.href} class={cn("text-sm", (it.current ? "font-medium" : it.disabled ? "text-muted-foreground" : "text-primary underline-offset-4 hover:underline"))} aria-current={it.current ? "page" : undefined}>{it.label}</Link>
              : <Text tag="span" class={cn("text-sm", (it.current ? "font-medium" : it.disabled ? "text-muted-foreground" : "text-primary underline-offset-4 hover:underline"))} aria-current={it.current ? "page" : undefined}>{it.label}</Text>}
          </Box>
        )}</For>
      </Box>
    </Box>
  );
}

    // Code generated by packages/kit-compose/src/build.ts from src/map.ts + variants/. DO NOT EDIT.

package compose

import (
	"github.com/ui8kit/codegen/generated/templ/ui/box"
	"github.com/ui8kit/codegen/generated/templ/ui/link"
	"github.com/ui8kit/codegen/generated/templ/ui/text"
)

// BreadcrumbItem is one step of the trail.
type BreadcrumbItem struct {
	Label    string
	Href     string
	Current  bool
	Disabled bool
}

// BreadcrumbProps: Box nav > ol > li, Link (href, not disabled) or Text span.
type BreadcrumbProps struct {
	Items     []BreadcrumbItem
	AriaLabel string
	box.BoxProps
}

func breadcrumbItemClass(it BreadcrumbItem) string {
	switch {
	case it.Current:
		return "text-sm font-medium"
	case it.Disabled:
		return "text-sm text-muted-foreground"
	}
	return "text-sm text-primary underline-offset-4 hover:underline"
}

func breadcrumbCurrent(it BreadcrumbItem) templ.Attributes {
	if it.Current {
		return templ.Attributes{"aria-current": "page"}
	}
	return templ.Attributes{}
}

func breadcrumbNav(p BreadcrumbProps) box.BoxProps {
	in := p.BoxProps
	in.Tag = "nav"
	in.Class = Join("text-sm", p.BoxProps.Class)
	in.Attrs = WithAttrs(in.Attrs, map[string]string{"aria-label": Or(p.AriaLabel, "Breadcrumb")})
	return in
}

// Breadcrumb renders the trail; aria-current="page" on the current item.
templ Breadcrumb(p BreadcrumbProps) {
	@box.Box(breadcrumbNav(p)) {
		@box.Box(box.BoxProps{Tag: "ol", Class: "flex flex-wrap items-center gap-2"}) {
			for _, it := range p.Items {
				@box.Box(box.BoxProps{Tag: "li", Class: "inline-flex items-center"}) {
					if it.Href != "" && !it.Disabled {
						@link.Link(link.LinkProps{Href: it.Href, Class: breadcrumbItemClass(it), Attrs: breadcrumbCurrent(it)}) {
							{ it.Label }
						}
					} else {
						@text.Text(text.TextProps{Tag: "span", Class: breadcrumbItemClass(it), Attrs: breadcrumbCurrent(it)}) {
							{ it.Label }
						}
					}
				}
			}
		}
	}
}
