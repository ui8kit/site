/**
 * Breadcrumb shows the current page location within a hierarchy.
 * Renders nav with ordered links and aria-current on the active item.
 */

import { forwardRef, type ReactNode, type HTMLAttributes } from "react";
import { cn } from "../../utils";
import { isSetStr, orUndef } from "../../utils/expr";

export type BreadcrumbItem = {
  label?: string;
  href?: string;
  current?: boolean;
  disabled?: boolean;
};

/**
 * Breadcrumb renders the full navigation trail.
 */
export type BreadcrumbProps = Omit<
  HTMLAttributes<HTMLElement>,
  | "className" | "children" | "items"
> & {
  items?: BreadcrumbItem[];
  className?: string;
};

export const Breadcrumb = forwardRef<HTMLElement, BreadcrumbProps>(function Breadcrumb(
  { items, className, ...rest },
  ref
) {
  const cls = cn("text-sm", className);
  return (
    <nav ref={ref} className={cls || undefined} {...rest}>
      <ol className="flex flex-wrap items-center gap-2">
        {(items ?? []).map((item, index) => (
          <li key={index} className="inline-flex items-center">
            {(isSetStr(item.href) && !(item.disabled)) ? (
              <a
                href={orUndef(item.href)}
                className={cn(
                  "text-sm",
                  (
                    item.current ? "font-medium"
                    : item.disabled ? "text-muted-foreground"
                    : "text-primary underline-offset-4 hover:underline"
                  )
                )}
                aria-current={item.current ? "page" : undefined}
              >
                {item.label}
              </a>
            ) : (
              <span
                className={cn(
                  "text-sm",
                  (
                    item.current ? "font-medium"
                    : item.disabled ? "text-muted-foreground"
                    : "text-primary underline-offset-4 hover:underline"
                  )
                )}
                aria-current={item.current ? "page" : undefined}
              >
                {item.label}
              </span>
            )}
          </li>
        ))}
      </ol>
    </nav>
  );
});

export type BreadcrumbItem = {
  label?: string;
  href?: string;
  current?: boolean;
  disabled?: boolean;
};

 * Breadcrumb renders the full navigation trail.
 */

  JSX.HTMLAttributes<HTMLElement>,
  "class" | "children" | "items"
> & {
  items?: BreadcrumbItem[];
  class?: string;
};

export const Breadcrumb: Component<BreadcrumbProps> = (props) => {
  const [local, rest] = splitProps(props, ["items", "class"]);
  const cls = () => cn("text-sm", local.class);
  return (
    <nav class={cls() || undefined} {...rest}>
      <ol class="flex flex-wrap items-center gap-2">
        <For each={local.items ?? []}>

            <li class="inline-flex items-center">
              <Show when={(isSetStr(item.href) && !(item.disabled))} fallback={
                <span
                  class={cn(
                    "text-sm",

                      : item.disabled ? "text-muted-foreground"
                      : "text-primary underline-offset-4 hover:underline"
                    )
                  )}
                  aria-current={item.current ? "page" : undefined}
                >
                  {item.label}
                </span>
              }>
                <a
                  href={orUndef(item.href)}
                  class={cn(
                    "text-sm",
                    (
                      item.current ? "font-medium"

// Source: xgen/specs/ui/breadcrumb.json

/*
Breadcrumb shows the current page location within a hierarchy.
Renders nav with ordered links and aria-current on the active item.
*/

package breadcrumb

import (
	uiutils "github.com/codegen-x/ui8kit/packages/templ/ui/utils"
)

// BreadcrumbItem is one BreadcrumbItem entry.
type BreadcrumbItem struct {
	Label string
	Href string
	// marks the active page (aria-current=page)
	Current bool
	Disabled bool
}

// BreadcrumbProps configures Breadcrumb.
type BreadcrumbProps struct {
	// Class — adds Tailwind utilities
	Class string
	Items []BreadcrumbItem
	AriaLabel string
	// DataUI8Kit — opt-in behavior hook; empty emits nothing
	DataUI8Kit string
	Attrs templ.Attributes
}

// BreadcrumbClasses composes Breadcrumb classes without rendering —
// the server-composition escape hatch for custom root elements.
func BreadcrumbClasses(p BreadcrumbProps) string {
	return uiutils.Cn("text-sm", p.Class)
}

// Breadcrumb renders the Breadcrumb contract.
templ Breadcrumb(p BreadcrumbProps) {
	<nav
		class={ BreadcrumbClasses(p) }
		if uiutils.IsSetStr(p.AriaLabel) {
			aria-label={ p.AriaLabel }
		}
		if uiutils.IsSetStr(p.DataUI8Kit) {
			data-ui8kit={ p.DataUI8Kit }
		}
		{ p.Attrs... }
	>
		<ol
			class="flex flex-wrap items-center gap-2"
		>
			for _, item := range p.Items {
				<li
					class="inline-flex items-center"
				>
					if (uiutils.IsSetStr(item.Href) && !(item.Disabled)) {
						<a
							if uiutils.IsSetStr(item.Href) {
								href={ item.Href }
							}
							class={ uiutils.Cn("text-sm", uiutils.If(item.Current, "font-medium", uiutils.If(item.Disabled, "text-muted-foreground", "text-primary underline-offset-4 hover:underline"))) }
							if item.Current {
								aria-current={ "page" }
							}
						>
							{ item.Label }
						</a>
					} else {
						<span
							class={ uiutils.Cn("text-sm", uiutils.If(item.Current, "font-medium", uiutils.If(item.Disabled, "text-muted-foreground", "text-primary underline-offset-4 hover:underline"))) }
							if item.Current {
								aria-current={ "page" }
							}
						>
							{ item.Label }
						</span>
					}
				</li>
			}
		</ol>
	</nav>
}
