// Source: xgen/specs/ui/card.json
/**
 * Card groups related content in one bordered surface.
 * Card uses header, content, and footer slots.
 * For semantic root elements (section, article), use CardClasses on a manual wrapper.
 */

import { forwardRef, type ReactNode, type HTMLAttributes } from "react";
import { cn, variants, type VariantKey } from "../../utils";
import { Slot } from "../slot/slot";

/** Card: the classes of each value of each prop (shadcn/cva style). */
export const cardVariants = variants({
  base: "flex flex-col gap-8 bg-card p-6 text-card-foreground inset-ring inset-ring-border rounded",
  variants: {
    variant: {
      default: "",
      muted: "bg-muted",
      ghost: "bg-transparent",
      compact: "gap-4 p-4",
    },
  },
  defaultVariants: { variant: "default" },
});

export type CardVariant = VariantKey<typeof cardVariants, "variant">;

/**
 * Card renders the bordered surface wrapper.
 */
export type CardProps = Omit<
  HTMLAttributes<HTMLDivElement>,
  | "className" | "children" | "variant"
> & {
  variant?: CardVariant;
  className?: string;
  children?: ReactNode;
  asChild?: boolean;
};

export const Card = forwardRef<HTMLDivElement, CardProps>(function Card(
  { variant, className, children, asChild, ...rest },
  ref
) {
  const cls = cn(cardVariants({ variant }), className);
  if (asChild) {
    return (
      <Slot ref={ref} className={cls} {...rest}>
        {children}
      </Slot>
    );
  }
  return (
    <div ref={ref} className={cls || undefined} data-slot="card" {...rest}>
      {children}
    </div>
  );
});
Card.displayName = "Card";

/*
Card groups related content in one bordered surface.
Card uses header, content, and footer slots.
For semantic root elements (section, article), use CardClasses on a manual wrapper.
*/

package card

import (
	uiutils "github.com/codegen-x/ui8kit/packages/templ/ui/utils"
)

// CardProps configures Card.
type CardProps struct {
	// Class — adds Tailwind utilities
	Class string
	// Variant — surface: default, muted, ghost, compact
	Variant string
	Attrs templ.Attributes
}

// CardClasses composes Card classes without rendering —
// the server-composition escape hatch for custom root elements.
func CardClasses(p CardProps) string {
	return uiutils.Compose(CardVariants, map[string]string{
		"variant": p.Variant,
	}, p.Class)
}

// Card renders the Card contract.
templ Card(p CardProps) {
	<div
		class={ CardClasses(p) }
		data-slot="card"
		{ p.Attrs... }
	>
		{ children... }
	</div>
}
