/**
 * LinkPlain is a bare anchor: no styling beyond Class.
 */

import { forwardRef, type ReactNode, type AnchorHTMLAttributes } from "react";
import { cn } from "../../utils";
import { eqStr, isSetStr, orUndef } from "../../utils/expr";

/**
 * LinkPlain is a bare anchor: no styling beyond Class.
 */
export type LinkPlainProps = Omit<
  AnchorHTMLAttributes<HTMLAnchorElement>,
  | "className" | "children" | "target" | "rel" | "external" | "disabled"
> & {
  target?: string;
  rel?: string;
  external?: boolean;
  disabled?: boolean;
  className?: string;
  children?: ReactNode;
};

export const LinkPlain = forwardRef<HTMLAnchorElement, LinkPlainProps>(function LinkPlain(
  { target, rel, external, disabled, className, children, ...rest },
  ref
) {
  const cls = cn((disabled ? "pointer-events-none opacity-50" : ""), className);
  const targetValue = (isSetStr(target) ? target : (external ? "_blank" : ""));
  const relValue = (isSetStr(rel) ? rel : ((external || eqStr(targetValue, "_blank")) ? "noopener noreferrer" : ""));
  return (
    <a
      ref={ref}
      className={cls || undefined}
      target={orUndef(targetValue)}
      rel={orUndef(relValue)}
      aria-disabled={disabled ? "true" : undefined}
      tabIndex={disabled ? -1 : undefined}
      {...rest}
    >
      {children}
    </a>
  );
});
LinkPlain.displayName = "LinkPlain";
