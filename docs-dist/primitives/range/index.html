/**
 * Range renders a styled input[type=range] (single thumb slider). The value is controlled from outside: Value
 * in, an input event out (canary used a bindable prop).
 */

import { forwardRef, type ReactNode, type InputHTMLAttributes } from "react";
import { cn, variants } from "../../utils";
import { isSetNum, orUndef } from "../../utils/expr";

/** Range: the classes of each value of each prop (shadcn/cva style). */
export const rangeVariants = variants({
  base:
    "h-5 w-full cursor-pointer appearance-none bg-transparent accent-primary disabled:cursor-not-allowed " +
    "disabled:opacity-50 [&::-webkit-slider-runnable-track]:h-1 [&::-webkit-slider-runnable-track]:rounded " +
    "[&::-webkit-slider-runnable-track]:bg-muted [&::-webkit-slider-thumb]:-mt-1 [&::-webkit-slider-thumb]:size-4 " +
    "[&::-webkit-slider-thumb]:appearance-none [&::-webkit-slider-thumb]:rounded " +
    "[&::-webkit-slider-thumb]:bg-primary [&::-moz-range-track]:h-1 [&::-moz-range-track]:rounded " +
    "[&::-moz-range-track]:bg-muted [&::-moz-range-thumb]:size-4 [&::-moz-range-thumb]:rounded " +
    "[&::-moz-range-thumb]:border-0 [&::-moz-range-thumb]:bg-primary",
  variants: {},
  defaultVariants: {},
});

/**
 * Range renders input type=range.
 */
export type RangeProps = Omit<
  InputHTMLAttributes<HTMLInputElement>,
  | "className" | "children" | "min" | "max" | "step" | "disabled" | "aria-label"
> & {
  min?: number;
  max?: number;
  step?: number;
  disabled?: boolean;
  "aria-label"?: string;
  className?: string;
};

export const Range = forwardRef<HTMLInputElement, RangeProps>(function Range(
  { min, max, step, disabled, "aria-label": ariaLabel, className, ...rest },
  ref
) {
  const cls = cn(rangeVariants(), className);
  return (
    <input
      ref={ref}
      type="range"
      className={cls || undefined}
      min={(isSetNum(min) ? String(min ?? 0) : "0")}
      max={(isSetNum(max) ? String(max ?? 0) : "100")}
      step={(isSetNum(step) ? String(step ?? 0) : "1")}
      disabled={disabled || undefined}
      aria-label={orUndef(ariaLabel)}
      {...rest}
    />
  );
});
Range.displayName = "Range";
