/**

import { forwardRef, type ReactNode, type HTMLAttributes } from "react";
import { cn, variants, type VariantKey } from "../../utils";
import { eqStr } from "../../utils/expr";

/** Separator: the classes of each value of each prop (shadcn/cva style). */
export const separatorVariants = variants({
  base: "shrink-0 border-0 bg-border",
  variants: {
    orientation: {
      horizontal: "h-px w-full",
      vertical: "h-full w-px",
    },
    variant: {
      default: "",
      spaced: "my-6",
    },
  },
  defaultVariants: { orientation: "horizontal", variant: "default" },
});

export type SeparatorVariant = VariantKey<typeof separatorVariants, "variant">;
export type SeparatorOrientation = VariantKey<typeof separatorVariants, "orientation">;

/**
 * Separator renders hr with optional decorative semantics.
 */
export type SeparatorProps = Omit<
  HTMLAttributes<HTMLHRElement>,
  | "className" | "children" | "variant" | "orientation" | "decorative"
> & {
  variant?: SeparatorVariant;
  orientation?: SeparatorOrientation;
  decorative?: boolean;
  className?: string;
};

export const Separator = forwardRef<HTMLHRElement, SeparatorProps>(function Separator(
  { variant, orientation, decorative, className, ...rest },
  ref
) {
  const cls = cn(separatorVariants({ orientation, variant }), className);
  return (
    <hr
      ref={ref}
      className={cls || undefined}
      role={decorative ? "presentation" : undefined}
      aria-hidden={decorative ? "true" : undefined}
      aria-orientation={eqStr(orientation, "vertical") ? "vertical" : undefined}
      {...rest}
    />
  );
});
Separator.displayName = "Separator";
