 * Group renders a horizontal flex container.

export type GroupProps = Omit<
  HTMLAttributes<HTMLElement>,
  | "className" | "children" | "gap" | "align" | "justify" | "tag"
> & {

  align?: GroupAlign;
  justify?: GroupJustify;
  tag?: string;

  children?: ReactNode;
  asChild?: boolean;

export const Group = forwardRef<HTMLElement, GroupProps>(function Group(
  { gap, align, justify, tag, className, children, asChild, ...rest },
  ref
) {
  const cls = cn(groupVariants({ gap, align, justify }), className);
  const resolvedTagValue = resolveTag(tag, "div", TagGroup.Group);
  const Tag = resolvedTagValue as ElementType;

        {children}
      </Slot>
    );
  }
  return (
    <Tag ref={ref as Ref<HTMLElement>} className={cls || undefined} {...rest}>
      {children}
    </Tag>
  );
});

/*
Group arranges children in a horizontal flex row.
Supports fieldset for related form controls.
*/

package group

import (
	uiutils "github.com/codegen-x/ui8kit/packages/templ/ui/utils"
)

// GroupProps configures Group.
type GroupProps struct {
	// Gap — sets gap: a Tailwind spacing step (0, 2, 4, 6, 8, 10, 12 = gap-N) or auto
	Gap string
	// Align — sets align
	Align string
	// Justify — sets justify
	Justify string
	// Class — adds Tailwind utilities
	Class string
	// Tag — div, fieldset, or dl. Use fieldset for related form controls.
	Tag string
	Attrs templ.Attributes
}

// GroupClasses composes Group classes without rendering —
// the server-composition escape hatch for custom root elements.
func GroupClasses(p GroupProps) string {
	return uiutils.Compose(GroupVariants, map[string]string{
		"gap": p.Gap,
		"align": p.Align,
		"justify": p.Justify,
	}, p.Class)
}

// Group renders the Group contract.
templ Group(p GroupProps) {
	switch uiutils.ResolveTag(p.Tag, "div", uiutils.TagGroupGroup) {
		case "fieldset":
			<fieldset
				class={ GroupClasses(p) }
				{ p.Attrs... }
			>
				{ children... }
			</fieldset>
		case "dl":
			<dl
				class={ GroupClasses(p) }
				{ p.Attrs... }
			>
				{ children... }
			</dl>
		default:
			<div
				class={ GroupClasses(p) }
				{ p.Attrs... }
			>
				{ children... }
			</div>
	}
}
