/**
 * Select is a native dropdown built from Option items.
 */

import { forwardRef, type ReactNode, type SelectHTMLAttributes } from "react";
import { cn, variants, type VariantKey } from "../../utils";
import { eqStr } from "../../utils/expr";

/** Select: the classes of each value of each prop (shadcn/cva style). */
export const selectVariants = variants({
  base:
    "w-full min-w-0 bg-background text-foreground inset-ring inset-ring-input outline-none " +
    "placeholder:text-muted-foreground disabled:cursor-not-allowed disabled:opacity-50 " +
    "focus-visible:inset-ring-ring rounded",
  variants: {
    size: {
      sm: "h-8 px-3 text-sm",
      default: "h-10 px-3 text-sm",
      lg: "h-12 px-4 text-base",
    },
    variant: {
      default: "",
      ghost: "bg-muted inset-ring-transparent",
    },
  },
  defaultVariants: { size: "default", variant: "default" },
});

export type SelectOptionItem = {
  value?: string;
  label?: string;
};

export type SelectVariant = VariantKey<typeof selectVariants, "variant">;
export type SelectSize = VariantKey<typeof selectVariants, "size">;

/**
 * Select renders option children from the Options list plus manual children.
 */
export type SelectProps = Omit<
  SelectHTMLAttributes<HTMLSelectElement>,
  | "className" | "children" | "variant" | "size" | "value" | "disabled" | "required" | "options"
> & {
  variant?: SelectVariant;
  size?: SelectSize;
  value?: string;
  disabled?: boolean;
  required?: boolean;
  options?: SelectOptionItem[];
  className?: string;
  children?: ReactNode;
};

export const Select = forwardRef<HTMLSelectElement, SelectProps>(function Select(
  { variant, size, value, disabled, required, options, className, children, ...rest },
  ref
) {
  const cls = cn(selectVariants({ size, variant }), className);
  return (
    <select
      ref={ref}
      className={cls || undefined}
      disabled={disabled || undefined}
      required={required || undefined}
      {...rest}
    >
      {(options ?? []).map((item, index) => (
        <option key={index} value={(item.value) ?? ""} selected={eqStr(item.value, value) || undefined}>
          {item.label}
        </option>
      ))}
      {children}
    </select>
  );
});
Select.displayName = "Select";

Select is a native dropdown built from Option items.

package selectfield

import (
	uiutils "github.com/codegen-x/ui8kit/packages/templ/ui/utils"
)

// SelectOptionItem is one SelectOptionItem entry.
type SelectOptionItem struct {
	Value string
	Label string
}

// SelectProps configures Select.
type SelectProps struct {
	// Variant — sets appearance preset
	Variant string
	// Size — sets density preset
	Size string
	// Class — adds Tailwind utilities
	Class string
	Name string
	// Value — currently selected option value
	Value string
	Disabled bool
	Required bool
	Options []SelectOptionItem
	// ID — element id
	ID string
	// Role — ARIA role override
	Role string
	// TabIndex — tab order override
	TabIndex string
	// AriaLabel — accessible name for unlabeled controls
	AriaLabel string
	Attrs templ.Attributes
}

// SelectClasses composes Select classes without rendering —
// the server-composition escape hatch for custom root elements.
func SelectClasses(p SelectProps) string {
	return uiutils.Compose(SelectVariants, map[string]string{
		"size": p.Size,
		"variant": p.Variant,
	}, p.Class)
}

// Select renders the Select contract.
templ Select(p SelectProps) {
	<select
		if uiutils.IsSetStr(p.ID) {
			id={ p.ID }
		}
		if uiutils.IsSetStr(p.Name) {
			name={ p.Name }
		}
		class={ SelectClasses(p) }
		disabled?={ p.Disabled }
		required?={ p.Required }
		if uiutils.IsSetStr(p.Role) {
			role={ p.Role }
		}
		if uiutils.IsSetStr(p.TabIndex) {
			tabindex={ p.TabIndex }
		}
		if uiutils.IsSetStr(p.AriaLabel) {
			aria-label={ p.AriaLabel }
		}
		{ p.Attrs... }
	>
		for _, item := range p.Options {
			<option
				value={ item.Value }
				selected?={ item.Value == p.Value }
			>
				{ item.Label }
			</option>
		}
		{ children... }
	</select>
}

// SelectOptionProps configures SelectOption.
type SelectOptionProps struct {
	Value string
	Label string
	Selected bool
	Disabled bool
	Attrs templ.Attributes
}

templ SelectOption(p SelectOptionProps) {
	<option
		if uiutils.IsSetStr(p.Value) {
			value={ p.Value }
		}
		if uiutils.IsSetStr(p.Label) {
			label={ p.Label }
		}
		selected?={ p.Selected }
		disabled?={ p.Disabled }
		{ p.Attrs... }
	>
		{ children... }

}
