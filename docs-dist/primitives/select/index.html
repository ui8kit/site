/**
 * Select is a native dropdown built from Option items.
 */

import { forwardRef, type ReactNode, type SelectHTMLAttributes } from "react";
import { cn, variants, type VariantKey } from "../../utils";
import { eqStr } from "../../utils/expr";

/** Select: the classes of each value of each prop (shadcn/cva style). */
export const selectVariants = variants({
  base:
    "w-full min-w-0 bg-background text-foreground inset-ring inset-ring-input outline-none " +
    "placeholder:text-muted-foreground disabled:cursor-not-allowed disabled:opacity-50 " +
    "focus-visible:inset-ring-ring rounded",
  variants: {
    size: {
      sm: "h-8 px-3 text-sm",
      default: "h-10 px-3 text-sm",
      lg: "h-12 px-4 text-base",
    },
    variant: {
      default: "",
      ghost: "bg-muted inset-ring-transparent",
    },
  },
  defaultVariants: { size: "default", variant: "default" },
});

export type SelectOptionItem = {
  value?: string;
  label?: string;
};

export type SelectVariant = VariantKey<typeof selectVariants, "variant">;
export type SelectSize = VariantKey<typeof selectVariants, "size">;

/**
 * Select renders option children from the Options list plus manual children.
 */
export type SelectProps = Omit<
  SelectHTMLAttributes<HTMLSelectElement>,
  | "className" | "children" | "variant" | "size" | "value" | "disabled" | "required" | "options"
> & {
  variant?: SelectVariant;
  size?: SelectSize;
  value?: string;
  disabled?: boolean;
  required?: boolean;
  options?: SelectOptionItem[];
  className?: string;
  children?: ReactNode;
};

export const Select = forwardRef<HTMLSelectElement, SelectProps>(function Select(
  { variant, size, value, disabled, required, options, className, children, ...rest },
  ref
) {
  const cls = cn(selectVariants({ size, variant }), className);
  return (
    <select
      ref={ref}
      className={cls || undefined}
      disabled={disabled || undefined}
      required={required || undefined}
      {...rest}
    >
      {(options ?? []).map((item, index) => (
        <option key={index} value={(item.value) ?? ""} selected={eqStr(item.value, value) || undefined}>
          {item.label}
        </option>
      ))}
      {children}
    </select>
  );
});
Select.displayName = "Select";
