/**
 * Box is a generic layout wrapper for internal sections.
 * Not for top-level landmarks — use Block for page regions.
 */

import { forwardRef, type ReactNode, type HTMLAttributes, type ElementType, type Ref } from "react";
import { cn } from "../../utils";
import { resolveTag, TagGroup } from "../../utils/tags";
import { Slot } from "../slot/slot";

/**
 * Box renders a div layout wrapper.
 */
export type BoxProps = Omit<HTMLAttributes<HTMLElement>, "className" | "children" | "tag"> & {
  tag?: string;
  className?: string;
  children?: ReactNode;
  asChild?: boolean;
};

export const Box = forwardRef<HTMLElement, BoxProps>(function Box(
  { tag, className, children, asChild, ...rest },
  ref
) {
  const cls = cn(className);
  const resolvedTagValue = resolveTag(tag, "div", TagGroup.BoxAllowed);
  const Tag = resolvedTagValue as ElementType;
  if (asChild) {
    return (
      <Slot ref={ref} className={cls} {...rest}>
        {children}
      </Slot>
    );
  }
  return (
    <Tag ref={ref as Ref<HTMLElement>} className={cls || undefined} {...rest}>
      {children}
    </Tag>
  );
});
Box.displayName = "Box";

/**
 * Box is a generic layout wrapper for internal sections.
 * Not for top-level landmarks — use Block for page regions.

import { splitProps, type Component, type JSX } from "solid-js";
import { Dynamic } from "solid-js/web";
import { cn } from "../../utils";
import { resolveTag, TagGroup } from "../../utils/tags";

/**
 * Box renders a div layout wrapper.
 */
export type BoxProps = Omit<JSX.HTMLAttributes<HTMLElement>, "class" | "children" | "tag"> & {
  tag?: string;
  class?: string;
  children?: JSX.Element;
};

export const Box: Component<BoxProps> = (props) => {
  const [local, rest] = splitProps(props, ["tag", "class", "children"]);
  const cls = () => cn(local.class);
  const resolvedTagValue = () => resolveTag(local.tag, "div", TagGroup.BoxAllowed);
  return (
    <Dynamic component={resolvedTagValue()} class={cls() || undefined} {...(rest as object)}>
      {local.children}
    </Dynamic>
  );
};

/*
Box is a generic layout wrapper for internal sections.
Not for top-level landmarks — use Block for page regions.
*/

package box

import (
	uiutils "github.com/codegen-x/ui8kit/packages/templ/ui/utils"
)

// BoxProps configures Box.
type BoxProps struct {
	// Class — adds Tailwind utilities
	Class string
	// Tag — div only. Use Block for landmarks (aside, section, header, nav).
	Tag string
	Attrs templ.Attributes
}

// BoxClasses composes Box classes without rendering —
// the server-composition escape hatch for custom root elements.
func BoxClasses(p BoxProps) string {
	return uiutils.Compose(BoxVariants, map[string]string{}, p.Class)
}

// Box renders the Box contract.
templ Box(p BoxProps) {
	switch uiutils.ResolveTag(p.Tag, "div", uiutils.TagGroupBoxAllowed) {
		default:
			<div
				class={ BoxClasses(p) }
				{ p.Attrs... }
			>
				{ children... }
			</div>
	}
}
