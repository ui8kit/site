/**
 * ListItem: one part of List (one file per part, as in the Svelte package).
 */

import { forwardRef, type ReactNode, type HTMLAttributes, type ElementType, type Ref } from "react";
import { cn } from "../../utils";
import { eqStr, isSetNum } from "../../utils/expr";
import { resolveTag, TagGroup } from "../../utils/tags";

/**
 * ListItem renders one li with optional value attribute.
 */
export type ListItemProps = Omit<
  HTMLAttributes<HTMLElement>,
  | "className" | "children" | "tag" | "value"
> & {
  tag?: string;
  value?: number;
  className?: string;
  children?: ReactNode;
};

export const ListItem = forwardRef<HTMLElement, ListItemProps>(function ListItem(
  { tag, value, className, children, ...rest },
  ref
) {
  const cls = cn(className);
  const resolvedTagValue = resolveTag(tag, "li", TagGroup.ListItem);
  const Tag = resolvedTagValue as ElementType;
  return (
    <Tag
      ref={ref as Ref<HTMLElement>}
      className={cls || undefined}
      value={(isSetNum(value) && eqStr(resolvedTagValue, "li")) ? String(value ?? 0) : undefined}
      {...rest}
    >
      {children}
    </Tag>
  );
});
ListItem.displayName = "ListItem";
