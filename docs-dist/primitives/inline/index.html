/**
 * Inline renders inline copy as a span. Use Text for paragraph blocks and
 * Title (as 1–6) for headings.
 */

import { forwardRef, type ReactNode, type HTMLAttributes } from "react";
import { cn } from "../../utils";
import { Slot } from "../slot/slot";

/**
 * Inline renders a span element with inline copy classes.
 */
export type InlineProps = Omit<HTMLAttributes<HTMLSpanElement>, "className" | "children"> & {
  className?: string;
  children?: ReactNode;
  asChild?: boolean;
};

export const Inline = forwardRef<HTMLSpanElement, InlineProps>(function Inline(
  { className, children, asChild, ...rest },
  ref
) {
  const cls = cn(className);
  if (asChild) {
    return (
      <Slot ref={ref} className={cls} {...rest}>
        {children}
      </Slot>
    );
  }
  return (
    <span ref={ref} className={cls || undefined} {...rest}>
      {children}
    </span>
  );
});
Inline.displayName = "Inline";
