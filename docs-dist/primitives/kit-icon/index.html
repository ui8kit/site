/**
 * Icon is a sized frame for a consumer-supplied glyph (svg, sprite, font).
 * No registry: the glyph is the children/slot. Decorative unless Label is set.
 */

import { forwardRef, type ReactNode, type HTMLAttributes } from "react";
import { iconClasses, type IconSize } from "./icon.shared";
import { isSetStr, safeRest } from "../../utils/expr";

/**
 * Icon renders span[data-slot=icon]; Label switches to role=img + aria-label.
 */
export type IconProps = Omit<HTMLAttributes<HTMLSpanElement>, "className" | "children" | "size" | "label"> & {
  size?: IconSize;
  label?: string;
  className?: string;
  children?: ReactNode;
};

export const Icon = forwardRef<HTMLSpanElement, IconProps>(function Icon(
  { size, label, className, children, ...rawRest },
  ref
) {
  const rest = safeRest(rawRest);
  const cls = iconClasses({ size, className });
  return (
    <span ref={ref} data-slot="icon" className={cls || undefined} aria-hidden={!(isSetStr(label)) ? "true" : undefined} role={isSetStr(label) ? "img" : undefined} aria-label={isSetStr(label) ? label : undefined} {...rest}>
      {children}
    </span>
  );
});

/*
Icon is a sized frame for a consumer-supplied glyph (svg, sprite, font).
No registry: the glyph is the children/slot. Decorative unless Label is set.
*/

package icon

import (
	uiutils "github.com/ui8kit/codegen/generated/templ/utils"
)

// IconProps configures Icon.
type IconProps struct {
	// Size — size token: xs, sm, md, lg, xl (default sm)
	Size string
	// Class — token class names supplied by the consumer
	Class string
	// Label — accessible name; omitted = decorative (aria-hidden)
	Label string
	Attrs templ.Attributes
}

// IconClasses composes Icon classes without rendering —
// the server-composition escape hatch for custom root elements (no asChild in templ).
func IconClasses(p IconProps) string {
	return uiutils.Compose(IconVariants, map[string]string{
		"size": p.Size,
	}, p.Class)
}

// Icon renders the Icon contract.
templ Icon(p IconProps) {
	<span
		if !uiutils.HasAttr(p.Attrs, "data-slot") {
			data-slot="icon"
		}
		if IconClasses(p) != "" {
			class={ IconClasses(p) }
		}
		if !uiutils.HasAttr(p.Attrs, "aria-hidden") {
			if !uiutils.IsSetStr(p.Label) {
				aria-hidden={ "true" }
			}
		}
		if !uiutils.HasAttr(p.Attrs, "role") {
			if uiutils.IsSetStr(p.Label) {
				role={ "img" }
			}
		}
		if !uiutils.HasAttr(p.Attrs, "aria-label") {
			if uiutils.IsSetStr(p.Label) {
				aria-label={ p.Label }
			}
		}
		{ uiutils.RestAttrs(p.Attrs, false)... }
	>
		{ children... }
	</span>
}
