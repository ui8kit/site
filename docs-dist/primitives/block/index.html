    // Code generated by xgen. DO NOT EDIT.
// Source: xgen/specs/ui/block.json
/**
 * Block is a top-level page landmark (main, header, section, nav, …).
 * Use for document structure; do not nest Block inside Block.
 */

import { forwardRef, type ReactNode, type HTMLAttributes, type ElementType, type Ref } from "react";
import { cn } from "../../utils";
import { resolveTag, TagGroup } from "../../utils/tags";
import { Slot } from "../slot/slot";

/**
 * Block renders a resolved landmark element with children.
 */
export type BlockProps = Omit<HTMLAttributes<HTMLElement>, "className" | "children" | "tag"> & {
  tag?: string;
  className?: string;
  children?: ReactNode;
  asChild?: boolean;
};

export const Block = forwardRef<HTMLElement, BlockProps>(function Block(
  { tag, className, children, asChild, ...rest },
  ref
) {
  const cls = cn(className);
  const resolvedTagValue = resolveTag(tag, "div", TagGroup.Layout);
  const Tag = resolvedTagValue as ElementType;
  if (asChild) {
    return (
      <Slot ref={ref} className={cls} {...rest}>
        {children}
      </Slot>
    );
  }
  return (
    <Tag ref={ref as Ref<HTMLElement>} className={cls || undefined} {...rest}>
      {children}
    </Tag>
  );
});
Block.displayName = "Block";
