/**
 * Stack arranges children vertically in a flex column.
 * Default layout primitive for spacing sections and form fields.
 */

import { forwardRef, type ReactNode, type HTMLAttributes, type ElementType, type Ref } from "react";
import { cn, variants, type VariantKey } from "../../utils";
import { resolveTag, TagGroup } from "../../utils/tags";
import { Slot } from "../slot/slot";

/** Stack: the classes of each value of each prop (shadcn/cva style). */
export const stackVariants = variants({
  base: "flex flex-col",
  variants: {
    gap: {
      "0": "gap-0",
      "2": "gap-2",
      "4": "gap-4",
      "6": "gap-6",
      "8": "gap-8",
      "10": "gap-10",
      "12": "gap-12",
      auto: "",
    },
    align: {
      start: "items-start",
      center: "items-center",
      end: "items-end",
      stretch: "items-stretch",
      baseline: "items-baseline",
    },
    justify: {
      start: "justify-start",
      center: "justify-center",
      end: "justify-end",
      between: "justify-between",
    },
  },
  defaultVariants: { gap: "auto", align: "start", justify: "start" },
});

export type StackGap = VariantKey<typeof stackVariants, "gap">;
export type StackAlign = VariantKey<typeof stackVariants, "align">;
export type StackJustify = VariantKey<typeof stackVariants, "justify">;

/**
 * Stack renders a vertical flex container.
 */
export type StackProps = Omit<
  HTMLAttributes<HTMLElement>,
  | "className" | "children" | "gap" | "align" | "justify" | "tag"
> & {
  gap?: StackGap;
  align?: StackAlign;
  justify?: StackJustify;
  tag?: string;
  className?: string;
  children?: ReactNode;
  asChild?: boolean;
};

export const Stack = forwardRef<HTMLElement, StackProps>(function Stack(
  { gap, align, justify, tag, className, children, asChild, ...rest },
  ref
) {
  const cls = cn(stackVariants({ gap, align, justify }), className);
  const resolvedTagValue = resolveTag(tag, "div", TagGroup.Stack);
  const Tag = resolvedTagValue as ElementType;
  if (asChild) {
    return (
      <Slot ref={ref} className={cls} {...rest}>
        {children}
      </Slot>
    );
  }
  return (
    <Tag ref={ref as Ref<HTMLElement>} className={cls || undefined} {...rest}>
      {children}
    </Tag>
  );
});
Stack.displayName = "Stack";
