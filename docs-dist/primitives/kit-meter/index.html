/**
 * Meter is a native gauge for a known range (value/min/max).
 */

import { forwardRef, type ReactNode, type MeterHTMLAttributes } from "react";
import { meterClasses } from "./meter.shared";
import { safeRest } from "../../utils/expr";

/**
 * Meter renders a native meter; children are the fallback text.
 */
export type MeterProps = Omit<MeterHTMLAttributes<HTMLMeterElement>, "className" | "children"> & {
  className?: string;
  children?: ReactNode;
};

export const Meter = forwardRef<HTMLMeterElement, MeterProps>(function Meter(
  { className, children, ...rawRest },
  ref
) {
  const rest = safeRest(rawRest, { blank: ["min", "max", "low", "high", "optimum", "aria-label"] });
  const cls = meterClasses({ className });
  return (
    <meter ref={ref} className={cls || undefined} {...rest}>
      {children}
    </meter>
  );
});
Meter.displayName = "Meter";

import { splitProps, type JSX, type Component } from "solid-js";
import { meterClasses } from "./meter.shared";
import { safeRest } from "../../utils/expr";

/**
 * Meter renders a native meter; children are the fallback text.
 */
export type MeterProps = Omit<JSX.MeterHTMLAttributes<HTMLMeterElement>, "class" | "className" | "children"> & {
  class?: string;
  className?: string;
  children?: JSX.Element;
};

export const Meter: Component<MeterProps> = (props) => {
  const [local, rest] = splitProps(props, ["class", "className", "children"]);
  const cls = () => meterClasses({ className: local.class ?? local.className });

      {local.children}
    </meter>
  );
};
