// Source: xgen/specs/ui/label.json
/**
 * Label associates visible text with a form control via the for attribute.
 */

import { forwardRef, type ReactNode, type LabelHTMLAttributes } from "react";
import { cn, variants } from "../../utils";

/** Label: the classes of each value of each prop (shadcn/cva style). */
export const labelVariants = variants({
  base: "text-xs font-bold text-foreground",
  variants: {},
  defaultVariants: {},
});

/**
 * Label renders accessible text linked to a control.
 */
export type LabelProps = Omit<LabelHTMLAttributes<HTMLLabelElement>, "className" | "children"> & {
  className?: string;
  children?: ReactNode;
};

export const Label = forwardRef<HTMLLabelElement, LabelProps>(function Label(
  { className, children, ...rest },
  ref
) {
  const cls = cn(labelVariants(), className);
  return (
    <label ref={ref} className={cls || undefined} {...rest}>
      {children}
    </label>
  );
});
Label.displayName = "Label";
