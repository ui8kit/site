/**
 * Link renders a semantic anchor for navigation and prose.
 * Use Button for actions; use Link for destinations.
 */

import { forwardRef, type ReactNode, type AnchorHTMLAttributes } from "react";
import { cn, variants, type VariantKey } from "../../utils";
import { eqStr, isSetStr, orUndef } from "../../utils/expr";

/** Link: the classes of each value of each prop (shadcn/cva style). */
export const linkVariants = variants({
  base: "inline-flex items-center underline-offset-4",
  variants: {
    size: {
      sm: "gap-2 text-sm font-medium",
      default: "gap-2 text-base font-semibold",
      md: "gap-2 text-xl font-semibold",
      lg: "gap-2 text-2xl font-semibold",
    },
    variant: {
      block: "flex w-full text-foreground",
      default: "text-foreground hover:underline",
      external: "text-foreground hover:underline",
      muted: "text-muted-foreground hover:text-foreground",
      nav: "text-foreground",
    },
  },
  defaultVariants: { size: "default", variant: "default" },
});

export type LinkVariant = VariantKey<typeof linkVariants, "variant">;
export type LinkSize = VariantKey<typeof linkVariants, "size">;

/**
 * Link renders a semantic anchor element.
 */
export type LinkProps = Omit<
  AnchorHTMLAttributes<HTMLAnchorElement>,
  | "className" | "children" | "variant" | "size" | "target" | "rel" | "external" | "disabled"
> & {
  variant?: LinkVariant;
  size?: LinkSize;
  target?: string;
  rel?: string;
  external?: boolean;
  disabled?: boolean;
  className?: string;
  children?: ReactNode;
};

export const Link = forwardRef<HTMLAnchorElement, LinkProps>(function Link(
  { variant, size, target, rel, external, disabled, className, children, ...rest },
  ref
) {
  const cls = cn(linkVariants({ size, variant }), (disabled ? "pointer-events-none opacity-50" : ""), className);
  const targetValue = (isSetStr(target) ? target : (external ? "_blank" : ""));
  const relValue = (isSetStr(rel) ? rel : ((external || eqStr(targetValue, "_blank")) ? "noopener noreferrer" : ""));
  return (
    <a
      ref={ref}
      className={cls || undefined}
      target={orUndef(targetValue)}
      rel={orUndef(relValue)}
      aria-disabled={disabled ? "true" : undefined}
      tabIndex={disabled ? -1 : undefined}
      {...rest}
    >
      {children}
    </a>
  );
});
Link.displayName = "Link";
