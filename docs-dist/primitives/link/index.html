/**
 * Link renders a semantic anchor for navigation and prose.
 * Use Button for actions; use Link for destinations.
 */

import { forwardRef, type ReactNode, type AnchorHTMLAttributes } from "react";
import { cn, variants, type VariantKey } from "../../utils";
import { eqStr, isSetStr, orUndef } from "../../utils/expr";

/** Link: the classes of each value of each prop (shadcn/cva style). */
export const linkVariants = variants({
  base: "inline-flex items-center underline-offset-4",
  variants: {
    size: {
      sm: "gap-2 text-sm font-medium",
      default: "gap-2 text-base font-semibold",
      md: "gap-2 text-xl font-semibold",
      lg: "gap-2 text-2xl font-semibold",
    },
    variant: {
      block: "flex w-full text-foreground",
      default: "text-foreground hover:underline",
      external: "text-foreground hover:underline",
      muted: "text-muted-foreground hover:text-foreground",
      nav: "text-foreground",
    },
  },
  defaultVariants: { size: "default", variant: "default" },
});

export type LinkVariant = VariantKey<typeof linkVariants, "variant">;
export type LinkSize = VariantKey<typeof linkVariants, "size">;

/**
 * Link renders a semantic anchor element.
 */
export type LinkProps = Omit<
  AnchorHTMLAttributes<HTMLAnchorElement>,
  | "className" | "children" | "variant" | "size" | "target" | "rel" | "external" | "disabled"
> & {
  variant?: LinkVariant;
  size?: LinkSize;
  target?: string;
  rel?: string;
  external?: boolean;
  disabled?: boolean;
  className?: string;
  children?: ReactNode;
};

export const Link = forwardRef<HTMLAnchorElement, LinkProps>(function Link(
  { variant, size, target, rel, external, disabled, className, children, ...rest },
  ref
) {
  const cls = cn(linkVariants({ size, variant }), (disabled ? "pointer-events-none opacity-50" : ""), className);
  const targetValue = (isSetStr(target) ? target : (external ? "_blank" : ""));
  const relValue = (isSetStr(rel) ? rel : ((external || eqStr(targetValue, "_blank")) ? "noopener noreferrer" : ""));
  return (
    <a
      ref={ref}
      className={cls || undefined}
      target={orUndef(targetValue)}
      rel={orUndef(relValue)}
      aria-disabled={disabled ? "true" : undefined}
      tabIndex={disabled ? -1 : undefined}
      {...rest}
    >
      {children}
    </a>
  );
});
Link.displayName = "Link";

/*
Link renders a semantic anchor for navigation and prose.
Use Button for actions; use Link for destinations.
*/

package link

import (
	"strconv"

	uiutils "github.com/codegen-x/ui8kit/packages/templ/ui/utils"
)

// LinkProps configures Link.
type LinkProps struct {
	// Variant — sets appearance preset
	Variant string
	// Size — sets density preset
	Size string
	// Class — adds Tailwind utilities
	Class string
	Href string
	// Target — explicit target; External implies _blank
	Target string
	// Rel — explicit rel; external links default to noopener noreferrer
	Rel string
	Download string
	// External — opens in a new tab with safe rel defaults
	External bool
	// Disabled — visually and semantically disables the link
	Disabled bool
	AriaCurrent string
	AriaLabel string
	ID string
	Role string
	Attrs templ.Attributes
}

// LinkClasses composes Link classes without rendering —
// the server-composition escape hatch for custom root elements.
func LinkClasses(p LinkProps) string {
	return uiutils.Compose(LinkVariants, map[string]string{
		"size": p.Size,
		"variant": p.Variant,
	}, uiutils.If(p.Disabled, "pointer-events-none opacity-50", ""), p.Class)
}

func linkTarget(p LinkProps) string {
	return uiutils.If(uiutils.IsSetStr(p.Target), p.Target, uiutils.If(p.External, "_blank", ""))
}

func linkRel(p LinkProps) string {
	return uiutils.If(uiutils.IsSetStr(p.Rel), p.Rel, uiutils.If((p.External || linkTarget(p) == "_blank"), "noopener noreferrer", ""))
}

// Link renders the Link contract.
templ Link(p LinkProps) {
	<a
		if uiutils.IsSetStr(p.Href) {
			href={ p.Href }
		}
		class={ LinkClasses(p) }
		if uiutils.IsSetStr(p.ID) {
			id={ p.ID }
		}
		if uiutils.IsSetStr(linkTarget(p)) {
			target={ linkTarget(p) }
		}
		if uiutils.IsSetStr(linkRel(p)) {
			rel={ linkRel(p) }
		}
		if uiutils.IsSetStr(p.Download) {
			download={ p.Download }
		}
		if uiutils.IsSetStr(p.Role) {
			role={ p.Role }
		}
		if uiutils.IsSetStr(p.AriaLabel) {
			aria-label={ p.AriaLabel }
		}
		if uiutils.IsSetStr(p.AriaCurrent) {
			aria-current={ p.AriaCurrent }
		}
		if p.Disabled {
			aria-disabled={ "true" }
		}
		if p.Disabled {
			tabindex={ strconv.Itoa(-1) }
		}
		{ p.Attrs... }
	>
		{ children... }
	</a>
}
