import { forwardRef, type InputHTMLAttributes } from "react";
import { inputClasses } from "./input.shared";
import { defaultIfEmpty, safeRest } from "../../utils/expr";

/**
 * Input renders a native input element (default type text).
 */
export type InputProps = Omit<InputHTMLAttributes<HTMLInputElement>, "className" | "children" | "type" | "required" | "disabled" | "readOnly"> & {
  type?: "text" | "email" | "password" | "search" | "tel" | "url" | "number" | "range" | "date" | "datetime-local" | "month" | "week" | "time" | "color" | "file" | "hidden";
  required?: boolean;
  disabled?: boolean;
  readOnly?: boolean;
  className?: string;
};

export const Input = forwardRef<HTMLInputElement, InputProps>(function Input(
  { type, required, disabled, readOnly, className, ...rawRest },
  ref
) {
  const rest = safeRest(rawRest, { blank: ["id", "name", "placeholder", "min", "max", "step", "aria-invalid", "aria-describedby"] });
  const cls = inputClasses({ className });
  return (
    <input ref={ref} type={defaultIfEmpty(type, "text")} className={cls || undefined} required={required} disabled={disabled} readOnly={readOnly} {...rest} />
  );
});
Input.displayName = "Input";

  className?: string;
};
