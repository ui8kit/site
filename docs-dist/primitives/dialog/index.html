 * Dialog renders a native dialog root with an optional behavior hook.
 * The data-ui8kit hook defaults off; the registry ships no client JS.

import { forwardRef, type ReactNode, type DialogHTMLAttributes } from "react";
import { cn, variants, type VariantKey } from "../../utils";

/** Dialog: the classes of each value of each prop (shadcn/cva style). */
export const dialogVariants = variants({
  base:
    "m-auto flex flex-col bg-background text-foreground inset-ring inset-ring-border shadow-lg " +
    "backdrop:bg-foreground/20 rounded",
  variants: {
    size: {
      default: "w-70",
      full: "h-full w-full",
      lg: "w-100",
      xl: "w-140",
    },
    variant: {
      card: "bg-card text-card-foreground",
      default: "",
      sheet: "min-h-screen rounded-none",
    },
  },
  defaultVariants: { size: "default", variant: "default" },
});

export type DialogVariant = VariantKey<typeof dialogVariants, "variant">;
export type DialogSize = VariantKey<typeof dialogVariants, "size">;

/**
 * Dialog renders a native dialog element.
 */
export type DialogProps = Omit<
  DialogHTMLAttributes<HTMLDialogElement>,
  | "className" | "children" | "variant" | "size" | "open"
> & {
  variant?: DialogVariant;
  size?: DialogSize;
  open?: boolean;
  className?: string;
  children?: ReactNode;
};

export const Dialog = forwardRef<HTMLDialogElement, DialogProps>(function Dialog(
  { variant, size, open, className, children, ...rest },
  ref
) {
  const cls = cn(dialogVariants({ size, variant }), className);
  return (
    <dialog ref={ref} className={cls || undefined} open={open || undefined} {...rest}>
      {children}
    </dialog>
  );
});
Dialog.displayName = "Dialog";
