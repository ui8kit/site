/**
 * Badge shows a short status label.
 * Badge uses variant and size presets.
 */

import { forwardRef, type ReactNode, type HTMLAttributes } from "react";
import { cn, variants, type VariantKey } from "../../utils";

/** Badge: the classes of each value of each prop (shadcn/cva style). */
export const badgeVariants = variants({
  base: "inline-flex shrink-0 items-center justify-center whitespace-nowrap font-medium inset-ring rounded-full",
  variants: {
    size: {
      sm: "h-5 px-2 text-xs",
      default: "h-6 px-3 text-xs",
      lg: "h-8 px-4 text-sm",
    },
    variant: {
      default: "bg-primary text-primary-foreground inset-ring-primary",
      secondary: "bg-muted text-foreground inset-ring-transparent",
      outline: "bg-transparent text-foreground inset-ring-border",
      destructive:
        "bg-destructive text-white inset-ring-transparent focus-visible:ring-destructive/20 dark:bg-destructive/60 " +
        "dark:focus-visible:ring-destructive/40",
    },
  },
  defaultVariants: { size: "default", variant: "default" },
});

export type BadgeVariant = VariantKey<typeof badgeVariants, "variant">;
export type BadgeSize = VariantKey<typeof badgeVariants, "size">;

/**
 * Badge renders label text in a div wrapper.
 */
export type BadgeProps = Omit<
  HTMLAttributes<HTMLDivElement>,
  | "className" | "children" | "variant" | "size"
> & {
  variant?: BadgeVariant;
  size?: BadgeSize;
  className?: string;
  children?: ReactNode;
};

export const Badge = forwardRef<HTMLDivElement, BadgeProps>(function Badge(
  { variant, size, className, children, ...rest },
  ref
) {
  const cls = cn(badgeVariants({ size, variant }), className);
  return (
    <div ref={ref} className={cls || undefined} data-slot="badge" {...rest}>
      {children}
    </div>
  );
});
Badge.displayName = "Badge";
