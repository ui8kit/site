/**
 * Skeleton renders a pulsing placeholder box (loading state). It is decorative: aria-hidden; the loading state
 * is announced by the container (aria-busy).
 */

import { forwardRef, type ReactNode, type HTMLAttributes } from "react";
import { cn, variants, type VariantKey } from "../../utils";

/** Skeleton: the classes of each value of each prop (shadcn/cva style). */
export const skeletonVariants = variants({
  base: "animate-pulse bg-muted motion-reduce:animate-none rounded",
  variants: {
    shape: {
      line: "h-4 w-full",
      block: "h-20 w-full",
    },
  },
  defaultVariants: { shape: "line" },
});

export type SkeletonShape = VariantKey<typeof skeletonVariants, "shape">;

/**
 * Skeleton renders a decorative placeholder.
 */
export type SkeletonProps = Omit<
  HTMLAttributes<HTMLDivElement>,
  | "className" | "children" | "shape"
> & {
  shape?: SkeletonShape;
  className?: string;
};

export const Skeleton = forwardRef<HTMLDivElement, SkeletonProps>(function Skeleton(
  { shape, className, ...rest },
  ref
) {
  const cls = cn(skeletonVariants({ shape }), className);
  return (
    <div ref={ref} className={cls || undefined} aria-hidden="true" data-slot="skeleton" {...rest} />
  );
});
Skeleton.displayName = "Skeleton";
