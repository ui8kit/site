/**
 * Title renders a heading (h1–h6) from As (1–6). Defaults to h2 when As is 0
 * or out of range.
 */

import { forwardRef, type ReactNode, type HTMLAttributes, type ElementType, type Ref } from "react";
import { cn, variants, type VariantKey } from "../../utils";
import { titleTag } from "../../utils/expr";
import { Slot } from "../slot/slot";

/** Title: the classes of each value of each prop (shadcn/cva style). */
export const titleVariants = variants({
  base: "text-foreground",
  variants: {
    size: {
      xs: "text-xs",
      sm: "text-sm",
      base: "text-base",
      lg: "text-lg",
      xl: "text-xl",
      "2xl": "text-2xl",
      "3xl": "text-3xl",
      "4xl": "text-4xl",
      "5xl": "text-5xl",
      inherit: "",
    },
    weight: {
      regular: "font-normal",
      medium: "font-medium",
      semibold: "font-semibold",
      bold: "font-bold",
    },
  },
  defaultVariants: { size: "inherit", weight: "semibold" },
});

export type TitleSize = VariantKey<typeof titleVariants, "size">;
export type TitleWeight = VariantKey<typeof titleVariants, "weight">;

/**
 * Title renders the resolved heading element.
 */
export type TitleProps = Omit<
  HTMLAttributes<HTMLElement>,
  | "className" | "children" | "size" | "weight" | "as"
> & {
  size?: TitleSize;
  weight?: TitleWeight;
  as?: 1 | 2 | 3 | 4 | 5 | 6;
  className?: string;
  children?: ReactNode;
  asChild?: boolean;
};

export const Title = forwardRef<HTMLElement, TitleProps>(function Title(
  { size, weight, as, className, children, asChild, ...rest },
  ref
) {
  const cls = cn(titleVariants({ size, weight }), className);
  const resolvedTagValue = titleTag(as);
  const Tag = resolvedTagValue as ElementType;
  if (asChild) {
    return (
      <Slot ref={ref} className={cls} {...rest}>
        {children}
      </Slot>
    );
  }
  return (
    <Tag ref={ref as Ref<HTMLElement>} className={cls || undefined} {...rest}>
      {children}
    </Tag>
  );
});
Title.displayName = "Title";

export type TitleWeight = VariantKey<typeof titleVariants, "weight">;

/**
 * Title renders the resolved heading element.

export type TitleProps = Omit<
  JSX.HTMLAttributes<HTMLElement>,
  "class" | "children" | "size" | "weight" | "as"
> & {
  size?: TitleSize;
  weight?: TitleWeight;
  as?: 1 | 2 | 3 | 4 | 5 | 6;
  class?: string;
  children?: JSX.Element;
};

export const Title: Component<TitleProps> = (props) => {
  const [local, rest] = splitProps(props, ["size", "weight", "as", "class", "children"]);
  const cls = () => cn(titleVariants({ size: local.size, weight: local.weight }), local.class);
  const resolvedTagValue = () => titleTag(local.as);
  return (
    <Dynamic component={resolvedTagValue()} class={cls() || undefined} {...(rest as object)}>
      {local.children}
    </Dynamic>
  );
};

/*
Title renders a heading (h1–h6) from As (1–6). Defaults to h2 when As is 0
or out of range.
*/

package title

import (
	uiutils "github.com/codegen-x/ui8kit/packages/templ/ui/utils"
)

// TitleProps configures Title.
type TitleProps struct {
	// Size — sets size
	Size string
	// Weight — sets weight
	Weight string
	// Class — adds Tailwind utilities
	Class string
	// As — maps to h1–h6; 0 and 2 default to h2
	As uiutils.HeadingLevel
	Attrs templ.Attributes
}

// TitleClasses composes Title classes without rendering —
// the server-composition escape hatch for custom root elements.
func TitleClasses(p TitleProps) string {
	return uiutils.Compose(TitleVariants, map[string]string{
		"size": p.Size,
		"weight": p.Weight,
	}, p.Class)
}

// Title renders the Title contract.
templ Title(p TitleProps) {
	switch uiutils.TitleTag(p.As) {
		case "h1":
			<h1
				class={ TitleClasses(p) }
				{ p.Attrs... }
			>
				{ children... }
			</h1>
		case "h3":
			<h3
				class={ TitleClasses(p) }
				{ p.Attrs... }
			>
				{ children... }
			</h3>
		case "h4":
			<h4
				class={ TitleClasses(p) }
				{ p.Attrs... }
			>
				{ children... }
			</h4>
		case "h5":
			<h5
				class={ TitleClasses(p) }
				{ p.Attrs... }
			>
				{ children... }
			</h5>
		case "h6":
			<h6
				class={ TitleClasses(p) }
				{ p.Attrs... }
			>
				{ children... }
			</h6>
		default:
			<h2
				class={ TitleClasses(p) }
				{ p.Attrs... }
			>
				{ children... }
			</h2>
	}
}
