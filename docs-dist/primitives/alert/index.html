/**
 * Alert displays a static status or informational message.
 * Renders a live region with role=status by default; children carry the message body.
 */

import { forwardRef, type ReactNode, type HTMLAttributes } from "react";
import { cn, variants, type VariantKey } from "../../utils";
import { defaultIfEmpty, oneOfOr } from "../../utils/expr";
import { Slot } from "../slot/slot";

/** Alert: the classes of each value of each prop (shadcn/cva style). */
export const alertVariants = variants({
  base: "flex min-h-8 items-center gap-2 px-2 text-xs font-semibold inset-ring rounded",
  variants: {
    variant: {
      default: "bg-card text-card-foreground inset-ring-border",
      destructive: "bg-muted text-destructive inset-ring-destructive",
      success: "bg-muted text-foreground inset-ring-border",
      warning: "bg-muted text-foreground inset-ring-border",
    },
  },
  defaultVariants: { variant: "default" },
});

export type AlertVariant = VariantKey<typeof alertVariants, "variant">;

/**
 * Alert renders a div with alert or status semantics.
 */
export type AlertProps = Omit<
  HTMLAttributes<HTMLDivElement>,
  | "className" | "children" | "variant" | "role" | "aria-live"
> & {
  variant?: AlertVariant;
  role?: "status" | "alert";
  "aria-live"?: "off" | "polite" | "assertive";
  className?: string;
  children?: ReactNode;
  asChild?: boolean;
};

export const Alert = forwardRef<HTMLDivElement, AlertProps>(function Alert(
  { variant, role, "aria-live": ariaLive, className, children, asChild, ...rest },
  ref
) {
  const cls = cn(alertVariants({ variant }), className);
  if (asChild) {
    return (
      <Slot ref={ref} className={cls} {...rest}>
        {children}
      </Slot>
    );
  }
  return (
    <div
      ref={ref}
      className={cls || undefined}
      data-slot="alert"
      role={oneOfOr(role, "status", "status", "alert")}
      aria-live={defaultIfEmpty(ariaLive, "polite")}
      {...rest}
    >
      {children}
    </div>
  );
});
Alert.displayName = "Alert";

/*
Alert displays a static status or informational message.
Renders a live region with role=status by default; children carry the message body.
*/

package alert

import (
	uiutils "github.com/codegen-x/ui8kit/packages/templ/ui/utils"
)

// AlertProps configures Alert.
type AlertProps struct {
	// Class — adds Tailwind utilities
	Class string
	// Variant — default, destructive, or wireframe preset
	Variant string
	// Role — status or alert (defaults to status)
	Role string
	// AriaLive — polite or assertive (defaults to polite)
	AriaLive string
	Attrs templ.Attributes
}

// AlertClasses composes Alert classes without rendering —
// the server-composition escape hatch for custom root elements.
func AlertClasses(p AlertProps) string {
	return uiutils.Compose(AlertVariants, map[string]string{
		"variant": p.Variant,
	}, p.Class)
}

// Alert renders the Alert contract.
templ Alert(p AlertProps) {
	<div
		class={ AlertClasses(p) }
		data-slot="alert"
		role={ uiutils.OneOfOr(p.Role, "status", "status", "alert") }
		aria-live={ uiutils.DefaultIfEmpty(p.AriaLive, "polite") }
		{ p.Attrs... }
	>
		{ children... }
	</div>
}
