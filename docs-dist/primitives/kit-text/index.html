/**
 * Text is the polymorphic text primitive. `tag` accepts HTML5 text elements
 * only (p, span, h1–h6, blockquote, code, …); anything else falls back to p.
 */

import { forwardRef, type ReactNode, type HTMLAttributes, type ElementType, type Ref } from "react";
import { textClasses } from "./text.shared";
import { safeRest } from "../../utils/expr";
import { resolveTag, TagGroup } from "../../utils/tags";
import { Slot } from "../slot/slot";

/**
 * Text renders the requested text element (default p).
 */
export type TextProps = Omit<HTMLAttributes<HTMLElement>, "className" | "children" | "tag"> & {
  tag?: "p" | "span" | "h1" | "h2" | "h3" | "h4" | "h5" | "h6" | "blockquote" | "figcaption" | "pre" | "code" | "small" | "strong" | "em" | "time" | "legend" | "del" | "ins";
  className?: string;
  children?: ReactNode;
  asChild?: boolean;
};

export const Text = forwardRef<HTMLElement, TextProps>(function Text(
  { tag, className, children, asChild, ...rawRest },
  ref
) {
  const rest = safeRest(rawRest);
  const cls = textClasses({ className });
  const resolvedTagValue = resolveTag(tag, "p", TagGroup.Text);
  const Tag = resolvedTagValue as ElementType;
  if (asChild) {
    return (
      <Slot ref={ref} className={cls} {...rest}>
        {children}
      </Slot>
    );
  }
  return (
    <Tag ref={ref as Ref<HTMLElement>} className={cls || undefined} {...rest}>
      {children}
    </Tag>
  );
});
Text.displayName = "Text";

/**
 * Text is the polymorphic text primitive. `tag` accepts HTML5 text elements
 * only (p, span, h1–h6, blockquote, code, …); anything else falls back to p.
 */

import { splitProps, type JSX, type Component } from "solid-js";
import { Dynamic } from "solid-js/web";
import { textClasses } from "./text.shared";
import { safeRest } from "../../utils/expr";
import { resolveTag, TagGroup } from "../../utils/tags";

/**
 * Text renders the requested text element (default p).
 */
export type TextProps = Omit<JSX.HTMLAttributes<HTMLElement>, "class" | "className" | "children" | "tag"> & {
  tag?: "p" | "span" | "h1" | "h2" | "h3" | "h4" | "h5" | "h6" | "blockquote" | "figcaption" | "pre" | "code" | "small" | "strong" | "em" | "time" | "legend" | "del" | "ins";
  class?: string;
  className?: string;
  children?: JSX.Element;
};

export const Text: Component<TextProps> = (props) => {
  const [local, rest] = splitProps(props, ["tag", "class", "className", "children"]);
  const resolvedTagValue = () => resolveTag(local.tag, "p", TagGroup.Text);
  const cls = () => textClasses({ className: local.class ?? local.className });
  return (
    <Dynamic component={resolvedTagValue()} {...(cls() ? { class: cls() } : {})} {...safeRest(rest)}>
      {local.children}
    </Dynamic>
  );
};
