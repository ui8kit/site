/**
 * Switch is an on/off toggle: button[role=switch] with aria-checked.
 * State changes are the consumer's (onClick via attrs/rest).
 */

import { forwardRef, type ReactNode, type ButtonHTMLAttributes } from "react";
import { switchClasses } from "./switch.shared";
import { boolStr, safeRest } from "../../utils/expr";

/**
 * Switch renders button[type=button][role=switch]; children render the thumb.
 */
export type SwitchProps = Omit<ButtonHTMLAttributes<HTMLButtonElement>, "className" | "children" | "checked" | "disabled"> & {
  checked?: boolean;
  disabled?: boolean;
  className?: string;
  children?: ReactNode;
};

export const Switch = forwardRef<HTMLButtonElement, SwitchProps>(function Switch(
  { checked, disabled, className, children, ...rawRest },
  ref
) {
  const rest = safeRest(rawRest, { blank: ["id", "name", "aria-label", "aria-labelledby"] });
  const cls = switchClasses({ className });
  return (
    <button ref={ref} type="button" role="switch" aria-checked={boolStr(checked)} data-state={boolStr(checked)} className={cls || undefined} disabled={disabled} {...rest}>
      {children}
    </button>
  );
});
Switch.displayName = "Switch";

/**
 * Switch is an on/off toggle: button[role=switch] with aria-checked.
 * State changes are the consumer's (onClick via attrs/rest).
 */

import { splitProps, type JSX, type Component } from "solid-js";
import { switchClasses } from "./switch.shared";
import { boolStr, safeRest } from "../../utils/expr";

/**
 * Switch renders button[type=button][role=switch]; children render the thumb.
 */
export type SwitchProps = Omit<JSX.ButtonHTMLAttributes<HTMLButtonElement>, "class" | "className" | "children" | "checked" | "disabled"> & {
  checked?: boolean;
  disabled?: boolean;
  class?: string;
  className?: string;
  children?: JSX.Element;
};

export const Switch: Component<SwitchProps> = (props) => {
  const [local, rest] = splitProps(props, ["checked", "disabled", "class", "className", "children"]);
  const cls = () => switchClasses({ className: local.class ?? local.className });
  return (
    <button type="button" role="switch" aria-checked={boolStr(local.checked)} data-state={boolStr(local.checked)} {...(cls() ? { class: cls() } : {})} disabled={local.disabled || undefined} {...safeRest(rest, { blank: ["id", "name", "aria-label", "aria-labelledby"] })}>
      {local.children}
    </button>
  );
};
