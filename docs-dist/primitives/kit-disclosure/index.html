 * Disclosure is native details/summary. `name` groups exclusive accordions.

import { forwardRef, type ReactNode, type DetailsHTMLAttributes, type HTMLAttributes } from "react";
import { disclosureClasses, summaryClasses } from "./disclosure.shared";
import { safeRest } from "../../utils/expr";

/**
 * Disclosure renders a native details element.
 */
export type DisclosureProps = Omit<DetailsHTMLAttributes<HTMLDetailsElement>, "className" | "children" | "open"> & {
  open?: boolean;
  className?: string;
  children?: ReactNode;
};

export const Disclosure = forwardRef<HTMLDetailsElement, DisclosureProps>(function Disclosure(
  { open, className, children, ...rawRest },
  ref
) {
  const rest = safeRest(rawRest, { blank: ["name"] });
  const cls = disclosureClasses({ className });
  return (
    <details ref={ref} className={cls || undefined} open={open} {...rest}>
      {children}
    </details>
  );
});
Disclosure.displayName = "Disclosure";

 * Summary renders the native summary (the disclosure trigger).

  children?: ReactNode;

  ref

import { splitProps, type JSX, type Component } from "solid-js";
import { disclosureClasses, summaryClasses } from "./disclosure.shared";
import { safeRest } from "../../utils/expr";

/**
 * Disclosure renders a native details element.
 */
export type DisclosureProps = Omit<JSX.DetailsHtmlAttributes<HTMLDetailsElement>, "class" | "className" | "children" | "open"> & {
  open?: boolean;
  class?: string;
  className?: string;
  children?: JSX.Element;
};

export const Disclosure: Component<DisclosureProps> = (props) => {
  const [local, rest] = splitProps(props, ["open", "class", "className", "children"]);
  const cls = () => disclosureClasses({ className: local.class ?? local.className });
  return (
    <details {...(cls() ? { class: cls() } : {})} open={local.open || undefined} {...safeRest(rest, { blank: ["name"] })}>
      {local.children}
    </details>
  );
};

 * Summary renders the native summary (the disclosure trigger).

  class?: string;
  className?: string;
  children?: JSX.Element;
};

  const [local, rest] = splitProps(props, ["class", "className", "children"]);

/*
Disclosure is native details/summary. `name` groups exclusive accordions.
*/

package disclosure

import (
	uiutils "github.com/ui8kit/codegen/generated/templ/utils"
)

// DisclosureProps configures Disclosure.
type DisclosureProps struct {
	// Class — token class names supplied by the consumer
	Class string
	// Open — initial open state
	Open bool
	Name string
	Attrs templ.Attributes
}

// DisclosureClasses composes Disclosure classes without rendering —
// the server-composition escape hatch for custom root elements (no asChild in templ).
func DisclosureClasses(p DisclosureProps) string {
	return uiutils.Cn(p.Class)
}

// Disclosure renders the Disclosure contract.
templ Disclosure(p DisclosureProps) {
	<details
		if DisclosureClasses(p) != "" {
			class={ DisclosureClasses(p) }
		}
		if !uiutils.HasAttr(p.Attrs, "open") {
			open?={ p.Open }
		}
		if !uiutils.HasAttr(p.Attrs, "name") {
			if uiutils.IsSetStr(p.Name) {
				name={ p.Name }
			}
		}
		{ uiutils.RestAttrs(p.Attrs, false, "name")... }
	>
		{ children... }
	</details>
}

// SummaryProps configures Summary.
type SummaryProps struct {
	// Class — token class names supplied by the consumer
	Class string
	Attrs templ.Attributes
}

// SummaryClasses composes Summary classes without rendering —
// the server-composition escape hatch for custom root elements (no asChild in templ).
func SummaryClasses(p SummaryProps) string {
	return uiutils.Cn(p.Class)
}

// Summary renders the Summary contract.
templ Summary(p SummaryProps) {
	<summary
		if SummaryClasses(p) != "" {
			class={ SummaryClasses(p) }
		}
		{ uiutils.RestAttrs(p.Attrs, false)... }
	>
		{ children... }
	</summary>
}
