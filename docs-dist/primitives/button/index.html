/**
 * Button renders a button element, or an anchor when Href is set.
 * Type/Form apply only in button mode; Disabled on anchors uses aria-disabled.
 */
export type ButtonProps = Omit<
  HTMLAttributes<HTMLElement>,
  | "className" | "children" | "variant" | "size" | "layout" | "type" | "href" | "target" | "rel"
  | "external" | "disabled"
> & {
  variant?: ButtonVariant;
  size?: ButtonSize;
  layout?: ButtonLayout;
  type?: string;
  href?: string;
  target?: string;
  rel?: string;
  external?: boolean;
  disabled?: boolean;
  className?: string;
  children?: ReactNode;
  asChild?: boolean;
};

export const Button = forwardRef<HTMLElement, ButtonProps>(function Button(
  { variant, size, layout, type, href, target, rel, external, disabled, className, children, asChild, ...rest },
  ref
) {
  const cls = cn(
    buttonVariants({ variant, size, layout }),
    (disabled ? "pointer-events-none opacity-50" : ""),
    className
  );
  const targetValue = (isSetStr(target) ? target : (external ? "_blank" : ""));
  const relValue = (isSetStr(rel) ? rel : ((external || eqStr(targetValue, "_blank")) ? "noopener noreferrer" : ""));
  if (asChild) {
    return (
      <Slot ref={ref} className={cls} {...rest}>
        {children}
      </Slot>
    );
  }
  return (
    isSetStr(href) ? (
      <a
        ref={ref as Ref<HTMLElementTagNameMap["a"]>}
        href={orUndef(href)}
        className={cls || undefined}
        data-slot="button"
        target={orUndef(targetValue)}
        rel={orUndef(relValue)}
        aria-disabled={disabled ? "true" : undefined}
        tabIndex={disabled ? -1 : undefined}
        {...rest}
      >
        {children}
      </a>
    ) : (
      <button
        ref={ref as Ref<HTMLElementTagNameMap["button"]>}
        type={oneOfOr(type, "button", "submit", "reset")}
        className={cls || undefined}
        data-slot="button"
        disabled={disabled || undefined}
        {...rest}
      >
        {children}
      </button>
    )
  );
});
Button.displayName = "Button";
