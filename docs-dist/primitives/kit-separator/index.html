/**
 * Separator is the native <hr> (implicit role=separator). Vertical adds
 * aria-orientation; Decorative removes the semantics with role=none.
 */

import { forwardRef, type HTMLAttributes } from "react";
import { separatorClasses } from "./separator.shared";
import { eqStr, oneOfOrLower, safeRest } from "../../utils/expr";

/**
 * Separator renders a void hr; no children.
 */
export type SeparatorProps = Omit<HTMLAttributes<HTMLHRElement>, "className" | "children" | "orientation" | "decorative"> & {
  orientation?: "horizontal" | "vertical";
  decorative?: boolean;
  className?: string;
};

export const Separator = forwardRef<HTMLHRElement, SeparatorProps>(function Separator(
  { orientation, decorative, className, ...rawRest },
  ref
) {
  const rest = safeRest(rawRest);
  const cls = separatorClasses({ className });
  return (
    <hr ref={ref} className={cls || undefined} role={decorative ? "none" : undefined} aria-orientation={(!(decorative) && eqStr(oneOfOrLower(orientation, "horizontal", "vertical"), "vertical")) ? "vertical" : undefined} {...rest} />
  );
});

/*
Separator is the native <hr> (implicit role=separator). Vertical adds
aria-orientation; Decorative removes the semantics with role=none.
*/

package separator

import (
	uiutils "github.com/ui8kit/codegen/generated/templ/utils"
)

// SeparatorProps configures Separator.
type SeparatorProps struct {
	// Class — token class names supplied by the consumer
	Class string
	// Orientation — horizontal (default) | vertical
	Orientation string
	// Decorative — purely visual: role=none, no separator semantics
	Decorative bool
	Attrs templ.Attributes
}

// SeparatorClasses composes Separator classes without rendering —
// the server-composition escape hatch for custom root elements (no asChild in templ).
func SeparatorClasses(p SeparatorProps) string {
	return uiutils.Cn(p.Class)
}

// Separator renders the Separator contract.
templ Separator(p SeparatorProps) {
	<hr
		if SeparatorClasses(p) != "" {
			class={ SeparatorClasses(p) }
		}
		if !uiutils.HasAttr(p.Attrs, "role") {
			if p.Decorative {
				role={ "none" }
			}
		}
		if !uiutils.HasAttr(p.Attrs, "aria-orientation") {
			if (!p.Decorative && (uiutils.OneOfOrLower(p.Orientation, "horizontal", "vertical") == "vertical")) {
				aria-orientation={ "vertical" }
			}
		}
		{ uiutils.RestAttrs(p.Attrs, false)... }
	/>
}
