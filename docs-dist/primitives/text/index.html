/**
 * Text renders a paragraph of body copy. Use Inline for inline span copy
 * and Title (as 1–6) for headings.
 */

import { forwardRef, type ReactNode, type HTMLAttributes } from "react";
import { cn, variants, type VariantKey } from "../../utils";
import { Slot } from "../slot/slot";

/** Text: the classes of each value of each prop (shadcn/cva style). */
export const textVariants = variants({
  variants: {
    size: {
      xs: "text-xs",
      sm: "text-sm",
      base: "text-base",
      lg: "text-lg",
      xl: "text-xl",
      "2xl": "text-2xl",
      "3xl": "text-3xl",
      "4xl": "text-4xl",
      "5xl": "text-5xl",
      inherit: "",
    },
    weight: {
      regular: "font-normal",
      medium: "font-medium",
      semibold: "font-semibold",
      bold: "font-bold",
      inherit: "",
    },
  },
  defaultVariants: { size: "inherit", weight: "inherit" },
});

export type TextSize = VariantKey<typeof textVariants, "size">;
export type TextWeight = VariantKey<typeof textVariants, "weight">;

/**
 * Text renders a p element with body copy classes.
 */
export type TextProps = Omit<
  HTMLAttributes<HTMLParagraphElement>,
  | "className" | "children" | "size" | "weight"
> & {
  size?: TextSize;
  weight?: TextWeight;
  className?: string;
  children?: ReactNode;
  asChild?: boolean;
};

export const Text = forwardRef<HTMLParagraphElement, TextProps>(function Text(
  { size, weight, className, children, asChild, ...rest },
  ref
) {
  const cls = cn(textVariants({ size, weight }), className);
  if (asChild) {
    return (
      <Slot ref={ref} className={cls} {...rest}>
        {children}
      </Slot>
    );
  }
  return (
    <p ref={ref} className={cls || undefined} {...rest}>
      {children}
    </p>
  );
});
Text.displayName = "Text";
