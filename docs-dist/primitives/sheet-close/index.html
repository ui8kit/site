package compose

import (
	"github.com/ui8kit/codegen/generated/templ/ui/button"
	"github.com/ui8kit/codegen/generated/templ/ui/icon"
)

// SheetCloseProps: Button + Lucide icons (data-ui8kit="sheet-close").
type SheetCloseProps struct {
	Variant   string
	Size      string
	AriaLabel string
	button.ButtonProps
}

func sheetCloseInner(p SheetCloseProps) button.ButtonProps {
	in := p.ButtonProps
	in.Type = "button"
	in.Class = Join(Rc("sheettrigger", map[string]string{"variant": p.Variant, "size": p.Size}), p.ButtonProps.Class)
	fixed := map[string]string{"data-ui8kit": "sheet-close", "aria-label": Or(p.AriaLabel, "Close menu")}
	in.Attrs = WithAttrs(in.Attrs, fixed)
	return in
}

// SheetClose renders the button with its Lucide icons.
templ SheetClose(p SheetCloseProps) {
	@button.Button(sheetCloseInner(p)) {
		@icon.Icon(icon.IconProps{Class: "size-5"}) {
			@templ.Raw(lucideX)
		}
	}
}
