/**
 * LucideIcon: a Lucide glyph (ISC license) by kebab-case name (any glyph of the `lucide` package, through ./lucide-icons), inside the Icon primitive
 * (the kit-compose LucideIcon frame: span[data-slot=icon]). Decorative (the frame is aria-hidden) unless a `label` is given.
 */
import { createElement } from "react";
import { LucideIcon as Frame } from "../compose";
import { lucideNodes } from "./lucide-icons";

export type LucideIconProps = { name: string; size?: number; strokeWidth?: number; label?: string; className?: string };

export function LucideIcon({ name, size = 16, strokeWidth = 2, label, className }: LucideIconProps) {
  const nodes = lucideNodes(name) ?? lucideNodes("circle") ?? [];
  return (
    <Frame label={label} className={className} data-lucide={name}>
      <svg xmlns="http://www.w3.org/2000/svg" viewBox="0 0 24 24" width={size} height={size} fill="none" stroke="currentColor"

        {nodes.map(([tag, attrs], i) => createElement(tag, { key: i, ...attrs }))}
      </svg>
    </Frame>
  );
}

// LucideIcon: a Lucide glyph (ISC license) by kebab-case name from the registry next to this file (lucide_icons.go), inside
// the Icon primitive (the kit-compose LucideIcon frame: span[data-slot=icon]). Decorative (the frame is aria-hidden) unless a
// Label is given. Size and StrokeWidth default to 16 and 2.

package icons

import (
	"html"
	"sort"
	"strconv"
	"strings"

	"github.com/ui8kit/codegen/generated/templ/compose"
	"github.com/ui8kit/codegen/generated/templ/ui/icon"
)

// LucideIconProps configures LucideIcon.
type LucideIconProps struct {
	Name        string
	Size        int
	StrokeWidth int
	Label       string
	Class       string
	Attrs       templ.Attributes
}

func orInt(v, d int) string {
	if v == 0 {
		return strconv.Itoa(d)
	}
	return strconv.Itoa(v)
}

// glyph is the inner markup of the icon: its nodes with their attributes in name order, escaped.
func glyph(name string) string {
	nodes, ok := lucide[name]
	if !ok {
		nodes = lucide["circle"]
	}
	var b strings.Builder
	for _, n := range nodes {
		keys := make([]string, 0, len(n.Attrs))
		for k := range n.Attrs {
			keys = append(keys, k)
		}
		sort.Strings(keys)
		b.WriteString("<" + n.Tag)
		for _, k := range keys {
			b.WriteString(" " + k + "=\"" + html.EscapeString(n.Attrs[k]) + "\"")
		}
		b.WriteString("></" + n.Tag + ">")
	}
	return b.String()
}

func frameAttrs(a templ.Attributes, name string) templ.Attributes {
	out := templ.Attributes{}
	for k, v := range a {
		out[k] = v
	}
	if name != "" {
		out["data-lucide"] = name
	}
	return out
}

// LucideIcon renders the glyph as an inline svg inside the Icon primitive.
templ LucideIcon(p LucideIconProps) {
	@compose.LucideIcon(compose.LucideIconProps{IconProps: icon.IconProps{Label: p.Label, Class: p.Class, Attrs: frameAttrs(p.Attrs, p.Name)}}) {
		<svg
			xmlns="http://www.w3.org/2000/svg"
			viewBox="0 0 24 24"
			width={ orInt(p.Size, 16) }
			height={ orInt(p.Size, 16) }
			fill="none"
			stroke="currentColor"
			stroke-width={ orInt(p.StrokeWidth, 2) }
			stroke-linecap="round"
			stroke-linejoin="round"
			aria-hidden="true"
		>
			@templ.Raw(glyph(p.Name))
		</svg>
	}
}
